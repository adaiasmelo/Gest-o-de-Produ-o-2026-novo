
import React, { useState, useMemo, useEffect } from 'react';
import { Menu, Plus, TrendingUp, BarChart2, Calendar, RotateCcw, Download, Clock, Zap, Trash2, Cpu, BrainCircuit, Sparkles, CheckCircle2, AlertTriangle, X, ChevronRight, Settings, Target, Users, UserPlus, Trophy, Sparkle, RefreshCw } from 'lucide-react';
import { BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, Cell, PieChart, Pie, Legend } from 'recharts';
import { GoogleGenAI } from "@google/genai";
import { ProductionEntry } from './types';
import { INITIAL_DATA, GOAL_VALUE as DEFAULT_GOAL } from './constants';
import LaunchModal from './components/LaunchModal';
import StatsCard from './components/StatsCard';

const App: React.FC = () => {
  const [productionData, setProductionData] = useState<ProductionEntry[]>(() => {
    const saved = localStorage.getItem('manupackaging_data');
    return saved ? JSON.parse(saved) : INITIAL_DATA;
  });

  const [goalValue, setGoalValue] = useState<number>(() => {
    const savedGoal = localStorage.getItem('manupackaging_goal');
    return savedGoal ? parseFloat(savedGoal) : DEFAULT_GOAL;
  });

  const [operators, setOperators] = useState<string[]>(() => {
    const savedOps = localStorage.getItem('manupackaging_operators');
    return savedOps ? JSON.parse(savedOps) : ['Franck', 'Sidoneide', 'Deyvis'];
  });

  const [newOperatorName, setNewOperatorName] = useState('');
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isMenuOpen, setIsMenuOpen] = useState(false);
  const [isAiLoading, setIsAiLoading] = useState(false);
  const [aiInsights, setAiInsights] = useState<string | null>(null);
  
  const [startDate, setStartDate] = useState<string>('');
  const [endDate, setEndDate] = useState<string>('');
  const [selectedMachine, setSelectedMachine] = useState<string>('Todas');

  useEffect(() => {
    localStorage.setItem('manupackaging_data', JSON.stringify(productionData));
  }, [productionData]);

  useEffect(() => {
    localStorage.setItem('manupackaging_goal', goalValue.toString());
  }, [goalValue]);

  useEffect(() => {
    localStorage.setItem('manupackaging_operators', JSON.stringify(operators));
  }, [operators]);

  const filteredData = useMemo(() => {
    return productionData.filter(entry => {
      const entryDate = entry.date;
      const matchesStart = !startDate || entryDate >= startDate;
      const matchesEnd = !endDate || entryDate <= endDate;
      const matchesMachine = selectedMachine === 'Todas' || entry.machine === selectedMachine;
      return matchesStart && matchesEnd && matchesMachine;
    });
  }, [productionData, startDate, endDate, selectedMachine]);

  const stats = useMemo(() => {
    const s = filteredData.reduce((acc, curr) => ({
      totalNet: acc.totalNet + curr.netWeight,
      totalGross: acc.totalGross + curr.grossWeight,
      ecoA: acc.ecoA + curr.ecoA,
      ecoBP: acc.ecoBP + curr.ecoBP,
      ecoBM: acc.ecoBM + curr.ecoBM,
      borra: acc.borra + curr.borraTotal,
      paradas: acc.paradas + curr.manutencaoMin + curr.processoMin + curr.outrosMin,
      manut: acc.manut + curr.manutencaoMin,
      proc: acc.proc + curr.processoMin,
      outros: acc.outros + curr.outrosMin,
    }), { totalNet: 0, totalGross: 0, ecoA: 0, ecoBP: 0, ecoBM: 0, borra: 0, paradas: 0, manut: 0, proc: 0, outros: 0 });
    
    const qualityEfficiency = s.totalGross > 0 ? (s.totalNet / s.totalGross) * 100 : 0;
    return { ...s, qualityEfficiency };
  }, [filteredData]);

  const isGoalAchieved = stats.totalNet >= goalValue && goalValue > 0;

  const machineStatus = useMemo(() => {
    const machines = ['Cast 1', 'Cast 2'];
    return machines.map(m => {
      const lastEntry = [...productionData].reverse().find(e => e.machine === m);
      const totalDowntime = lastEntry ? (lastEntry.manutencaoMin + lastEntry.processoMin + lastEntry.outrosMin) : 0;
      
      let status = 'Operando';
      let color = 'emerald';
      
      if (!lastEntry) { status = 'Sem Dados'; color = 'slate'; }
      else if (lastEntry.manutencaoMin > 30) { status = 'Manutenção'; color = 'orange'; }
      else if (totalDowntime > 60) { status = 'Alerta'; color = 'amber'; }

      return { name: m, status, color, lastOp: lastEntry?.operator || '---' };
    });
  }, [productionData]);

  const generateAiInsights = async () => {
    setIsAiLoading(true);
    try {
      const ai = new GoogleGenAI({ apiKey: process.env.API_KEY });
      const prompt = `Analise os seguintes dados de produção industrial e forneça 3 insights rápidos e acionáveis:
        Dados: Produção Líquida Total: ${stats.totalNet}kg, Eficiência: ${stats.qualityEfficiency.toFixed(1)}%, 
        Total Paradas: ${stats.paradas}min (Manutenção: ${stats.manut}m, Processo: ${stats.proc}m).
        Considere que a meta é ${goalValue}kg. Responda de forma curta e profissional.`;

      const response = await ai.models.generateContent({
        model: 'gemini-3-flash-preview',
        contents: prompt,
      });

      setAiInsights(response.text || "Não foi possível gerar insights no momento.");
    } catch (error) {
      setAiInsights("Erro ao conectar com o analista de IA.");
    } finally {
      setIsAiLoading(false);
    }
  };

  const handleSaveEntry = (newEntry: Omit<ProductionEntry, 'id'>) => {
    const entryWithId: ProductionEntry = {
      ...newEntry,
      id: Math.random().toString(36).substr(2, 9),
    };
    setProductionData(prev => [...prev, entryWithId]);
  };

  const handleDeleteEntry = (id: string) => {
    if (confirm('Deseja realmente excluir este lançamento?')) {
      setProductionData(prev => prev.filter(e => e.id !== id));
    }
  };

  const handleAddOperator = (e: React.FormEvent) => {
    e.preventDefault();
    if (newOperatorName.trim() && !operators.includes(newOperatorName.trim())) {
      setOperators(prev => [...prev, newOperatorName.trim()].sort());
      setNewOperatorName('');
    }
  };

  const handleRemoveOperator = (name: string) => {
    if (confirm(`Remover operador ${name}?`)) {
      setOperators(prev => prev.filter(op => op !== name));
    }
  };

  const handleResetData = () => {
    if (confirm('Deseja resetar todos os dados para os valores iniciais (Novembro/Dezembro)? Isso apagará seus lançamentos manuais.')) {
      setProductionData(INITIAL_DATA);
      setGoalValue(DEFAULT_GOAL);
      setIsMenuOpen(false);
    }
  };

  const downtimePieData = useMemo(() => [
    { name: 'Manutenção', value: stats.manut, color: '#f97316' },
    { name: 'Processo', value: stats.proc, color: '#f59e0b' },
    { name: 'Outros', value: stats.outros, color: '#94a3b8' }
  ].filter(d => d.value > 0), [stats]);

  const operatorPerformanceData = useMemo(() => {
    const performanceMap: Record<string, number> = {};
    filteredData.forEach(entry => {
      performanceMap[entry.operator] = (performanceMap[entry.operator] || 0) + entry.netWeight;
    });
    return Object.entries(performanceMap)
      .map(([name, kg]) => ({ name, kg }))
      .sort((a, b) => b.kg - a.kg);
  }, [filteredData]);

  return (
    <div className="min-h-screen bg-slate-50 pb-20 overflow-x-hidden">
      {/* Sidebar Menu */}
      <div className={`fixed inset-0 z-50 transition-all duration-300 ${isMenuOpen ? 'visible' : 'invisible'}`}>
        <div className={`absolute inset-0 bg-slate-900/40 backdrop-blur-sm transition-opacity duration-300 ${isMenuOpen ? 'opacity-100' : 'opacity-0'}`} onClick={() => setIsMenuOpen(false)} />
        <div className={`absolute right-0 top-0 bottom-0 w-80 bg-white shadow-2xl transition-transform duration-300 transform p-6 flex flex-col gap-6 ${isMenuOpen ? 'translate-x-0' : 'translate-x-full'} overflow-y-auto`}>
          <div className="flex items-center justify-between">
            <h2 className="text-lg font-black text-slate-800 flex items-center gap-2">
              <Settings size={20} className="text-blue-600" /> OPÇÕES
            </h2>
            <button onClick={() => setIsMenuOpen(false)} className="p-2 hover:bg-slate-100 rounded-full transition-colors text-slate-400">
              <X size={24} />
            </button>
          </div>

          {/* Edit Goal */}
          <div className="space-y-3">
            <h3 className="text-[10px] font-black text-slate-400 uppercase tracking-widest flex items-center gap-2">
              <Target size={14} className="text-emerald-500" /> Meta de Produção
            </h3>
            <div className="space-y-1">
              <input 
                type="number" 
                value={goalValue} 
                onChange={(e) => setGoalValue(parseFloat(e.target.value) || 0)} 
                className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2 text-sm font-black text-blue-600 focus:outline-none focus:ring-2 focus:ring-blue-500/10"
              />
            </div>
          </div>

          {/* Manage Operators */}
          <div className="space-y-4 pt-4 border-t border-slate-100">
            <h3 className="text-[10px] font-black text-slate-400 uppercase tracking-widest flex items-center gap-2">
              <Users size={14} className="text-blue-500" /> Gerenciar Operadores
            </h3>
            <form onSubmit={handleAddOperator} className="flex gap-2">
              <input 
                type="text" 
                placeholder="Novo nome..." 
                value={newOperatorName}
                onChange={(e) => setNewOperatorName(e.target.value)}
                className="flex-1 bg-slate-50 border border-slate-200 rounded-xl px-3 py-2 text-xs font-bold text-slate-700 focus:outline-none focus:ring-2 focus:ring-blue-500/10"
              />
              <button type="submit" className="p-2 bg-blue-600 text-white rounded-xl hover:bg-blue-700 transition-all">
                <UserPlus size={16} />
              </button>
            </form>
            <div className="max-h-32 overflow-y-auto space-y-1 pr-1 custom-scrollbar">
              {operators.map(op => (
                <div key={op} className="flex items-center justify-between bg-slate-50 px-3 py-2 rounded-lg group">
                  <span className="text-xs font-bold text-slate-700">{op}</span>
                  <button onClick={() => handleRemoveOperator(op)} className="text-slate-300 hover:text-red-500 transition-colors opacity-0 group-hover:opacity-100">
                    <Trash2 size={12} />
                  </button>
                </div>
              ))}
            </div>
          </div>

          {/* Search by Period */}
          <div className="space-y-4 pt-4 border-t border-slate-100">
            <h3 className="text-[10px] font-black text-slate-400 uppercase tracking-widest flex items-center gap-2">
              <Calendar size={14} /> Filtro por Período
            </h3>
            <div className="space-y-3">
              <div className="space-y-1">
                <label className="text-[10px] font-bold text-slate-500">DATA INICIAL</label>
                <input type="date" value={startDate} onChange={(e) => setStartDate(e.target.value)} className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2 text-sm font-bold text-slate-700 focus:outline-none focus:ring-2 focus:ring-blue-500/10" />
              </div>
              <div className="space-y-1">
                <label className="text-[10px] font-bold text-slate-500">DATA FINAL</label>
                <input type="date" value={endDate} onChange={(e) => setEndDate(e.target.value)} className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2 text-sm font-bold text-slate-700 focus:outline-none focus:ring-2 focus:ring-blue-500/10" />
              </div>
              <button 
                onClick={() => { setStartDate(''); setEndDate(''); }} 
                className="w-full flex items-center justify-center gap-2 py-2 text-xs font-black text-slate-400 hover:text-blue-600 transition-colors uppercase"
              >
                <RotateCcw size={14} /> Limpar Filtros
              </button>
            </div>
          </div>

          {/* AI Analyst in Menu */}
          <div className="space-y-4 pt-4 border-t border-slate-100">
            <div className="flex items-center justify-between">
              <h3 className="text-[10px] font-black text-purple-600 uppercase tracking-widest flex items-center gap-2">
                <BrainCircuit size={14} /> Analista de IA
              </h3>
              <button 
                onClick={generateAiInsights}
                disabled={isAiLoading}
                className="text-xs font-black text-blue-600 flex items-center gap-1 hover:underline disabled:opacity-50"
              >
                {isAiLoading ? 'Processando...' : 'Atualizar'}
              </button>
            </div>
            
            {aiInsights ? (
              <div className="bg-purple-50 rounded-2xl p-4 border border-purple-100 text-[11px] text-purple-800 leading-relaxed italic animate-in slide-in-from-right-2">
                "{aiInsights}"
              </div>
            ) : (
              <div className="bg-slate-50 rounded-2xl p-6 text-center border border-dashed border-slate-200">
                <Sparkles size={24} className="mx-auto text-slate-300 mb-2" />
                <p className="text-[10px] text-slate-400 font-bold uppercase">Pronto para analisar seus dados</p>
                <button 
                  onClick={generateAiInsights}
                  className="mt-4 px-4 py-2 bg-slate-900 text-white rounded-lg text-[10px] font-black uppercase tracking-widest hover:bg-black transition-all"
                >
                  Gerar Insights Agora
                </button>
              </div>
            )}
          </div>

          <div className="mt-auto space-y-2 pb-6">
             <button onClick={exportToCSV} className="w-full bg-blue-50 text-blue-600 p-4 rounded-2xl font-black text-[10px] uppercase tracking-widest flex items-center justify-between hover:bg-blue-100 transition-all">
                Exportar CSV <Download size={16} />
             </button>
             <button onClick={handleResetData} className="w-full bg-slate-100 text-slate-500 p-4 rounded-2xl font-black text-[10px] uppercase tracking-widest flex items-center justify-between hover:bg-slate-200 transition-all mt-2">
                Resetar p/ Fake Data <RefreshCw size={16} />
             </button>
          </div>
        </div>
      </div>

      <header className="bg-white px-6 py-4 flex items-center justify-between sticky top-0 z-40 border-b border-slate-100">
        <h1 className="text-xl font-bold text-slate-800 flex items-center gap-2">
          <div className="w-8 h-8 bg-blue-600 rounded-lg flex items-center justify-center text-white">
            <Cpu size={18} />
          </div>
          <span className="tracking-tight">Manupackaging <span className="text-blue-600">Pro</span></span>
        </h1>
        <div className="flex items-center gap-2">
           <button onClick={() => setIsModalOpen(true)} className="bg-blue-600 hover:bg-blue-700 text-white font-bold px-4 py-2 rounded-lg flex items-center gap-2 shadow-lg shadow-blue-200 transition-all active:scale-95 text-sm">
              <Plus size={18} />
              <span className="hidden sm:inline">Novo</span>
           </button>
           <button onClick={() => setIsMenuOpen(true)} className="p-2 hover:bg-slate-100 rounded-lg text-slate-600 transition-colors">
             <Menu size={24} />
           </button>
        </div>
      </header>

      <main className="max-w-4xl mx-auto p-4 space-y-6">
        
        {/* Goal Achieved Alert */}
        {isGoalAchieved && (
          <div className="bg-emerald-600 text-white p-4 rounded-2xl shadow-xl shadow-emerald-200 flex items-center justify-between animate-bounce">
            <div className="flex items-center gap-4">
              <div className="w-12 h-12 bg-white/20 rounded-full flex items-center justify-center">
                <Trophy size={28} className="text-emerald-100" />
              </div>
              <div>
                <h2 className="font-black text-sm uppercase tracking-widest">Meta Atingida!</h2>
                <p className="text-[11px] font-medium text-emerald-100">Excelente performance. A meta global foi superada!</p>
              </div>
            </div>
            <div className="flex gap-1">
              <Sparkle size={16} className="text-yellow-400 animate-pulse" />
              <Sparkle size={16} className="text-yellow-200 animate-ping" />
            </div>
          </div>
        )}

        {/* Machine Status Trackers */}
        <div className="grid grid-cols-2 gap-4">
          {machineStatus.map((m) => (
            <div key={m.name} className="bg-white rounded-2xl p-4 shadow-sm border border-slate-100 flex items-center gap-4">
              <div className={`w-10 h-10 rounded-full flex items-center justify-center bg-${m.color}-100 text-${m.color}-600`}>
                {m.status === 'Operando' ? <CheckCircle2 size={20} /> : <AlertTriangle size={20} />}
              </div>
              <div>
                <span className="text-[10px] font-black text-slate-400 uppercase tracking-widest">{m.name}</span>
                <div className="flex items-center gap-2">
                  <span className="text-sm font-bold text-slate-700">{m.status}</span>
                  <div className={`w-2 h-2 rounded-full bg-${m.color}-500 animate-pulse`}></div>
                </div>
                <span className="text-[9px] text-slate-400 italic">Último op: {m.lastOp}</span>
              </div>
            </div>
          ))}
        </div>

        {/* Dynamic Filters - Machine Selector Only now */}
        <div className="bg-white rounded-2xl p-2 shadow-sm border border-slate-200 flex justify-center">
           <div className="flex bg-slate-100 p-1 rounded-xl w-full">
              {['Todas', 'Cast 1', 'Cast 2'].map(m => (
                <button 
                  key={m} onClick={() => setSelectedMachine(m)}
                  className={`flex-1 px-4 py-2 rounded-lg text-[10px] font-black uppercase tracking-tight transition-all ${selectedMachine === m ? 'bg-white text-blue-600 shadow-sm' : 'text-slate-500'}`}
                >
                  {m}
                </button>
              ))}
           </div>
        </div>

        {/* Global Stats Grid */}
        <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
          <div className="bg-white p-4 rounded-2xl shadow-sm border border-slate-100">
            <span className="text-[9px] font-black text-slate-400 uppercase block">Eficiência de Qualidade</span>
            <div className="flex items-baseline gap-1">
              <span className="text-xl font-black text-emerald-500">{stats.qualityEfficiency.toFixed(1)}%</span>
              <TrendingUp size={12} className="text-emerald-400" />
            </div>
          </div>
          <div className="bg-white p-4 rounded-2xl shadow-sm border border-slate-100">
            <span className="text-[9px] font-black text-slate-400 uppercase block">Total Borra (Perda)</span>
            <span className="text-xl font-black text-red-500">{stats.borra.toLocaleString('pt-BR')} kg</span>
          </div>
          <div className="bg-white p-4 rounded-2xl shadow-sm border border-slate-100">
            <span className="text-[9px] font-black text-slate-400 uppercase block">Horas de Parada</span>
            <span className="text-xl font-black text-orange-500">{(stats.paradas / 60).toFixed(1)}h</span>
          </div>
          <div className="bg-white p-4 rounded-2xl shadow-sm border border-slate-100">
            <span className="text-[9px] font-black text-slate-400 uppercase block">Produção Total</span>
            <span className="text-xl font-black text-blue-600">{(stats.totalNet / 1000).toFixed(1)}t</span>
          </div>
        </div>

        {/* Goal Tracking */}
        <div className={`bg-white rounded-2xl p-6 shadow-sm transition-all duration-500 border ${isGoalAchieved ? 'border-emerald-500 shadow-emerald-100 ring-4 ring-emerald-500/10 scale-[1.02]' : 'border-blue-500/10'}`}>
          <div className="flex items-center justify-between mb-4">
            <div>
               <span className={`text-[10px] font-black uppercase tracking-widest ${isGoalAchieved ? 'text-emerald-500' : 'text-blue-500'}`}>
                 {isGoalAchieved ? 'Meta Global Superada!' : 'Atingimento da Meta Global'}
               </span>
               <div className="flex items-baseline gap-2">
                  <span className="text-4xl font-black text-slate-800 tracking-tighter">{stats.totalNet.toLocaleString('pt-BR')}</span>
                  <span className="text-slate-400 font-medium text-xs">/ {goalValue.toLocaleString('pt-BR')} kg</span>
               </div>
            </div>
            <div className="text-right">
              <span className={`text-3xl font-black ${isGoalAchieved ? 'text-emerald-600' : 'text-blue-600'}`}>
                {Math.round((stats.totalNet / goalValue) * 100)}%
              </span>
            </div>
          </div>
          <div className="w-full bg-slate-100 h-3 rounded-full overflow-hidden">
            <div 
              className={`h-full rounded-full transition-all duration-1000 shadow-lg ${isGoalAchieved ? 'bg-gradient-to-r from-emerald-500 to-teal-400 shadow-emerald-500/30' : 'bg-gradient-to-r from-blue-600 to-indigo-400 shadow-blue-500/30'}`} 
              style={{ width: `${Math.min((stats.totalNet / goalValue) * 100, 100)}%` }}
            ></div>
          </div>
        </div>

        {/* Charts Row */}
        <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
          <div className="bg-white rounded-2xl p-6 shadow-sm border border-slate-100">
            <h3 className="font-black text-slate-800 text-xs uppercase tracking-widest mb-6 flex items-center gap-2">
               <BarChart2 size={16} className="text-blue-500" /> Rendimento Operacional
            </h3>
            <div className="h-[240px] w-full">
              {operatorPerformanceData.length > 0 ? (
                <ResponsiveContainer width="100%" height="100%">
                  <BarChart data={operatorPerformanceData}>
                    <CartesianGrid vertical={false} stroke="#f1f5f9" />
                    <XAxis dataKey="name" axisLine={false} tickLine={false} tick={{ fontSize: 9, fontWeight: 'bold', fill: '#94a3b8' }} />
                    <YAxis axisLine={false} tickLine={false} tick={{ fontSize: 9, fill: '#94a3b8' }} />
                    <Tooltip cursor={{ fill: '#f8fafc' }} contentStyle={{ borderRadius: '12px', border: 'none', boxShadow: '0 4px 12px rgba(0,0,0,0.05)' }} />
                    <Bar dataKey="kg" fill="#3b82f6" radius={[4, 4, 0, 0]} barSize={25} />
                  </BarChart>
                </ResponsiveContainer>
              ) : <NoDataPlaceholder /> }
            </div>
          </div>

          <div className="bg-white rounded-2xl p-6 shadow-sm border border-slate-100">
            <h3 className="font-black text-slate-800 text-xs uppercase tracking-widest mb-6 flex items-center gap-2">
               <Clock size={16} className="text-orange-500" /> Motivos de Parada
            </h3>
            <div className="h-[240px] w-full">
              {stats.paradas > 0 ? (
                <ResponsiveContainer width="100%" height="100%">
                  <PieChart>
                    <Pie 
                      data={downtimePieData} 
                      cx="50%" cy="50%" innerRadius={60} outerRadius={85} paddingAngle={8} dataKey="value"
                    >
                      {downtimePieData.map((entry, index) => (
                        <Cell key={`cell-${index}`} fill={entry.color} />
                      ))}
                    </Pie>
                    <Tooltip />
                    <Legend verticalAlign="bottom" height={36} iconType="circle" wrapperStyle={{ fontSize: '10px', fontWeight: 'bold', textTransform: 'uppercase', paddingTop: '10px' }} />
                  </PieChart>
                </ResponsiveContainer>
              ) : <NoDataPlaceholder /> }
            </div>
          </div>
        </div>

        {/* History / Recent Entries */}
        <div className="bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden">
          <div className="p-4 border-b border-slate-100 bg-slate-50 flex items-center justify-between">
            <h3 className="font-black text-slate-800 text-[10px] uppercase tracking-widest">Histórico de Lançamentos</h3>
            <div className="flex items-center gap-2 text-[10px] font-bold text-slate-400">
              <span className="px-2 py-0.5 bg-slate-200 rounded">{filteredData.length} registros</span>
            </div>
          </div>
          <div className="divide-y divide-slate-50 max-h-[500px] overflow-y-auto custom-scrollbar">
            {filteredData.slice().reverse().map((entry) => (
              <div key={entry.id} className="p-4 flex items-center justify-between hover:bg-slate-50/50 transition-colors">
                <div className="flex items-center gap-4">
                  <div className={`w-2 h-10 rounded-full ${entry.shift === 'Diurno' ? 'bg-amber-400' : 'bg-indigo-600'}`}></div>
                  <div>
                    <div className="flex items-center gap-2">
                      <span className="font-black text-slate-800 text-sm">{entry.operator}</span>
                      <span className="text-[10px] bg-slate-100 text-slate-500 px-2 py-0.5 rounded font-bold uppercase">{entry.machine}</span>
                    </div>
                    <p className="text-[10px] text-slate-400 font-medium">{entry.date} • {entry.shift}</p>
                  </div>
                </div>
                <div className="text-right flex items-center gap-6">
                  <div>
                    <span className="block font-black text-blue-600 text-sm">{entry.netWeight.toLocaleString('pt-BR')} kg</span>
                    <span className="text-[9px] text-slate-400 font-bold uppercase">Líquido</span>
                  </div>
                  <button onClick={() => handleDeleteEntry(entry.id)} className="text-slate-300 hover:text-red-500 transition-colors p-2">
                    <Trash2 size={16} />
                  </button>
                </div>
              </div>
            ))}
          </div>
        </div>
      </main>

      <LaunchModal 
        isOpen={isModalOpen} 
        onClose={() => setIsModalOpen(false)} 
        onSave={handleSaveEntry} 
        operators={operators}
      />
      
      {/* Footer Branding */}
      <footer className="py-8 text-center">
        <p className="text-[10px] font-black text-slate-300 uppercase tracking-[0.2em]">Manupackaging Industrial System • 2026</p>
      </footer>
    </div>
  );

  function exportToCSV() {
    const headers = ['Data', 'Operador', 'Maquina', 'Turno', 'Peso Liquido', 'Paradas'];
    const rows = filteredData.map(e => [e.date, e.operator, e.machine, e.shift, e.netWeight, e.manutencaoMin + e.processoMin + e.outrosMin]);
    const csvContent = [headers, ...rows].map(r => r.join(';')).join('\n');
    const blob = new Blob([csvContent], { type: 'text/csv' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `relatorio_producao_${new Date().toISOString().split('T')[0]}.csv`;
    a.click();
    setIsMenuOpen(false);
  }
};
