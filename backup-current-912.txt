const renderCustomizedLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) => {
  if (percent < 0.05) return null;
  const RADIAN = Math.PI / 180;
  const radius = outerRadius * 1.15;
  const x = cx + radius * Math.cos(-midAngle * RADIAN);
  const y = cy + radius * Math.sin(-midAngle * RADIAN);
  return (
    <text x={x} y={y} fill="#1e293b" textAnchor={x > cx ? 'start' : 'end'} dominantBaseline="central" className="text-[10px] font-black">
      {`${(percent * 100).toFixed(0)}%`}
    </text>
  );
};

                        <button onClick={() => setIsShiftModalOpen(true)} className="w-full px-5 py-3.5 bg-slate-50 text-slate-700 rounded-2xl font-black text-[11px] uppercase flex items-center gap-3 border border-slate-100 hover:bg-slate-100 transition-all"><Clock size={18} className="text-orange-500"/> Gerenciar Turnos</button>
                    </div>

                    <div className="pt-8 border-t border-slate-100 space-y-4">
                        <button onClick={() => fileInputRef.current?.click()} className="w-full py-4 bg-blue-50 text-blue-600 rounded-2xl font-black text-[11px] uppercase flex items-center justify-center gap-2 border border-blue-100 hover:bg-blue-100 transition-all"><Upload size={18}/> Restaurar Backup</button>
                        <button onClick={() => { localStorage.removeItem('manupackaging_session'); setIsAuthenticated(false); }} className="w-full py-4 bg-slate-800 text-white rounded-2xl font-black text-[11px] uppercase flex items-center justify-center gap-2 shadow-xl hover:bg-slate-900 transition-all"><LogOut size={18}/> Sair</button>
                    </div>
                </div>
            </div>
        </div>
      )}

      <input type="file" ref={fileInputRef} onChange={(e) => {
        const file = e.target.files?.[0];
        if (file) {
            const reader = new FileReader();
            reader.onload = (ev) => {
                const data = JSON.parse(ev.target?.result as string);
                if (data.productionData) setProductionData(data.productionData);
                if (data.employees) setEmployees(data.employees);
                if (data.operators) setOperators(data.operators);
            };
            reader.readAsText(file);
        }
      }} accept=".json" className="hidden" />

      <LaunchModal isOpen={isModalOpen} onClose={() => setIsModalOpen(false)} onSave={(e) => {
        if(editingEntry) setProductionData(p => p.map(x => x.id === editingEntry.id ? {...e, id: editingEntry.id} as any : x));
        else setProductionData(p => [{...e, id: Math.random().toString(36).substr(2, 9)} as any, ...p]);
        setEditingEntry(null);
      }} operators={operators} activeMachines={activeMachines} availableShifts={availableShifts} initialData={editingEntry} />
      
      <EmployeeModal isOpen={isEmployeeModalOpen} onClose={() => { setIsEmployeeModalOpen(false); setSelectedEmployee(null); setSelectedSlot(null); }} onSave={(data, action, details) => {
          const now = new Date().toISOString();
          if (action === 'Contratação') setEmployees(prev => [...prev, {...data, id: Math.random().toString(36).substr(2, 9), updatedAt: now} as Employee]);
          else if (action === 'Exclusão') setEmployees(prev => prev.filter(e => e.id !== data.id));
          else setEmployees(prev => prev.map(e => e.id === data.id ? {...e, ...data, updatedAt: now} : e));
          setPersonnelLogs(prev => [{id: Math.random().toString(36).substr(2, 9), date: now, employeeName: data.name || 'Vaga', action: action as any, details: details || '', user: 'Admin'}, ...prev]);
      }} availableShifts={availableShifts} availableMachines={activeMachines} availableRoles={availableRoles} initialData={selectedEmployee} slotInfo={selectedSlot} />
      
      <OperatorModal isOpen={isOperatorModalOpen} onClose={() => setIsOperatorModalOpen(false)} operators={operators} onUpdate={setOperators} />
      <RoleModal isOpen={isRoleModalOpen} onClose={() => setIsRoleModalOpen(false)} roles={availableRoles} onUpdate={setAvailableRoles} />
      <ShiftModal isOpen={isShiftModalOpen} onClose={() => setIsShiftModalOpen(false)} onSave={(s) => setAvailableShifts(prev => [...prev, {...s, id: Math.random().toString(36).substr(2, 9)}])} />
      <HistoryModal isOpen={isHistoryModalOpen} onClose={() => setIsHistoryModalOpen(false)} logs={personnelLogs} />
    </div>
  );
};
