const renderCustomizedLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) => {
  if (percent < 0.03) return null;
  const RADIAN = Math.PI / 180;
  const radius = outerRadius * 1.15;
  const x = cx + radius * Math.cos(-midAngle * RADIAN);
  const y = cy + radius * Math.sin(-midAngle * RADIAN);
  
  return (
    <text x={x} y={y} fill="#1e293b" textAnchor={x > cx ? 'start' : 'end'} dominantBaseline="central" className="text-[10px] font-black">
      {`${(percent * 100).toFixed(0)}%`}
    </text>
  );
};

interface OperatorStat {
  name: string;
  prod: number;
  ecoA: number;
  ecoBP: number;
  ecoBM: number;
  borra: number;
  p_manut: number;
  p_proc: number;
  p_outros: number;
  entries: number;
}

export const App: React.FC = () => {
  const [isAuthenticated, setIsAuthenticated] = useState(() => !!localStorage.getItem('manupackaging_session'));
  const [activeTab, setActiveTab] = useState<'dashboard' | 'reports' | 'personnel'>(() => {
    return (localStorage.getItem('manupackaging_active_tab') as 'dashboard' | 'reports' | 'personnel') || 'dashboard';
  });

  const [productionData, setProductionData] = useState<ProductionEntry[]>(() => {
    try {
      const saved = localStorage.getItem('manupackaging_data');
      return saved ? JSON.parse(saved) : INITIAL_DATA;
    } catch (e) { return INITIAL_DATA; }
  });

  const [goalValue, setGoalValue] = useState<number>(() => {
    const saved = localStorage.getItem('manupackaging_goal');
    return saved ? parseFloat(saved) : DEFAULT_GOAL;
  });

  const [dashboardMonth, setDashboardMonth] = useState(() => {
    const savedMonth = localStorage.getItem('manupackaging_dashboard_month');
    return savedMonth || new Date().toISOString().slice(0, 7);
  });

  const [operators, setOperators] = useState<string[]>(() => {
    const saved = localStorage.getItem('manupackaging_operators');
    return saved ? JSON.parse(saved) : DEFAULT_OPERATORS;
  });

  const [shifts, setShifts] = useState<Shift[]>(() => {
    const saved = localStorage.getItem('manupackaging_shifts');
    return saved ? JSON.parse(saved) : [
      { id: '1', name: 'Diurno 1', startTime: '07:00', endTime: '19:00', type: 'Diurno' },
      { id: '2', name: 'Noturno 1', startTime: '19:00', endTime: '07:00', type: 'Noturno' },
      { id: '3', name: 'Diurno 2', startTime: '07:00', endTime: '19:00', type: 'Diurno' },
      { id: '4', name: 'Noturno 2', startTime: '19:00', endTime: '07:00', type: 'Noturno' }
    ];
  });

    const saved = localStorage.getItem('manupackaging_employees');
    return saved ? JSON.parse(saved) : [];
  });

  const [personnelLogs, setPersonnelLogs] = useState<PersonnelLog[]>(() => {
    const saved = localStorage.getItem('manupackaging_personnel_logs');
    return saved ? JSON.parse(saved) : [];
  });

  const [jobRoles, setJobRoles] = useState<string[]>(['Gerente', 'Supervisor', 'Líder', 'Operador 1', 'Operador 2', 'Operador 3', 'Auxiliar']);
  
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isMenuOpen, setIsMenuOpen] = useState(false);
  const [isShiftModalOpen, setIsShiftModalOpen] = useState(false);
  const [isOperatorModalOpen, setIsOperatorModalOpen] = useState(false);
  const [isRoleModalOpen, setIsRoleModalOpen] = useState(false);
  const [isHistoryModalOpen, setIsHistoryModalOpen] = useState(false);
  const [isEmployeeModalOpen, setIsEmployeeModalOpen] = useState(false);
  
  const [editingEntry, setEditingEntry] = useState<ProductionEntry | null>(null);
  const [selectedEmployee, setSelectedEmployee] = useState<Employee | null>(null);
  const [searchTerm, setSearchTerm] = useState('');
  const [selectedIds, setSelectedIds] = useState<string[]>([]);
  
  const fileInputRef = useRef<HTMLInputElement>(null);

  useEffect(() => { localStorage.setItem('manupackaging_data', JSON.stringify(productionData)); }, [productionData]);
  useEffect(() => { localStorage.setItem('manupackaging_goal', goalValue.toString()); }, [goalValue]);
  useEffect(() => { localStorage.setItem('manupackaging_dashboard_month', dashboardMonth); }, [dashboardMonth]);
  useEffect(() => { localStorage.setItem('manupackaging_active_tab', activeTab); }, [activeTab]);
  useEffect(() => { localStorage.setItem('manupackaging_employees', JSON.stringify(employees)); }, [employees]);
  useEffect(() => { localStorage.setItem('manupackaging_personnel_logs', JSON.stringify(personnelLogs)); }, [personnelLogs]);
  useEffect(() => { localStorage.setItem('manupackaging_shifts', JSON.stringify(shifts)); }, [shifts]);
  useEffect(() => { localStorage.setItem('manupackaging_operators', JSON.stringify(operators)); }, [operators]);

  const activeMachines = ['Cast 1', 'Cast 2', 'Erema 1', 'Ghezzi', 'Lintech', 'Wutec', 'Supervisão'];

  const formatWeight = (val: number) => {
    if (isNaN(val)) return '0 KG';
    if (val >= 1000) return (val / 1000).toLocaleString('pt-BR', { minimumFractionDigits: 3, maximumFractionDigits: 3 }) + ' T';
    return val.toLocaleString('pt-BR') + ' KG';
  };

  const formatMinutes = (val: number) => val >= 60 ? `${Math.floor(val / 60)}h ${val % 60}m` : `${val} min`;

  const handleImport = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (event) => {
      try {
        const raw = JSON.parse(event.target?.result as string);
        const data = Array.isArray(raw) ? { productionData: raw } : raw;
        if (data.productionData) {
            const normalized = data.productionData.map((item: any) => ({
                ...item,
                date: item.date && item.date.includes('/') ? item.date.split('/').reverse().join('-') : item.date
            }));
            setProductionData(normalized);
            const sortedDates = normalized.map((i: any) => i.date).filter(Boolean).sort();
            if (sortedDates.length > 0) {
                setDashboardMonth(sortedDates[sortedDates.length - 1].slice(0, 7));
            }
        }
        if (data.employees) setEmployees(data.employees);
        if (data.goalValue) setGoalValue(Number(data.goalValue));
        if (data.operators) setOperators(data.operators);
        if (data.shifts) setShifts(data.shifts);
        alert('Backup restaurado com sucesso!');
        setIsMenuOpen(false);
      } catch (err) { alert('Erro ao processar backup.'); }
    };
    reader.readAsText(file);

  const handleExportCSV = () => {

    ].join(';'));
    const csvContent = "\uFEFF" + [headers.join(';'), ...rows].join('\n');

    link.href = URL.createObjectURL(blob);

    const refDay = isCurrentMonth ? now.getDate() : daysInMonth;

    productionData.forEach(e => {
      if (e.date.startsWith(dashboardMonth)) {

        results.totalBorra += (e.borraTotal || 0);
        results.totalStops += (e.manutencaoMin || 0) + (e.processoMin || 0) + (e.outrosMin || 0);
        results.totalActiveMinutes += 720; // 12h base per entry

    });

    results.dailyRequired = daysLeft > 0 ? results.remaining / daysLeft : results.remaining;
    return results;

      if (!e.machine.toLowerCase().includes('erema')) is[e.operator].prod += (e.netWeight || 0);

      is[e.operator].borra += (e.borraTotal || 0);
      is[e.operator].p_manut += (e.manutencaoMin || 0);
      is[e.operator].p_proc += (e.processoMin || 0);
      is[e.operator].p_outros += (e.outrosMin || 0);
      is[e.operator].ecoBP += (e.ecoBP || 0);
      is[e.operator].ecoBM += (e.ecoBM || 0);
      is[e.operator].entries += 1;
    });
    return Object.values(is);

    });
    return Object.values(ms).filter(m => m.time > 0);

    return productionData.filter(e => {
      const ms = searchTerm ? Object.values(e).some(v => String(v).toLowerCase().includes(searchTerm.toLowerCase())) : true;
      const mm = e.date.startsWith(dashboardMonth);
      return ms && mm;
    }).sort((a, b) => b.date.localeCompare(a.date));

      ecoA: acc.ecoA + (item.ecoA || 0),

  }, [filteredReports]);

    if (entry.id) setProductionData(prev => prev.map(e => e.id === entry.id ? { ...e, ...entry } : e));
    else setProductionData(prev => [{ ...entry, id: Math.random().toString(36).substr(2, 9) }, ...prev]);
  };

  const handleEmployeeAction = (data: Partial<Employee>, action: string, details?: string) => {
    const timestamp = new Date().toISOString();
    if (action === 'Exclusão') {
      setEmployees(prev => prev.filter(e => e.id !== data.id));
    } else if (data.id) {
      setEmployees(prev => prev.map(e => e.id === data.id ? { ...e, ...data, updatedAt: timestamp } : e));
    } else {
      setEmployees(prev => [...prev, { ...data, id: Math.random().toString(36).substr(2, 9), updatedAt: timestamp } as Employee]);
    }
    setPersonnelLogs(prev => [{ id: Math.random().toString(36).substr(2, 9), date: timestamp, employeeName: data.name || 'Desconhecido', action: action as any, details: details || '', user: 'Admin' }, ...prev]);
  };

  const chartConfigs = [
    { title: 'Produção Líquida por Operador', key: 'prod', formatter: formatWeight },
    { title: 'Eco A Produção por Operador', key: 'ecoA', formatter: formatWeight },
    { title: 'Eco B Produção por Operador', key: 'ecoB', formatter: formatWeight },
    { title: 'Borra por Operador', key: 'borra', formatter: formatWeight },
    { title: 'Tempo Manutenção (Min)', key: 'manut', formatter: formatMinutes },
    { title: 'Tempo Processo (Min)', key: 'proc', formatter: formatMinutes },
    { title: 'Total Paradas (Min)', key: 'stops', formatter: formatMinutes },
  ];

  if (!isAuthenticated) return (
    <div className="min-h-screen bg-slate-900 flex items-center justify-center p-4">
      <div className="bg-white rounded-3xl p-8 w-full max-w-md shadow-2xl space-y-6 text-center">
        <div className="w-16 h-16 bg-blue-600 rounded-2xl flex items-center justify-center text-white mx-auto shadow-lg mb-4"><ShieldCheck size={32} /></div>
        <h1 className="text-2xl font-black text-slate-800 tracking-tight uppercase">Manupackaging</h1>
        <button onClick={() => { localStorage.setItem('manupackaging_session', 'true'); setIsAuthenticated(true); }} className="w-full bg-blue-600 hover:bg-blue-700 text-white font-black py-4 rounded-2xl shadow-xl transition-all">Acessar Sistema</button>
      </div>
    </div>
  );

  return (
    <div className="min-h-screen bg-slate-50 pb-20 md:pb-24">
      <header className="bg-white px-4 md:px-6 py-3 md:py-4 flex items-center justify-between sticky top-0 z-40 border-b border-slate-100 no-print shadow-sm">
        <div className="flex items-center gap-2 md:gap-3">
          <div className="w-8 h-8 md:w-10 md:h-10 bg-blue-600 rounded-lg md:rounded-xl flex items-center justify-center text-white shadow-lg"><Cpu size={18} /></div>
          <h1 className="text-base md:text-lg font-black text-slate-800 tracking-tighter uppercase">Controle de Produção</h1>
        </div>
        <div className="flex items-center gap-1.5 md:gap-2">
          <button onClick={() => { setEditingEntry(null); setIsModalOpen(true); }} className="bg-blue-600 text-white p-2 md:p-2.5 rounded-lg md:rounded-xl shadow-lg hover:bg-blue-700 transition-all active:scale-95"><Plus size={20} md:size={24} /></button>

        </div>

'bg-slate-100 text-blue-600' : 'text-slate-400'}`}>Painel de Indicadores</button>

                 </div>

            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
              {/* Comparações Globais (Novos Gráficos de Eficiência) */}
              <div className="bg-white p-6 rounded-2xl md:rounded-[2rem] border border-slate-100 shadow-sm flex flex-col h-[400px]">
                <h3 className="text-xs font-black uppercase text-slate-800 mb-4 text-center">Operação vs Parada (Tempo)</h3>

                  <PieChart>
                    <Pie data={[
                        { name: 'Em Operação', value: Math.max(0, totals.totalActiveMinutes - totals.totalStops) },
                        { name: 'Parado', value: totals.totalStops }
                      ]} cx="50%" cy="50%" innerRadius={60} outerRadius={80} paddingAngle={5} dataKey="value" label={renderCustomizedLabel}>
                      <Cell fill="#10b981" />
                      <Cell fill="#ef4444" />
                    </Pie>
                    <Tooltip formatter={(v: number) => formatMinutes(v)} />
                    <Legend iconType="circle" wrapperStyle={{fontSize: '10px', fontWeight: 'bold', paddingTop: '10px'}} />
                  </PieChart>
                </ResponsiveContainer>

              <div className="bg-white p-6 rounded-2xl md:rounded-[2rem] border border-slate-100 shadow-sm flex flex-col h-[400px]">

                <ResponsiveContainer width="100%" height="100%">

                    <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="#f1f5f9" />

                    <YAxis tickFormatter={(v) => `${(v/1000).toFixed(0)}T`} tick={{fontSize: 9, fontWeight: 'bold'}} />
                    <Tooltip formatter={(v: number) => formatWeight(v)} />

                <h3 className="text-xs font-black uppercase text-slate-800 mb-4 text-center">Produção vs Borra</h3>
                <ResponsiveContainer width="100%" height="100%">
                  <BarChart data={[{ name: 'Comparativo', prod: totals.month, borra: totals.totalBorra }]}>
                    <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="#f1f5f9" />
                    <XAxis dataKey="name" hide />
                    <YAxis tickFormatter={(v) => `${(v/1000).toFixed(0)}T`} tick={{fontSize: 9, fontWeight: 'bold'}} />

                    <Legend />

                    <Bar name="Borra" dataKey="borra" fill="#ef4444" radius={[4, 4, 0, 0]} />
                  </BarChart>
                </ResponsiveContainer>
              </div>

              <div className="bg-white p-6 rounded-2xl md:rounded-[2rem] border border-slate-100 shadow-sm flex flex-col h-[400px] lg:col-span-2">
                <h3 className="text-xs font-black uppercase text-slate-800 mb-4 text-center">Tempo Efetivo de Máquina (Min)</h3>
                <ResponsiveContainer width="100%" height="100%">
                  <BarChart data={machineStats} layout="vertical">
                    <CartesianGrid strokeDasharray="3 3" horizontal={false} stroke="#f1f5f9" />
                    <XAxis type="number" tickFormatter={(v) => `${(v/60).toFixed(0)}h`} tick={{fontSize: 9, fontWeight: 'bold'}} />
                    <YAxis dataKey="name" type="category" width={80} tick={{fontSize: 9, fontWeight: 'bold'}} />
                    <Tooltip formatter={(v: number) => formatMinutes(v)} />
                    <Bar name="Operação" dataKey="time" fill="#8b5cf6" radius={[0, 4, 4, 0]} />
                  </BarChart>
                </ResponsiveContainer>
              </div>

              {/* Gráficos de Operadores */}
              {chartConfigs.map((config, idx) => {
                const data = stats.map(s => {
                  let value = 0;
                  if (config.key === 'prod') value = s.prod;
                  else if (config.key === 'ecoA') value = s.ecoA;
                  else if (config.key === 'borra') value = s.borra;
                  else if (config.key === 'manut') value = s.p_manut;
                  else if (config.key === 'proc') value = s.p_proc;
                  else if (config.key === 'stops') value = s.p_manut + s.p_proc + s.p_outros;
                  else if (config.key === 'ecoB') value = s.ecoBP + s.ecoBM;
                  return { name: s.name, value };
                }).filter(i => i.value > 0).sort((a, b) => b.value - a.value);
                if (data.length === 0) return null;
                return (
                  <div key={idx} className="bg-white p-6 rounded-2xl md:rounded-[2rem] border border-slate-100 shadow-sm flex flex-col h-[400px]">
                    <h3 className="text-xs font-black uppercase text-slate-800 mb-4 text-center">{config.title}</h3>
                    <ResponsiveContainer width="100%" height="100%">
                      <PieChart>
                        <Pie data={data} cx="50%" cy="50%" innerRadius={55} outerRadius={75} paddingAngle={3} dataKey="value" label={renderCustomizedLabel}>
                          {data.map((_, i) => <Cell key={`cell-${i}`} fill={COLORS[i % COLORS.length]} />)}
                        </Pie>
                        <Tooltip formatter={(v: number) => config.formatter(v)} />
                        <Legend iconType="circle" wrapperStyle={{fontSize: '9px', fontWeight: 'bold', paddingTop: '10px'}} />
                      </PieChart>
                    </ResponsiveContainer>
                  </div>
                );
              })}
            </div>
          </div>
        )}

        {activeTab === 'reports' && (
          <div className="space-y-4 md:space-y-6 animate-in fade-in duration-300">
            <div className="bg-white p-3 md:p-4 rounded-2xl md:rounded-3xl border border-slate-100 shadow-sm flex flex-col gap-3">
              <div className="flex items-center gap-2 bg-slate-50 rounded-xl md:rounded-2xl px-3 md:px-4 py-2.5 md:py-3 border border-slate-100 w-full">
                <Search className="text-slate-400" size={16} />
                <input type="text" placeholder="Filtrar lançamentos..." value={searchTerm} onChange={e => setSearchTerm(e.target.value)} className="bg-transparent border-none w-full text-xs md:text-sm font-bold outline-none" />
              </div>
              <div className="flex items-center gap-2 w-full">
                <button onClick={handleExportCSV} className="flex-1 bg-emerald-600 text-white px-3 md:px-6 py-2.5 md:py-3 rounded-xl md:rounded-2xl text-[10px] font-black uppercase flex items-center justify-center gap-2 hover:bg-emerald-700 transition-all active:scale-95"><FileDown size={14} /> CSV</button>
                {selectedIds.length > 0 && <button onClick={handleBulkDelete} className="flex-1 bg-red-50 text-red-600 px-3 md:px-6 py-2.5 md:py-3 rounded-xl md:rounded-2xl text-[10px] font-black uppercase flex items-center justify-center gap-2 border border-red-100 hover:bg-red-100 transition-all active:scale-95"><Trash2 size={14} /> ({selectedIds.length})</button>}
              </div>
            </div>
            
            <div className="bg-white rounded-2xl md:rounded-[2rem] border border-slate-100 overflow-hidden shadow-sm">
              <div className="overflow-x-auto custom-scrollbar">
                <table className="w-full text-left min-w-[850px]">
                  <thead className="bg-slate-50 border-b border-slate-100 text-[8px] md:text-[9px] font-black uppercase tracking-widest text-slate-400">
                    <tr>
                      <th className="px-4 md:px-6 py-3 md:py-4 w-10 text-center"><input type="checkbox" className="rounded" onChange={e => setSelectedIds(e.target.checked ? filteredReports.map(x => x.id) : [])} /></th>
                      <th className="px-4 md:px-6 py-3 md:py-4">Data/Ref</th>
                      <th className="px-4 md:px-6 py-3 md:py-4">Máquina</th>
                      <th className="px-4 md:px-6 py-3 md:py-4">Operador</th>
                      <th className="px-4 md:px-6 py-3 md:py-4 text-right">Peso Líq</th>
                      <th className="px-4 md:px-6 py-3 md:py-4 text-right">Eco A</th>
                      <th className="px-4 md:px-6 py-3 md:py-4 text-right">Borra</th>
                      <th className="px-4 md:px-6 py-3 md:py-4 text-right">Paradas</th>
                      <th className="px-4 md:px-6 py-3 md:py-4"></th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-slate-50">
                    {filteredReports.map(item => (
                      <tr key={item.id} className={`hover:bg-blue-50/20 transition-colors text-[10px] md:text-[11px] font-bold ${selectedIds.includes(item.id) ? 'bg-blue-50/40' : ''}`}>
                        <td className="px-4 md:px-6 py-3 md:py-4 text-center"><input type="checkbox" checked={selectedIds.includes(item.id)} onChange={() => setSelectedIds(prev => prev.includes(item.id) ? prev.filter(x => x !== item.id) : [...prev, item.id])} /></td>
                        <td className="px-4 md:px-6 py-3 md:py-4"><span>{item.date.split('-').reverse().join('/')}</span><br/><span className="text-[8px] text-slate-400 font-bold uppercase">{item.shift}</span></td>
                        <td className="px-4 md:px-6 py-3 md:py-4"><span className="bg-slate-100 px-2 py-0.5 rounded text-slate-600 uppercase text-[8px] md:text-[9px] font-black border border-slate-200">{item.machine}</span></td>
                        <td className="px-4 md:px-6 py-3 md:py-4">{item.operator}</td>
                        <td className="px-4 md:px-6 py-3 md:py-4 text-right font-black text-blue-600">{formatWeight(item.netWeight)}</td>
                        <td className="px-4 md:px-6 py-3 md:py-4 text-right text-emerald-600">{item.ecoA}kg</td>
                        <td className="px-4 md:px-6 py-3 md:py-4 text-right text-red-500">{item.borraTotal}kg</td>
                        <td className="px-4 md:px-6 py-3 md:py-4 text-right text-orange-400">{item.manutencaoMin + item.processoMin + item.outrosMin}m</td>
                        <td className="px-4 md:px-6 py-3 md:py-4"><button onClick={() => { setEditingEntry(item); setIsModalOpen(true); }} className="p-1.5 text-blue-600 hover:bg-blue-100 rounded-lg transition-all"><Edit2 size={12}/></button></td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
            </div>
          </div>
        )}

        {activeTab === 'personnel' && (
          <div className="space-y-6 md:space-y-8 animate-in fade-in duration-300">
             <div className="grid grid-cols-2 md:grid-cols-4 gap-3 md:gap-4">
              <div className="bg-white p-4 md:p-6 rounded-2xl md:rounded-[2rem] shadow-sm border border-slate-100 flex flex-col justify-center">
                <p className="text-[8px] md:text-[10px] font-black text-slate-400 uppercase tracking-widest mb-1">Equipe Ativa</p>
                <p className="text-xl md:text-3xl font-black text-slate-800">{employees.filter(e => e.status === 'Ativo').length}</p>
              </div>
              <div className="bg-white p-4 md:p-6 rounded-2xl md:rounded-[2rem] shadow-sm border border-slate-100 flex flex-col justify-center">
                <p className="text-[8px] md:text-[10px] font-black text-slate-400 uppercase tracking-widest mb-1">Afastados</p>
                <p className="text-xl md:text-3xl font-black text-orange-600">{employees.filter(e => e.status !== 'Ativo' && e.status !== 'Desligado').length}</p>
              </div>
              <button onClick={() => setIsHistoryModalOpen(true)} className="col-span-2 bg-slate-800 text-white rounded-2xl md:rounded-[2rem] p-4 md:p-6 flex items-center justify-between hover:bg-slate-900 transition-all group active:scale-[0.98]">
                <div><p className="text-[8px] md:text-[10px] font-black opacity-60 uppercase tracking-widest mb-1">Auditoria RH</p><p className="text-sm md:text-xl font-black uppercase">Movimentações</p></div>
                <div className="w-10 h-10 md:w-12 md:h-12 bg-white/10 rounded-xl md:rounded-2xl flex items-center justify-center group-hover:bg-white/20 transition-all"><History size={20} md:size={24} /></div>
              </button>
            </div>

            <div className="space-y-6 md:space-y-8">
               <div className="bg-white rounded-2xl md:rounded-[2rem] shadow-sm border border-slate-100 overflow-hidden">
                  <div className="bg-indigo-600 p-4 md:p-6 flex items-center justify-between text-white">
                    <div className="flex items-center gap-3 md:gap-4">
                        <div className="p-2 md:p-3 bg-white/20 rounded-xl md:rounded-2xl"><HardHat size={20} md:size={24} /></div>
                        <div><h3 className="text-base md:text-lg font-black uppercase tracking-tight leading-none">Liderança</h3><p className="text-[8px] md:text-[10px] opacity-70 font-bold uppercase mt-1">Gestão de Operação</p></div>
                    </div>
                    <button onClick={() => { setSelectedEmployee(null); setIsEmployeeModalOpen(true); }} className="bg-white text-indigo-600 px-3 md:px-4 py-1.5 md:py-2 rounded-lg md:rounded-xl text-[9px] md:text-[10px] font-black uppercase hover:bg-indigo-50 transition-all active:scale-95">Contratar</button>
                  </div>
                  <div className="p-4 md:p-6 grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
                    {employees.filter(e => e.sector === 'Liderança' || e.sector === 'Supervisão').map(emp => (
                      <div key={emp.id} onClick={() => { setSelectedEmployee(emp); setIsEmployeeModalOpen(true); }} className="bg-slate-50 p-4 rounded-xl border border-slate-100 hover:border-indigo-300 transition-all cursor-pointer group active:bg-white">
                        <div className="flex justify-between items-start mb-2">
                            <span className={`px-2 py-0.5 rounded-lg text-[7px] md:text-[8px] font-black uppercase ${emp.status === 'Ativo' ? 'bg-emerald-100 text-emerald-600' : 'bg-orange-100 text-orange-600'}`}>{emp.status}</span>
                            <ChevronRight size={14} className="text-slate-300 group-hover:text-indigo-600" />
                        </div>
                        <p className="text-sm font-black text-slate-800">{emp.name}</p>
                        <p className="text-[9px] font-bold text-slate-400 uppercase">{emp.role}</p>
                      </div>
                    ))}
                  </div>
               </div>

               <div className="bg-white rounded-2xl md:rounded-[2rem] shadow-sm border border-slate-100 overflow-hidden">
                  <div className="bg-slate-800 p-4 md:p-6 flex items-center justify-between text-white">
                    <div className="flex items-center gap-3 md:gap-4">
                        <div className="p-2 md:p-3 bg-white/20 rounded-xl md:rounded-2xl"><Factory size={20} md:size={24} /></div>
                        <div><h3 className="text-base md:text-lg font-black uppercase tracking-tight leading-none">Extrusão Cast 1 & 2</h3><p className="text-[8px] md:text-[10px] opacity-70 font-bold uppercase mt-1">Grade por Turno</p></div>
                    </div>
                    <button onClick={() => setIsEmployeeModalOpen(true)} className="bg-white/10 hover:bg-white/20 p-2 md:p-2.5 rounded-xl transition-all"><Plus size={18} md:size={20} /></button>
                  </div>
                  <div className="p-4 md:p-6">
                    <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-6 md:gap-8">
                      {['Diurno 1', 'Noturno 1', 'Diurno 2', 'Noturno 2'].map(shiftName => (
                        <div key={shiftName} className="space-y-3">
                          <h4 className="text-[9px] md:text-[10px] font-black text-slate-400 uppercase tracking-widest border-b border-slate-100 pb-2 flex items-center gap-2"><Clock size={12}/> {shiftName}</h4>
                          <div className="space-y-2">
                            {employees.filter(e => e.shift === shiftName && (e.machine === 'Cast 1' || e.machine === 'Cast 2')).map(emp => (
                              <div key={emp.id} onClick={() => { setSelectedEmployee(emp); setIsEmployeeModalOpen(true); }} className="bg-slate-50 p-3 rounded-xl border border-slate-100 flex items-center justify-between cursor-pointer hover:bg-white hover:shadow-md transition-all group">
                                <div className="truncate pr-2"><p className="text-[10px] md:text-[11px] font-black text-slate-700 truncate">{emp.name}</p><p className="text-[8px] font-bold text-slate-400 uppercase">{emp.machine} · {emp.role}</p></div>
                                <div className={`w-2 h-2 shrink-0 rounded-full ${emp.status === 'Ativo' ? 'bg-emerald-500' : 'bg-orange-500'}`}></div>
                              </div>
                            ))}
                          </div>
                        </div>
                      ))}
                    </div>
                  </div>
               </div>
            </div>
          </div>
        )}
      </main>

      {/* Menu Lateral Oculto */}
      {isMenuOpen && (
        <div className="fixed inset-0 z-50 flex justify-end bg-black/40 backdrop-blur-sm" onClick={() => setIsMenuOpen(false)}>
            <div className="w-[85%] sm:w-80 bg-white h-full shadow-2xl p-5 md:p-6 animate-in slide-in-from-right duration-300" onClick={e => e.stopPropagation()}>
                <div className="flex justify-between items-center mb-6 md:mb-8"><h2 className="text-lg md:text-xl font-black uppercase text-slate-800">Opções</h2><button onClick={() => setIsMenuOpen(false)}><X size={24} /></button></div>
                <div className="space-y-5">
                    <div><label className="text-[9px] md:text-[10px] font-black text-slate-400 mb-1.5 md:mb-2 block uppercase tracking-widest">Meta Produção (KG)</label><input type="number" value={goalValue} onChange={e => setGoalValue(Number(e.target.value))} className="w-full bg-slate-50 border rounded-xl px-4 py-3 font-bold border-slate-200 focus:border-blue-500 outline-none text-sm" /></div>
                    <div><label className="text-[9px] md:text-[10px] font-black text-slate-400 mb-1.5 md:mb-2 block uppercase tracking-widest">Mês Referência</label><input type="month" value={dashboardMonth} onChange={e => setDashboardMonth(e.target.value)} className="w-full bg-slate-50 border rounded-xl px-4 py-3 font-bold border-slate-200 focus:border-blue-500 outline-none text-sm" /></div>
                    
                    <div className="pt-5 border-t space-y-2">
                      <p className="text-[8px] md:text-[9px] font-black text-slate-400 uppercase tracking-widest mb-1.5 md:mb-2">Configurar Tabelas</p>
                      <button onClick={() => { setIsOperatorModalOpen(true); setIsMenuOpen(false); }} className="w-full py-3 bg-slate-50 text-slate-700 rounded-xl font-bold text-xs uppercase flex items-center justify-between px-4 hover:bg-slate-100 border border-slate-100"><span className="flex items-center gap-2"><Users size={14}/> Operadores</span> <ChevronRight size={14}/></button>
                      <button onClick={() => { setIsShiftModalOpen(true); setIsMenuOpen(false); }} className="w-full py-3 bg-slate-50 text-slate-700 rounded-xl font-bold text-xs uppercase flex items-center justify-between px-4 hover:bg-slate-100 border border-slate-100"><span className="flex items-center gap-2"><Clock size={14}/> Turnos</span> <ChevronRight size={14}/></button>
                      <button onClick={() => { setIsRoleModalOpen(true); setIsMenuOpen(false); }} className="w-full py-3 bg-slate-50 text-slate-700 rounded-xl font-bold text-xs uppercase flex items-center justify-between px-4 hover:bg-slate-100 border border-slate-100"><span className="flex items-center gap-2"><Briefcase size={14}/> Funções</span> <ChevronRight size={14}/></button>
                    </div>

                    <div className="pt-5 border-t space-y-3">
                        <button onClick={() => fileInputRef.current?.click()} className="w-full py-3.5 bg-blue-50 text-blue-600 rounded-xl font-black text-[10px] uppercase flex items-center justify-center gap-2 hover:bg-blue-100 border border-blue-100"><Upload size={14}/> Restaurar Backup</button>
                        <button onClick={() => { localStorage.removeItem('manupackaging_session'); setIsAuthenticated(false); }} className="w-full py-3.5 bg-slate-800 text-white rounded-xl font-black text-[10px] uppercase flex items-center justify-center gap-2 hover:bg-slate-900 transition-all shadow-xl shadow-slate-200"><LogOut size={14}/> Encerrar Sessão</button>
                    </div>
                </div>
            </div>
        </div>
      )}

      {/* Modais e Inputs */}
      <input type="file" ref={fileInputRef} onChange={handleImport} accept=".json" className="hidden" />
      <LaunchModal isOpen={isModalOpen} onClose={() => { setIsModalOpen(false); setEditingEntry(null); }} onSave={handleSaveEntry} operators={operators} activeMachines={activeMachines} availableShifts={shifts} initialData={editingEntry} />
      <ShiftModal isOpen={isShiftModalOpen} onClose={() => setIsShiftModalOpen(false)} onSave={s => setShifts(prev => [...prev, { ...s, id: Math.random().toString(36).substr(2, 9) }])} />
      <OperatorModal isOpen={isOperatorModalOpen} onClose={() => setIsOperatorModalOpen(false)} operators={operators} onUpdate={setOperators} />
      <RoleModal isOpen={isRoleModalOpen} onClose={() => setIsRoleModalOpen(false)} roles={jobRoles} onUpdate={setJobRoles} />
      <HistoryModal isOpen={isHistoryModalOpen} onClose={() => setIsHistoryModalOpen(false)} logs={personnelLogs} />
      <EmployeeModal isOpen={isEmployeeModalOpen} onClose={() => { setIsEmployeeModalOpen(false); setSelectedEmployee(null); }} onSave={handleEmployeeAction} availableShifts={shifts} availableMachines={activeMachines} availableRoles={jobRoles} initialData={selectedEmployee} />
    </div>
  );
};
