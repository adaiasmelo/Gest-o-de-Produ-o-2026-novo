const renderCustomizedLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) => {
  if (!percent || percent < 0.01) return null;
  const RADIAN = Math.PI / 180;
  const radius = outerRadius * 1.25;
  const x = cx + radius * Math.cos(-midAngle * RADIAN);
  const y = cy + radius * Math.sin(-midAngle * RADIAN);
  return (
    <text x={x} y={y} fill="#64748b" textAnchor={x > cx ? 'start' : 'end'} dominantBaseline="central" className="text-[10px] font-black">
      {`${(percent * 100).toFixed(1)}%`}
    </text>
  );
};

  name: string;
  prod: number;
  ecoA: number;
  ecoBP: number;

  borra: number;

const App: React.FC = () => {
  const [isAuthenticated, setIsAuthenticated] = useState(() => !!localStorage.getItem('manupackaging_session'));
  const [activeTab, setActiveTab] = useState<'dashboard' | 'reports'>(() => {
    return (localStorage.getItem('manupackaging_active_tab') as 'dashboard' | 'reports') || 'dashboard';
  });
  const [selectedTimeframe, setSelectedTimeframe] = useState<'month' | 'today' | 'yesterday' | 'week'>(() => {

  const [confirmDialog, setConfirmDialog] = useState<{
    isOpen: boolean;
    title: string;
    message: string;
    onConfirm: () => void;
  }>({
    isOpen: false,
    title: '',
    message: '',
    onConfirm: () => {},
  });

  const [dashboardMonth, setDashboardMonth] = useState(() => {
    const savedMonth = localStorage.getItem('manupackaging_dashboard_month');
    return savedMonth || new Date().toISOString().slice(0, 7);
  });

    try {
      const saved = localStorage.getItem('manupackaging_data');
      let data = saved ? JSON.parse(saved) : INITIAL_DATA;
      if (Array.isArray(data)) {
        return data.map((item: any) => ({
          ...item,
          id: item.id || Math.random().toString(36).substr(2, 9)
        }));
      }
      return INITIAL_DATA;
    } catch (e) {
      return INITIAL_DATA;
    }
  });

  const [goalValue, setGoalValue] = useState<number>(() => {
    const saved = localStorage.getItem('manupackaging_goal');
    return saved ? parseFloat(saved) : DEFAULT_GOAL;
  });

  const [operators, setOperators] = useState<string[]>(() => {
    try {
      const saved = localStorage.getItem('manupackaging_operators');
      return saved ? JSON.parse(saved) : DEFAULT_OPERATORS;
    } catch (e) {
      return DEFAULT_OPERATORS;
    }
  });

  const [shifts, setShifts] = useState<Shift[]>(() => {
    try {
      const saved = localStorage.getItem('manupackaging_shifts');
      return saved ? JSON.parse(saved) : [
        { id: '1', name: 'Diurno 1', startTime: '07:00', endTime: '19:00', type: 'Diurno' },
        { id: '2', name: 'Noturno 1', startTime: '19:00', endTime: '07:00', type: 'Noturno' }

    } catch (e) {
      return [
        { id: '1', name: 'Diurno 1', startTime: '07:00', endTime: '19:00', type: 'Diurno' },
        { id: '2', name: 'Noturno 1', startTime: '19:00', endTime: '07:00', type: 'Noturno' }
      ];
    }
  });

  useEffect(() => { localStorage.setItem('manupackaging_shifts', JSON.stringify(shifts)); }, [shifts]);
  useEffect(() => { localStorage.setItem('manupackaging_goal', goalValue.toString()); }, [goalValue]);
  useEffect(() => { localStorage.setItem('manupackaging_operators', JSON.stringify(operators)); }, [operators]);
  useEffect(() => { localStorage.setItem('manupackaging_active_tab', activeTab); }, [activeTab]);
  useEffect(() => { localStorage.setItem('manupackaging_timeframe', selectedTimeframe); }, [selectedTimeframe]);
  useEffect(() => { localStorage.setItem('manupackaging_dashboard_month', dashboardMonth); }, [dashboardMonth]);

  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isShiftModalOpen, setIsShiftModalOpen] = useState(false);
  const [isOperatorModalOpen, setIsOperatorModalOpen] = useState(false);

  const [isMenuOpen, setIsMenuOpen] = useState(false);
  const [searchTerm, setSearchTerm] = useState('');
  const [filterStartDate, setFilterStartDate] = useState('');
  const [filterEndDate, setFilterEndDate] = useState('');
  const [selectedIds, setSelectedIds] = useState<string[]>([]);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const handleSaveEntry = (newEntry: Omit<ProductionEntry, 'id'> & { id?: string }) => {
    if (newEntry.id) {
      setProductionData(prev => prev.map(e => e.id === newEntry.id ? (newEntry as ProductionEntry) : e));
    } else {
      setProductionData(prev => [{ ...newEntry, id: Math.random().toString(36).substr(2, 9) } as ProductionEntry, ...prev]);
    }
  };

  const handleDeleteReportItem = (id: string) => {
    if (!id) return;
    setConfirmDialog({
      isOpen: true,
      title: 'Excluir Registro',
      message: 'Tem certeza que deseja excluir permanentemente este registro de produção?',
      onConfirm: () => {
        setProductionData(prev => prev.filter(item => item.id !== id));
        setSelectedIds(prev => prev.filter(sid => sid !== id));
        setConfirmDialog(prev => ({ ...prev, isOpen: false }));
      }
    });
  };

  const handleDeleteSelected = () => {
    if (selectedIds.length === 0) return;
    setConfirmDialog({
      isOpen: true,
      title: 'Excluir Selecionados',
      message: `Tem certeza que deseja excluir permanentemente os ${selectedIds.length} registros selecionados?`,
      onConfirm: () => {
        setProductionData(prev => prev.filter(item => !selectedIds.includes(item.id)));
        setSelectedIds([]);
        setConfirmDialog(prev => ({ ...prev, isOpen: false }));
      }
    });
  };

  const toggleSelectAll = (filtered: ProductionEntry[]) => {
    if (selectedIds.length === filtered.length) setSelectedIds([]);
    else setSelectedIds(filtered.map(i => i.id));
  };

  const toggleSelectId = (id: string) => {
    setSelectedIds(prev => prev.includes(id) ? prev.filter(i => i !== id) : [...prev, id]);
  };

    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (event) => {
      try {
        const data = JSON.parse(event.target?.result as string);
        if (data.productionData) setProductionData(data.productionData);
        if (data.goalValue) setGoalValue(Number(data.goalValue));
        if (data.shifts) setShifts(data.shifts);
        if (data.operators) setOperators(data.operators);
        alert('Backup carregado com sucesso!');
      } catch (err) {
        alert('Erro ao processar o backup.');
      }
    };
    reader.readAsText(file);
  };

  const formatWeight = (val: number) => {
    if (val >= 1000) {
      return (val / 1000).toLocaleString('pt-BR', { minimumFractionDigits: 3, maximumFractionDigits: 3 }) + ' T';
    }
    return val.toLocaleString('pt-BR') + ' KG';
  };
  
  const formatMinutes = (val: number) => val >= 60 ? `${Math.floor(val / 60)}h ${val % 60}m` : `${val} min`;

  const getShiftDuration = (shiftName: string) => {
    const shift = shifts.find(s => s.name === shiftName);
    if (!shift) return 720;
    const [h1, m1] = shift.startTime.split(':').map(Number);
    const [h2, m2] = shift.endTime.split(':').map(Number);
    let minutes = (h2 * 60 + m2) - (h1 * 60 + m1);
    if (minutes < 0) minutes += 1440;
    return minutes === 0 ? 1440 : minutes;
  };

  const filteredDataForCharts = useMemo(() => {
    const now = new Date();
    const todayStr = now.toLocaleDateString('sv-SE');
    const yesterday = new Date(now);
    yesterday.setDate(now.getDate() - 1);
    const yesterdayStr = yesterday.toLocaleDateString('sv-SE');

      if (selectedTimeframe === 'today') return e.date === todayStr;
      if (selectedTimeframe === 'yesterday') return e.date === yesterdayStr;
      if (selectedTimeframe === 'month') return e.date.startsWith(dashboardMonth);

        const startOfWeek = new Date(now);
        startOfWeek.setHours(0,0,0,0);
        startOfWeek.setDate(now.getDate() - now.getDay()); 
        const entryDate = new Date(e.date + 'T00:00:00');
        return entryDate >= startOfWeek;

      return true;
    });
  }, [productionData, selectedTimeframe, dashboardMonth]);

      const matchesStart = filterStartDate ? e.date >= filterStartDate : true;
      const matchesEnd = filterEndDate ? e.date <= filterEndDate : true;
      return matchesSearch && matchesStart && matchesEnd;
    });
  }, [productionData, searchTerm, filterStartDate, filterEndDate]);

    return filteredReports.reduce((acc, e) => {

      acc.volumes += (e.volumes || 0);

      acc.outrosMin += (e.outrosMin || 0);
      return acc;

    const csvContent = [headers, ...rows, totalRow].map(row => row.join(';')).join('\n');
    const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' });
    const link = document.createElement('a');
    const url = URL.createObjectURL(blob);
    link.setAttribute('href', url);
    link.setAttribute('download', `relatorio_producao_${new Date().toLocaleDateString('sv-SE')}.csv`);
    link.style.visibility = 'hidden';
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

  const handleSaveDashboardPDF = async () => {
    if (!dashboardRef.current || isGeneratingPDF) return;
    setIsGeneratingPDF(true);
    try {
      const canvas = await html2canvas(dashboardRef.current, {
        scale: 2,
        useCORS: true,
        logging: false,
        backgroundColor: '#f8fafc'
      });
      const imgData = canvas.toDataURL('image/png');
      const pdf = new jsPDF('p', 'mm', 'a4');
      
      const pageWidth = pdf.internal.pageSize.getWidth();
      const margin = 10;
      const contentWidth = pageWidth - (margin * 2);

      pdf.setFont("helvetica", "bold");
      pdf.setFontSize(18);
      pdf.setTextColor(15, 23, 42);
      pdf.text("Manupackaging - Controle de Produção", margin, 15);
      
      pdf.setFontSize(12);
      pdf.setFont("helvetica", "normal");
      pdf.setTextColor(71, 85, 105);
      pdf.text("Relatório Geral de Indicadores Industriais", margin, 22);
      
      pdf.setFontSize(10);
      pdf.setFont("helvetica", "bold");
      pdf.setTextColor(100, 116, 139);
      pdf.text(`Referência: ${getMonthName(dashboardMonth).toUpperCase()}`, margin, 28);
      
      pdf.setDrawColor(226, 232, 240);
      pdf.setLineWidth(0.5);
      pdf.line(margin, 32, pageWidth - margin, 32);

      const imgProps = pdf.getImageProperties(imgData);
      const pdfImgHeight = (imgProps.height * contentWidth) / imgProps.width;
      
      pdf.addImage(imgData, 'PNG', margin, 35, contentWidth, pdfImgHeight);
      
      pdf.save(`dashboard_producao_${dashboardMonth}.pdf`);
    } catch (error) {
      console.error(error);

    buttons.forEach(btn => (btn as HTMLElement).style.display = 'none');
    try {
      const canvas = await html2canvas(element, {
        scale: 3,
        useCORS: true,
        backgroundColor: '#ffffff',
        logging: false,
      });
      const url = canvas.toDataURL('image/png');
      const link = document.createElement('a');
      link.href = url;
      link.download = `grafico_${title.toLowerCase().replace(/[^a-z0-9]/g, '_')}.png`;
      link.click();
    } catch (error) {
      alert("Erro ao exportar imagem do gráfico.");

      buttons.forEach(btn => (btn as HTMLElement).style.display = 'flex');

    operators.forEach(op => {

    filteredDataForCharts.forEach(e => {

        initialStats[e.operator] = { name: e.operator, prod: 0, ecoA: 0, ecoBP: 0, ecoBM: 0, borra: 0, p_manut: 0, p_proc: 0, p_outros: 0 };
      }
      initialStats[e.operator].prod += (e.netWeight || 0);
      initialStats[e.operator].ecoA += (e.ecoA || 0);
      initialStats[e.operator].ecoBP += (e.ecoBP || 0);
      initialStats[e.operator].ecoBM += (e.ecoBM || 0);
      initialStats[e.operator].borra += (e.borraTotal || 0);
      initialStats[e.operator].p_manut += (e.manutencaoMin || 0);
      initialStats[e.operator].p_proc += (e.processoMin || 0);
      initialStats[e.operator].p_outros += (e.outrosMin || 0);

  }, [filteredDataForCharts, operators]);

    const now = new Date();

  }, [totals.month, goalValue, dashboardMonth]);

  useEffect(() => {
    if (totals.month >= goalValue && totals.month > 0) {
      const duration = 5 * 1000;
      const animationEnd = Date.now() + duration;
      const defaults = { startVelocity: 30, spread: 360, ticks: 60, zIndex: 100 };
      const randomInRange = (min: number, someMax: number) => Math.random() * (someMax - min) + min;
      const interval: any = setInterval(function() {
        const timeLeft = animationEnd - Date.now();
        if (timeLeft <= 0) return clearInterval(interval);
        const particleCount = 50 * (timeLeft / duration);
        confetti({ ...defaults, particleCount, origin: { x: randomInRange(0.1, 0.3), y: Math.random() - 0.2 } });
        confetti({ ...defaults, particleCount, origin: { x: randomInRange(0.7, 0.9), y: Math.random() - 0.2 } });
      }, 250);
      return () => clearInterval(interval);
    }

  const changeMonth = (direction: number) => {
    const [year, month] = dashboardMonth.split('-').map(Number);
    const date = new Date(year, month - 1 + direction, 1);
    setDashboardMonth(date.toISOString().slice(0, 7));
  };

  const getMonthName = (monthStr: string) => {
    const [year, month] = monthStr.split('-').map(Number);
    return new Date(year, month - 1).toLocaleDateString('pt-BR', { month: 'long', year: 'numeric' });
  };

  const allCharts = [
    { id: 'prod', title: '🏆 Produção Líquida', key: 'prod', color: COLORS, formatter: formatWeight, description: 'Distribuição da produção total de material acabado entre os operadores no período.' },
    { id: 'ecoA', title: '♻️ Eco A (KG)', key: 'ecoA', color: ['#10b981', '#34d399', '#6ee7b7'], formatter: formatWeight, description: 'Volume total de material reciclado do tipo Eco A gerado durante o processo.' },
    { id: 'ecoBP', title: '♻️ Eco B Produção', key: 'ecoBP', color: ['#3b82f6', '#60a5fa', '#93c5fd'], formatter: formatWeight, description: 'Resíduos tipo Eco B originados diretamente da operação normal das máquinas.' },
    { id: 'ecoBM', title: '♻️ Eco B Manutenção', key: 'ecoBM', color: ['#8b5cf6', '#a78bfa', '#c4b5fd'], formatter: formatWeight, description: 'Resíduos tipo Eco B gerados especificamente em processos de manutenção e ajuste.' },
    { id: 'borra', title: '🗑️ Borra Total', key: 'borra', color: ['#f43f5e', '#fb7185', '#fda4af'], formatter: formatWeight, description: 'Desperdício não reaproveitável (purga) gerado por cada operador/máquina.' },
    { id: 'p_manut', title: '⏱️ Paradas (Min)', key: 'p_manut', color: ['#ef4444', '#f59e0b', '#64748b'], formatter: formatMinutes, description: 'Tempo total em minutos que as máquinas ficaram inoperantes por diversos motivos.' },
    { id: 'comp_prod_ecob', title: '📊 Líquido vs Eco B', key: 'comparison_prod_ecob', color: ['#3b82f6', '#8b5cf6'], formatter: formatWeight, description: 'Comparação proporcional entre o material final produzido e o volume de Eco B gerado.' },
    { id: 'comp_ecob_rec', title: '📊 Eco B vs Reciclado', key: 'comparison_ecob_recycled', color: ['#f59e0b', '#10b981'], formatter: formatWeight, description: 'Análise entre o material de descarte (Eco B) gerado na extrusão e o volume recuperado pela Erema.' },
    { id: 'comp_prod_borra', title: '📊 Líquido vs Borra', key: 'comparison_prod_borra', color: ['#3b82f6', '#ef4444'], formatter: formatWeight, description: 'Análise de eficiência comparando o peso líquido produzido contra o desperdício total.' },
    { id: 'comp_time', title: '⏱️ Tempo Máquina (Min)', key: 'comparison_time', color: ['#3b82f6', '#ef4444'], formatter: formatMinutes, description: 'Indicador de disponibilidade: tempo de operação ativa vs tempo total de paradas.' }
  ];

  const visibleCharts = useMemo(() => {
    if (selectedChartId === null) return [];
    if (selectedChartId === 'all') return allCharts;
    return allCharts.filter(c => c.id === selectedChartId);
  }, [selectedChartId]);

  if (!isAuthenticated) {
    return (
      <div className="min-h-screen bg-slate-900 flex items-center justify-center p-4">
        <div className="bg-white rounded-3xl p-8 w-full max-w-md shadow-2xl space-y-8">

            <div className="w-16 h-16 bg-blue-600 rounded-2xl flex items-center justify-center text-white mx-auto shadow-lg mb-4">

            <p className="text-slate-400 text-[10px] font-bold uppercase tracking-widest">Painel de Acesso</p>
          </div>
          <button onClick={() => { localStorage.setItem('manupackaging_session', 'true'); setIsAuthenticated(true); }} className="w-full bg-blue-600 hover:bg-blue-700 text-white font-black py-4 rounded-2xl transition-all shadow-xl shadow-blue-200 uppercase text-xs">Entrar no Sistema</button>
        </div>
      </div>
    );
  }

    <div className="min-h-screen bg-slate-50 pb-24">
      <header className="bg-white px-6 py-4 flex items-center justify-between sticky top-0 z-40 border-b border-slate-100 no-print">
        <div className="flex items-center gap-3">
          <div className="w-10 h-10 bg-blue-600 rounded-xl flex items-center justify-center text-white shadow-lg"><Cpu size={20} /></div>

                      </td>
                    </tr>
                  )}
                </tbody>
              </table>
            </div>
          </div>
        )}
      </main>

      <div className={`fixed inset-0 z-50 transition-all ${isMenuOpen ? 'visible' : 'invisible'}`}>
        <div className={`absolute inset-0 bg-slate-900/40 backdrop-blur-sm transition-opacity ${isMenuOpen ? 'opacity-100' : 'opacity-0'}`} onClick={() => setIsMenuOpen(false)} />
        <div className={`absolute right-0 top-0 bottom-0 w-80 bg-white shadow-2xl transition-transform duration-300 transform p-8 flex flex-col gap-6 ${isMenuOpen ? 'translate-x-0' : 'translate-x-full'}`}>
          <div className="flex items-center justify-between border-b border-slate-100 pb-4">
            <h2 className="text-xl font-black text-slate-800 flex items-center gap-2"><LayoutDashboard size={22} className="text-blue-600" /> Menu</h2>
            <button onClick={() => setConfirmDialog({ isOpen: true, title: 'Encerrar Sessão', message: 'Tem certeza que deseja sair?', onConfirm: () => { localStorage.removeItem('manupackaging_session'); setIsAuthenticated(false); } })} className="p-2 text-slate-400 hover:text-red-500 transition-colors"><LogOut size={24} /></button>
          </div>
          
          <div className="space-y-6 overflow-y-auto custom-scrollbar pr-2 flex-1">
            <div className="space-y-2">
              <label className="text-[10px] font-black text-slate-400 uppercase tracking-widest ml-1 flex items-center gap-2">
                <Calendar size={12} className="text-blue-500" /> Referência Atual
              </label>
              <div className="bg-slate-50 border border-slate-200 rounded-2xl px-5 py-3 flex items-center justify-between shadow-sm">
                <span className="text-sm font-black text-slate-700 capitalize">{getMonthName(dashboardMonth)}</span>
                <div className="flex gap-1">
                   <button onClick={() => changeMonth(-1)} className="p-1 hover:bg-white rounded-lg text-slate-400 hover:text-blue-600 transition-all border border-transparent hover:border-slate-100"><ChevronLeft size={16} /></button>
                   <button onClick={() => changeMonth(1)} className="p-1 hover:bg-white rounded-lg text-slate-400 hover:text-blue-600 transition-all border border-transparent hover:border-slate-100"><ChevronRight size={16} /></button>
                </div>
              </div>
            </div>

            <div className="space-y-3">
              <label className="text-[10px] font-black text-slate-400 uppercase tracking-widest ml-1">Navegação Principal</label>
              <div className="grid grid-cols-1 gap-2">
                <button 
                  onClick={() => { setActiveTab('dashboard'); setIsMenuOpen(false); }}
                  className={`flex items-center gap-4 px-5 py-4 rounded-2xl text-xs font-black uppercase transition-all ${activeTab === 'dashboard' ? 'bg-blue-600 text-white shadow-lg shadow-blue-100' : 'bg-slate-50 text-slate-500 hover:bg-slate-100'}`}
                >
                  <BarChart3 size={20} /> Painel Dashboard
                </button>
                <button 
                  onClick={() => { setActiveTab('reports'); setIsMenuOpen(false); }}
                  className={`flex items-center gap-4 px-5 py-4 rounded-2xl text-xs font-black uppercase transition-all ${activeTab === 'reports' ? 'bg-blue-600 text-white shadow-lg shadow-blue-100' : 'bg-slate-50 text-slate-500 hover:bg-slate-100'}`}
                >
                  <FileText size={20} /> Relatórios Analíticos
                </button>
              </div>
            </div>

            <div className="space-y-4 pt-4 border-t border-slate-100">
              <div className="space-y-2">
                <label className="text-[10px] font-black text-slate-400 uppercase tracking-widest ml-1 flex items-center gap-2"><Settings size={12} /> Configuração de Meta</label>
                <div className="relative">
                  <input 
                    type="number" 
                    value={goalValue} 
                    onChange={(e) => setGoalValue(parseFloat(e.target.value) || 0)} 
                    className="w-full bg-slate-50 border border-slate-200 rounded-2xl px-5 py-3 text-sm font-black text-blue-600 outline-none focus:ring-2 focus:ring-blue-500/10" 
                  />
                  <span className="absolute right-4 top-1/2 -translate-y-1/2 text-[10px] font-black text-slate-400 uppercase">KG</span>
                </div>
              </div>

              <div className="space-y-2">
                <label className="text-[10px] font-black text-slate-400 uppercase tracking-widest ml-1">Equipe</label>
                <button 
                  onClick={() => { setIsOperatorModalOpen(true); setIsMenuOpen(false); }}
                  className="w-full flex items-center justify-between px-5 py-4 bg-slate-50 border border-slate-200 rounded-2xl text-[11px] font-black uppercase text-slate-600 hover:bg-slate-100 transition-all group"
                >
                  <div className="flex items-center gap-3">
                    <Users size={18} className="text-blue-500" />
                    Gerenciar Operadores
                  </div>
                  <Plus size={16} className="text-slate-400 group-hover:text-blue-600 transition-colors" />
                </button>
              </div>
            </div>

            <div className="pt-4 border-t border-slate-100 space-y-3">
              <button 
                onClick={handleSaveDashboardPDF} 
                disabled={isGeneratingPDF || activeTab !== 'dashboard'}
                className={`w-full flex items-center justify-center gap-3 py-4 rounded-2xl text-[10px] font-black uppercase transition-all ${activeTab !== 'dashboard' ? 'bg-slate-50 text-slate-300 cursor-not-allowed' : 'bg-slate-800 text-white shadow-lg shadow-slate-200'}`}
              >
                <FileDown size={16} /> {isGeneratingPDF ? 'Gerando...' : 'Exportar PDF Dashboard'}
              </button>
              <div className="grid grid-cols-2 gap-2">
                <button onClick={() => {
                  const data = { productionData, shifts, goalValue, operators };
                  const blob = new Blob([JSON.stringify(data)], { type: 'application/json' });
                  const url = URL.createObjectURL(blob);
                  const a = document.createElement('a'); a.href = url; a.download = 'backup_manupackaging.json'; a.click();
                }} className="flex flex-col items-center justify-center gap-2 bg-slate-100 py-3 rounded-2xl text-[9px] font-black uppercase text-slate-600 hover:bg-slate-200 transition-colors"><Download size={14} /> Exportar</button>
                <button onClick={() => fileInputRef.current?.click()} className="flex flex-col items-center justify-center gap-2 bg-blue-50 py-3 rounded-2xl text-[9px] font-black uppercase text-blue-600 hover:bg-blue-100 transition-colors"><Upload size={14} /> Importar</button>
              </div>
            </div>
          </div>
        </div>
      </div>

      {confirmDialog.isOpen && (
        <div className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-sm animate-in fade-in duration-200">
          <div className="bg-white rounded-3xl w-full max-sm shadow-2xl p-6 space-y-6 animate-in zoom-in-95 duration-200">
            <div className="flex flex-col items-center text-center space-y-4">
              <div className="w-16 h-16 bg-red-50 text-red-600 rounded-full flex items-center justify-center shadow-inner">
                <AlertCircle size={32} />
              </div>
              <div className="space-y-2">
                <h3 className="text-lg font-black text-slate-800 uppercase tracking-tight">{confirmDialog.title}</h3>
                <p className="text-xs text-slate-500 font-medium leading-relaxed px-4">{confirmDialog.message}</p>
              </div>
            </div>
            <div className="flex gap-3 pt-2">
              <button 
                onClick={() => setConfirmDialog(prev => ({ ...prev, isOpen: false }))}
                className="flex-1 px-4 py-3 bg-slate-50 text-slate-500 rounded-2xl font-black text-[10px] uppercase hover:bg-slate-100 transition-all"
              >
                Cancelar
              </button>
              <button 
                onClick={confirmDialog.onConfirm}
                className="flex-1 px-4 py-3 bg-red-600 text-white rounded-2xl font-black text-[10px] uppercase shadow-lg shadow-red-200 hover:bg-red-700 transition-all"
              >
                Confirmar
              </button>
            </div>
          </div>
        </div>
      )}

      <input type="file" ref={fileInputRef} onChange={handleImport} className="hidden" accept=".json" />
      <LaunchModal isOpen={isModalOpen} onClose={() => { setIsModalOpen(false); setEditingEntry(null); }} onSave={handleSaveEntry} operators={operators} activeMachines={['Cast 1', 'Cast 2', 'Erema 1']} availableShifts={shifts} initialData={editingEntry} />
      <ShiftModal isOpen={isShiftModalOpen} onClose={() => setIsShiftModalOpen(false)} onSave={(s) => setShifts(prev => [...prev, { ...s, id: Math.random().toString() }])} />
      <OperatorModal isOpen={isOperatorModalOpen} onClose={() => setIsOperatorModalOpen(false)} operators={operators} onUpdate={setOperators} />
    </div>
  );
};

export default App;
