const renderCustomizedLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) => {
  if (!percent || percent < 0.01) return null;
  const RADIAN = Math.PI / 180;
  const radius = outerRadius * 1.25;
  const x = cx + radius * Math.cos(-midAngle * RADIAN);
  const y = cy + radius * Math.sin(-midAngle * RADIAN);
  return (
    <text x={x} y={y} fill="#64748b" textAnchor={x > cx ? 'start' : 'end'} dominantBaseline="central" className="text-[10px] font-black">
      {`${(percent * 100).toFixed(1)}%`}
    </text>
  );
};

            </div>

            <div className="p-6 bg-slate-50 border-t border-slate-100">
              <button onClick={() => { localStorage.removeItem('manupackaging_session'); setIsAuthenticated(false); }} className="w-full p-4 bg-white border border-red-100 text-red-600 rounded-2xl text-[10px] font-black uppercase flex items-center justify-center gap-2 hover:bg-red-50 transition-all">
                <LogOut size={16}/> Sair do Sistema
              </button>
            </div>
          </div>
        </div>
      )}

      {confirmDialog.isOpen && (
        <div className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm">
          <div className="bg-white rounded-3xl p-8 w-full max-sm shadow-2xl space-y-6 animate-in zoom-in-95 duration-200">
            <div className="flex flex-col items-center text-center space-y-4">
              <div className="w-16 h-16 bg-red-100 text-red-600 rounded-2xl flex items-center justify-center"><AlertTriangle size={32} /></div>
              <h3 className="text-xl font-black text-slate-800 uppercase tracking-tight">{confirmDialog.title}</h3>
              <p className="text-slate-500 text-sm font-medium">{confirmDialog.message}</p>
            </div>
            <div className="flex gap-3">
              <button onClick={() => setConfirmDialog(prev => ({...prev, isOpen: false}))} className="flex-1 px-4 py-3 bg-slate-100 text-slate-600 rounded-xl font-black text-[10px] uppercase hover:bg-slate-200 transition-all">Cancelar</button>
              <button onClick={confirmDialog.onConfirm} className="flex-1 px-4 py-3 bg-red-600 text-white rounded-xl font-black text-[10px] uppercase shadow-lg shadow-red-100 hover:bg-red-700 transition-all">Confirmar</button>
            </div>
          </div>
        </div>
      )}

      <LaunchModal isOpen={isModalOpen} onClose={() => { setIsModalOpen(false); setEditingEntry(null); }} onSave={handleSaveEntry} operators={operators} activeMachines={activeMachines} availableShifts={shifts} initialData={editingEntry} />
      <OperatorModal isOpen={isOperatorModalOpen} onClose={() => setIsOperatorModalOpen(false)} operators={operators} onUpdate={setOperators} />
      <ShiftModal isOpen={isShiftModalOpen} onClose={() => setIsShiftModalOpen(false)} onSave={(newShift) => setShifts(prev => [...prev, { ...newShift, id: Math.random().toString(36).substr(2, 9) }])} />
    </div>
  );
};

export default App;
