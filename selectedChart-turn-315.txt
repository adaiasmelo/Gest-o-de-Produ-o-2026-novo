const renderCustomizedLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) => {
  if (!percent || percent < 0.01) return null;
  const RADIAN = Math.PI / 180;
  const radius = outerRadius * 1.25;
  const x = cx + radius * Math.cos(-midAngle * RADIAN);
  const y = cy + radius * Math.sin(-midAngle * RADIAN);
  return (
    <text x={x} y={y} fill="#64748b" textAnchor={x > cx ? 'start' : 'end'} dominantBaseline="central" className="text-[10px] font-black">
      {`${(percent * 100).toFixed(1)}%`}
    </text>
  );
};

        const timeLeft = animationEnd - Date.now();
        if (timeLeft <= 0) return clearInterval(interval);
        const particleCount = 50 * (timeLeft / duration);
        confetti({ ...defaults, particleCount, origin: { x: randomInRange(0.1, 0.3), y: Math.random() - 0.2 } });

    return (
      <div className="min-h-screen bg-slate-900 flex items-center justify-center p-4">
        <div className="bg-white rounded-3xl p-8 w-full max-w-md shadow-2xl space-y-8">
          <div className="text-center space-y-2">
            <div className="w-16 h-16 bg-blue-600 rounded-2xl flex items-center justify-center text-white mx-auto shadow-lg mb-4">
              <ShieldCheck size={32} />
            </div>
            <h1 className="text-2xl font-black text-slate-800">Manupackaging</h1>
            <p className="text-slate-400 text-[10px] font-bold uppercase tracking-widest">Painel de Acesso</p>
          </div>
          <button onClick={() => { localStorage.setItem('manupackaging_session', 'true'); setIsAuthenticated(true); }} className="w-full bg-blue-600 hover:bg-blue-700 text-white font-black py-4 rounded-2xl transition-all shadow-xl shadow-blue-200 uppercase text-xs">Entrar no Sistema</button>
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-slate-50 pb-24">
      <header className="bg-white px-6 py-4 flex items-center justify-between sticky top-0 z-40 border-b border-slate-100 no-print">
        <div className="flex items-center gap-3">
          <div className="w-10 h-10 bg-blue-600 rounded-xl flex items-center justify-center text-white shadow-lg"><Cpu size={20} /></div>
          <h1 className="text-lg font-black text-slate-800 tracking-tighter uppercase">Controle de Produção</h1>
        </div>
        <div className="flex items-center gap-2">
          <button 
            onClick={() => { setEditingEntry(null); setIsModalOpen(true); }} 
            className="bg-blue-600 hover:bg-blue-700 text-white p-2.5 rounded-xl shadow-lg shadow-blue-100 transition-all flex items-center justify-center"
            title="Novo Lançamento"
          >
            <Plus size={24} />
          </button>
          <button onClick={() => setIsMenuOpen(true)} className="p-2.5 text-slate-400 hover:bg-slate-100 rounded-xl transition-all relative">
            <Settings size={20} />
            <span className="absolute top-1 right-1 w-2 h-2 bg-blue-600 rounded-full border-2 border-white"></span>
          </button>
        </div>
      </header>

      <div className="max-w-6xl mx-auto px-4 mt-6 no-print">
        <div className="flex items-center p-1 bg-slate-200/50 rounded-2xl w-fit">
          <button 
            onClick={() => setActiveTab('dashboard')}
            className={`flex items-center gap-2 px-6 py-2.5 rounded-xl transition-all duration-300 ${
              activeTab === 'dashboard' 
              ? 'bg-white text-blue-600 shadow-md' 
              : 'text-slate-500 hover:text-slate-700'
            }`}
          >
            <BarChart3 size={18} />
            <span className="text-[11px] font-black uppercase tracking-tighter">Painel de Indicadores</span>
          </button>
          <button 
            onClick={() => setActiveTab('reports')}
            className={`flex items-center gap-2 px-6 py-2.5 rounded-xl transition-all duration-300 ${
              activeTab === 'reports' 
              ? 'bg-white text-blue-600 shadow-md' 
              : 'text-slate-500 hover:text-slate-700'
            }`}
          >
            <FileText size={18} />
            <span className="text-[11px] font-black uppercase tracking-tighter">Relatórios Analíticos</span>
          </button>
        </div>
      </div>

      <main className="max-w-[100vw] mx-auto p-4 space-y-6 overflow-x-hidden">
        {activeTab === 'dashboard' ? (
          <div ref={dashboardRef} className="space-y-6 max-w-6xl mx-auto">
            
            <div className="bg-white rounded-3xl p-5 shadow-sm border border-slate-100 flex flex-wrap items-center gap-4 no-print">
              <div className="flex items-center gap-2 mr-2">
                <div className="w-8 h-8 bg-blue-50 text-blue-600 rounded-lg flex items-center justify-center"><Filter size={16} /></div>
                <span className="text-[10px] font-black uppercase text-slate-400 tracking-widest">Filtros</span>
              </div>
              
              <div className="flex-1 min-w-[180px]">
                <label className="text-[9px] font-black text-slate-400 uppercase tracking-widest ml-1 mb-1 block">Operador</label>
                <select 
                  value={dashOperator} 
                  onChange={(e) => setDashOperator(e.target.value)}
                  className="w-full bg-slate-50 border border-slate-100 rounded-xl px-4 py-2.5 text-[11px] font-bold text-slate-700 outline-none"
                >
                  <option value="">Todos os Operadores</option>
                  {operators.map(op => <option key={op} value={op}>{op}</option>)}
                </select>
              </div>

              <div className="flex-1 min-w-[180px]">
                <label className="text-[9px] font-black text-slate-400 uppercase tracking-widest ml-1 mb-1 block">Máquina</label>
                <select 
                  value={dashMachine} 
                  onChange={(e) => setDashMachine(e.target.value)}
                  className="w-full bg-slate-50 border border-slate-100 rounded-xl px-4 py-2.5 text-[11px] font-bold text-slate-700 outline-none"
                >
                  <option value="">Todas as Máquinas</option>
                  {activeMachines.map(m => <option key={m} value={m}>{m}</option>)}
                </select>
              </div>

              <div className="flex-1 min-w-[150px]">
                <label className="text-[9px] font-black text-slate-400 uppercase tracking-widest ml-1 mb-1 block">Data Inicial</label>
                <input 
                  type="date" 
                  value={dashStartDate} 
                  onChange={(e) => setDashStartDate(e.target.value)}
                  className="w-full bg-slate-50 border border-slate-100 rounded-xl px-4 py-2.5 text-[11px] font-bold text-slate-700 outline-none"
                />
              </div>

              <div className="flex-1 min-w-[150px]">
                <label className="text-[9px] font-black text-slate-400 uppercase tracking-widest ml-1 mb-1 block">Data Final</label>
                <input 
                  type="date" 
                  value={dashEndDate} 
                  onChange={(e) => setDashEndDate(e.target.value)}
                  className="w-full bg-slate-50 border border-slate-100 rounded-xl px-4 py-2.5 text-[11px] font-bold text-slate-700 outline-none"
                />
              </div>

              {(dashOperator || dashMachine || dashStartDate || dashEndDate) && (
                <button 
                  onClick={() => { setDashOperator(''); setDashMachine(''); setDashStartDate(''); setDashEndDate(''); }}
                  className="self-end px-4 py-2.5 text-red-500 hover:bg-red-50 rounded-xl text-[10px] font-black uppercase transition-all"
                >
                  Limpar
                </button>
              )}
            </div>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
              <button 
                onClick={() => { if (!isRangeActive) setSelectedTimeframe('month'); }} 
                className={`flex flex-col p-6 rounded-2xl border transition-all duration-300 h-full ${
                  (selectedTimeframe === 'month' && !isRangeActive) || (isRangeActive)
                  ? 'bg-blue-600 border-blue-500 shadow-xl text-white' 
                  : 'bg-white border-slate-100 text-slate-800 shadow-sm'
                } ${totals.month >= goalValue && !isRangeActive ? 'animate-goal-reached' : ''}`}
              >
                <div className="flex items-center justify-between mb-4">
                  <div className={`p-3 rounded-xl ${(selectedTimeframe === 'month' && !isRangeActive) || (isRangeActive) ? 'bg-white/20' : 'bg-blue-50 text-blue-600'}`}>
                    {isRangeActive ? <Calendar size={24} /> : <Target size={24} />}
                  </div>
                  <span className={`text-[10px] font-black uppercase tracking-widest ${(selectedTimeframe === 'month' && !isRangeActive) || (isRangeActive) ? 'text-blue-100' : 'text-slate-400'}`}>
                    {isRangeActive ? 'Produção no Período' : 'Meta Mensal'}
                  </span>
                </div>
                <div className="space-y-3">
                  <div className="text-2xl font-black">{formatWeight(isRangeActive ? totals.range : totals.month)}</div>
                  {!isRangeActive && (
                    <div className="w-full bg-black/10 h-2 rounded-full overflow-hidden">
                      <div className={`h-full bg-white rounded-full`} style={{ width: `${Math.min((totals.month / (goalValue || 1)) * 100, 100)}%` }}></div>
                    </div>
                  )}
                  <p className="text-[9px] font-bold uppercase tracking-tighter opacity-80">
                    {!isRangeActive ? `${((totals.month / goalValue) * 100).toFixed(1)}% atingido` : 'Total no intervalo selecionado'}
                  </p>
                </div>
              </button>

              <button 
                onClick={() => { if (!isRangeActive) setSelectedTimeframe('yesterday'); }} 
                className={`flex flex-col p-6 rounded-2xl border transition-all duration-300 h-full ${
                  selectedTimeframe === 'yesterday' && !isRangeActive
                  ? 'bg-emerald-600 border-emerald-500 shadow-xl text-white' 
                  : 'bg-white border-slate-100 text-slate-800 shadow-sm'
                }`}
              >
                <div className="flex items-center justify-between mb-4">
                  <div className={`p-3 rounded-xl ${selectedTimeframe === 'yesterday' && !isRangeActive ? 'bg-white/20' : 'bg-emerald-50 text-emerald-600'}`}>
                    <TrendingUp size={24} />
                  </div>
                  <span className={`text-[10px] font-black uppercase tracking-widest ${selectedTimeframe === 'yesterday' && !isRangeActive ? 'text-emerald-100' : 'text-slate-400'}`}>
                    Ontem
                  </span>
                </div>
                <div className="space-y-3">
                  <div className="text-2xl font-black">{formatWeight(totals.yesterday)}</div>
                  <p className="text-[9px] font-bold uppercase tracking-tighter opacity-80">Produção do Dia Anterior</p>
                </div>
              </button>

              <button 
                onClick={() => { if (!isRangeActive) setSelectedTimeframe('week'); }} 
                className={`flex flex-col p-6 rounded-2xl border transition-all duration-300 h-full ${
                  selectedTimeframe === 'week' && !isRangeActive
                  ? 'bg-violet-600 border-violet-500 shadow-xl text-white' 
                  : 'bg-white border-slate-100 text-slate-800 shadow-sm'
                }`}
              >
                <div className="flex items-center justify-between mb-4">
                  <div className={`p-3 rounded-xl ${selectedTimeframe === 'week' && !isRangeActive ? 'bg-white/20' : 'bg-violet-50 text-violet-600'}`}>
                    <CalendarDays size={24} />
                  </div>
                  <span className={`text-[10px] font-black uppercase tracking-widest ${selectedTimeframe === 'week' && !isRangeActive ? 'text-violet-100' : 'text-slate-400'}`}>
                    Esta Semana
                  </span>
                </div>
                <div className="space-y-3">
                  <div className="text-2xl font-black">{formatWeight(totals.week)}</div>
                  <p className="text-[9px] font-bold uppercase tracking-tighter opacity-80">Acumulado da Semana</p>
                </div>
              </button>
            </div>

            <div className="bg-white rounded-3xl p-6 shadow-sm border border-slate-100 space-y-4 no-print">
              <div className="relative group">
                <select 
                  value={selectedChartId || ''} 
                  onChange={(e) => setSelectedChartId(e.target.value === '' ? null : e.target.value)}
                  className="w-full bg-slate-50 border border-slate-200 rounded-2xl px-5 py-4 text-xs font-black text-slate-700 outline-none appearance-none uppercase tracking-widest pr-12"
                >
                  <option value="all">📊 Mostrar Todos os Gráficos</option>
                  {allCharts.map(chart => (
                    <option key={chart.id} value={chart.id}>{chart.title}</option>
                  ))}
                  <option value="">Nenhum gráfico selecionado...</option>
                </select>
                <div className="absolute right-5 top-1/2 -translate-y-1/2 pointer-events-none text-slate-400">
                  <ChevronDown size={20} />
                </div>
              </div>
            </div>

            <div className={`grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6 transition-all duration-500 animate-in slide-in-from-bottom-4 ${selectedChartId !== 'all' ? 'lg:grid-cols-1 max-w-2xl mx-auto' : ''}`}>
              {visibleCharts.map((chart, idx) => {
                let chartData: any[] = [];
                let dataKey = chart.key;
                const isOperatorChart = !chart.key.startsWith('comparison_');
                
                if (chart.key === 'comparison_prod_ecob') {
                  const totalProd = filteredDataForCharts.reduce((acc, c) => acc + (c.netWeight || 0), 0);
                  const totalEcoB = filteredDataForCharts.reduce((acc, c) => acc + (c.ecoBP || 0) + (c.ecoBM || 0), 0);
                  chartData = [{ name: 'Produção', value: totalProd }, { name: 'Eco B', value: totalEcoB }].filter(d => d.value > 0);
                  dataKey = 'value';
                } else if (chart.key === 'comparison_time') {
                  let totalAvailable = 0;
                  let totalDowntime = 0;
                  filteredDataForCharts.forEach(item => {
                     const shiftDuration = getShiftDuration(item.shift);
                     totalAvailable += shiftDuration;
                     totalDowntime += (item.manutencaoMin || 0) + (item.processoMin || 0) + (item.outrosMin || 0);
                  });
                  const totalRunning = Math.max(0, totalAvailable - totalDowntime);
                  chartData = [{ name: 'Operação', value: totalRunning }, { name: 'Parada', value: totalDowntime }].filter(d => d.value > 0);
                  dataKey = 'value';
                } else {
                  chartData = stats.filter(s => (s[chart.key as keyof OperatorStat] as number) > 0).sort((a, b) => (b[chart.key as keyof OperatorStat] as number) - (a[chart.key as keyof OperatorStat] as number));
                }
                const cardId = `chart-card-${idx}`;
                return (
                  <div key={idx} id={cardId} className="bg-white rounded-3xl p-6 shadow-sm border border-slate-100 min-h-[460px] flex flex-col group/card transition-all hover:shadow-xl">
                    <div className="flex justify-between items-start mb-4">
                      <h3 className="font-black text-xs uppercase tracking-widest text-slate-800">{chart.title}</h3>
                      <button 
                        onClick={() => handleDownloadChartPNG(cardId, chart.title)}
                        className="p-1.5 text-slate-300 hover:text-blue-600 no-print"
                      >
                        <ImageIcon size={16} />
                      </button>
                    </div>
                    <div className="flex-1">
                      {chartData.length > 0 ? (
                        <ResponsiveContainer width="100%" height="100%">
                          <PieChart>
                            <Pie data={chartData} cx="50%" cy="50%" innerRadius={60} outerRadius={85} dataKey={isOperatorChart ? chart.key : dataKey} nameKey="name" label={renderCustomizedLabel} paddingAngle={2}>
                              {chartData.map((_, i) => <Cell key={i} fill={chart.color[i % chart.color.length]} />)}
                            </Pie>
                            <Tooltip formatter={(val: any) => chart.formatter(val)} />
                            <Legend iconType="circle" formatter={(value, entry: any) => <span className="text-[10px] font-bold text-slate-600">{value} ({chart.formatter(entry.payload.value || entry.payload[chart.key])})</span>} />
                          </PieChart>
                        </ResponsiveContainer>
                      ) : (
                        <div className="h-full flex items-center justify-center text-slate-300 font-bold text-xs uppercase text-center">Sem dados</div>
                      )}
                    </div>
                  </div>
                );
              })}
            </div>
          </div>
        ) : (
          <div className="bg-white rounded-3xl shadow-sm border border-slate-100 overflow-hidden">
            <div className="p-4 border-b border-slate-100 flex flex-wrap items-center justify-between gap-4">
              <div className="flex flex-wrap items-center gap-4 flex-1 min-w-[300px]">
                <input type="date" value={filterStartDate} onChange={(e) => setFilterStartDate(e.target.value)} className="bg-slate-50 border border-slate-200 rounded-xl px-3 py-2 text-xs font-bold focus:outline-none" />
                <input type="date" value={filterEndDate} onChange={(e) => setFilterEndDate(e.target.value)} className="bg-slate-50 border border-slate-200 rounded-xl px-3 py-2 text-xs font-bold focus:outline-none" />
                <div className="relative flex-1">
                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" size={16} />
                  <input type="text" value={searchTerm} onChange={(e) => setSearchTerm(e.target.value)} placeholder="Pesquisar..." className="w-full pl-10 pr-4 py-2 bg-slate-50 border border-slate-200 rounded-xl text-xs font-bold focus:outline-none" />
                </div>
              </div>
              <div className="flex items-center gap-2 no-print">
                <button onClick={handleExportCSV} className="p-2.5 bg-emerald-600 text-white rounded-xl flex items-center gap-2 text-[10px] font-black uppercase"><FileSpreadsheet size={18} /> CSV</button>
                <button onClick={() => window.print()} className="p-2.5 bg-slate-800 text-white rounded-xl"><Printer size={18} /></button>
              </div>
            </div>
            <div className="overflow-x-auto custom-scrollbar">
              <table className="w-full text-left border-collapse min-w-[1500px]">
                <thead>
                  <tr className="bg-slate-50 text-[9px] font-black text-slate-400 uppercase tracking-widest border-b border-slate-100">
                    <th className="px-4 py-4 text-center no-print">
                      <button onClick={() => toggleSelectAll(filteredReports)}><Square size={16} /></button>
                    </th>
                    <th className="px-4 py-4">Data</th>
                    <th className="px-4 py-4">Operador</th>
                    <th className="px-4 py-4">Máquina</th>
                    <th className="px-4 py-4">Turno</th>
                    <th className="px-4 py-4 text-right">Peso Bruto</th>
                    <th className="px-4 py-4 text-right">Peso Líquido</th>
                    <th className="px-4 py-4 text-right">Eco A</th>
                    <th className="px-4 py-4 text-right">Borra</th>
                    <th className="px-4 py-4 text-right no-print">Ações</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100 text-[10px] font-bold">
                  {filteredReports.map((e) => (
                    <tr key={e.id} className="hover:bg-slate-50">
                      <td className="px-4 py-4 text-center no-print">
                        <input type="checkbox" checked={selectedIds.includes(e.id)} onChange={() => toggleSelectId(e.id)} />
                      </td>
                      <td className="px-4 py-4">{e.date}</td>
                      <td className="px-4 py-4">{e.operator}</td>
                      <td className="px-4 py-4 uppercase">{e.machine}</td>
                      <td className="px-4 py-4">{e.shift}</td>
                      <td className="px-4 py-4 text-right">{e.grossWeight}</td>
                      <td className="px-4 py-4 text-right text-blue-600 font-black">{e.netWeight}</td>
                      <td className="px-4 py-4 text-right">{e.ecoA}</td>
                      <td className="px-4 py-4 text-right text-red-500">{e.borraTotal}</td>
                      <td className="px-4 py-4 text-right no-print">
                        <button onClick={() => { setEditingEntry(e); setIsModalOpen(true); }} className="text-blue-500 p-1.5"><Edit2 size={12} /></button>
                        <button onClick={() => handleDeleteReportItem(e.id)} className="text-red-400 p-1.5"><Trash2 size={12} /></button>
                      </td>
                    </tr>
                  ))}
                  <tr className="bg-slate-800 text-white font-black">
                    <td colSpan={5} className="px-4 py-3 text-right uppercase text-[10px]">TOTAIS:</td>
                    <td className="px-4 py-3 text-right">{reportSums.grossWeight}</td>
                    <td className="px-4 py-3 text-right text-blue-300">{reportSums.netWeight}</td>
                    <td className="px-4 py-3 text-right">{reportSums.ecoA}</td>
                    <td className="px-4 py-3 text-right text-red-300">{reportSums.borraTotal}</td>
                    <td className="no-print"></td>
                  </tr>
                </tbody>
              </table>
            </div>
          </div>
        )}
      </main>

      {isMenuOpen && (
        <div className="fixed inset-0 z-50 flex justify-end">
          <div className="absolute inset-0 bg-slate-900/40 backdrop-blur-sm" onClick={() => setIsMenuOpen(false)}></div>
          <div className="relative w-80 bg-white shadow-2xl p-8 space-y-8 animate-in slide-in-from-right duration-300">
            <h2 className="text-xl font-black border-b pb-4 uppercase tracking-tighter flex items-center gap-2"><Settings size={22} className="text-blue-600" /> Menu</h2>
            <div className="space-y-4">
              <label className="text-[10px] font-black uppercase tracking-widest text-slate-400">Meta Mensal (KG)</label>
              <input type="number" value={goalValue} onChange={(e) => setGoalValue(Number(e.target.value))} className="w-full bg-slate-50 border p-3 rounded-2xl font-black text-blue-600 outline-none" />
            </div>
            <button onClick={() => { setIsOperatorModalOpen(true); setIsMenuOpen(false); }} className="w-full flex items-center justify-between p-4 bg-slate-50 rounded-2xl text-[10px] font-black uppercase">Gerenciar Equipe <Users size={16}/></button>
            <button onClick={handleSaveDashboardPDF} className="w-full p-4 bg-slate-800 text-white rounded-2xl text-[10px] font-black uppercase">Exportar PDF Dashboard</button>
            <button onClick={() => { localStorage.removeItem('manupackaging_session'); setIsAuthenticated(false); }} className="w-full p-4 bg-red-50 text-red-600 rounded-2xl text-[10px] font-black uppercase flex items-center justify-center gap-2"><LogOut size={16}/> Sair</button>
          </div>
        </div>
      )}

      <LaunchModal isOpen={isModalOpen} onClose={() => { setIsModalOpen(false); setEditingEntry(null); }} onSave={handleSaveEntry} operators={operators} activeMachines={activeMachines} availableShifts={shifts} initialData={editingEntry} />
      <OperatorModal isOpen={isOperatorModalOpen} onClose={() => setIsOperatorModalOpen(false)} operators={operators} onUpdate={setOperators} />
    </div>
  );
};

export default App;
