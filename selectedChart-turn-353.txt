const renderCustomizedLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) => {
  if (!percent || percent < 0.01) return null;
  const RADIAN = Math.PI / 180;
  const radius = outerRadius * 1.25;
  const x = cx + radius * Math.cos(-midAngle * RADIAN);
  const y = cy + radius * Math.sin(-midAngle * RADIAN);
  return (
    <text x={x} y={y} fill="#64748b" textAnchor={x > cx ? 'start' : 'end'} dominantBaseline="central" className="text-[10px] font-black">
      {`${(percent * 100).toFixed(1)}%`}
    </text>
  );
};

  };

  const groupedPersonnel = useMemo(() => {
    const sectorsOrder = ['Extrusão', 'Erema', 'Fita', 'Ghezzi', 'Lintech', 'Wutec'];
    const groups: Record<string, any> = {};
    
    employees.forEach(emp => {
      const sector = emp.sector || 'Outros';
      if (!groups[sector]) groups[sector] = {};
      
      const machine = emp.machine || 'Geral';
      if (!groups[sector][machine]) groups[sector][machine] = {};
      
      const shift = emp.group || 'Geral';
      if (!groups[sector][machine][shift]) groups[sector][machine][shift] = [];
      
      groups[sector][machine][shift].push(emp);
    });
    
    return groups;
  }, [employees]);

  const EmployeeCard = ({ emp }: { emp: Employee }) => (
    <div className="bg-white rounded-2xl p-4 shadow-sm border border-slate-100 hover:shadow-md transition-all group relative overflow-hidden">
      <div className="absolute top-0 right-0 p-2 flex gap-1 no-print opacity-0 group-hover:opacity-100 transition-opacity">
        <button onClick={() => { setEditingEmployee(emp); setIsEmployeeModalOpen(true); }} className="p-1.5 text-slate-300 hover:text-blue-600 hover:bg-blue-50 rounded-lg"><Edit2 size={12} /></button>
        <button onClick={() => handleDeleteEmployee(emp.id)} className="p-1.5 text-slate-300 hover:text-red-600 hover:bg-red-50 rounded-lg"><Trash2 size={12} /></button>
      </div>

      <div className="flex items-center gap-3 mb-3">
        <div className={`w-10 h-10 rounded-xl flex items-center justify-center font-black text-sm ${emp.status === 'Ativo' ? 'bg-blue-50 text-blue-600' : emp.status === 'Férias' ? 'bg-orange-50 text-orange-600' : 'bg-red-50 text-red-600'}`}>
          {emp.name ? emp.name.substring(0, 2).toUpperCase() : '??'}
        </div>
        <div className="flex-1 min-w-0">
          <h4 className="font-black text-slate-800 text-xs truncate leading-none mb-1.5">{emp.name || 'Vaga Aberta'}</h4>
          <span className={`inline-block px-2 py-0.5 rounded-full text-[8px] font-black uppercase tracking-widest ${emp.status === 'Ativo' ? 'bg-emerald-100 text-emerald-700' : emp.status === 'Férias' ? 'bg-orange-100 text-orange-700' : 'bg-red-100 text-red-700'}`}>{emp.status}</span>
        </div>
      </div>

      <div className="space-y-1.5">
         <div className="flex items-center justify-between">
           <span className="text-[8px] font-black text-slate-400 uppercase tracking-widest">Cargo</span>
           <span className="text-[10px] font-bold text-slate-600">{emp.role}</span>
         </div>
         {emp.status === 'Férias' && (
           <div className="mt-2 pt-2 border-t border-slate-50 flex items-center justify-between">
              <span className="text-[8px] font-black text-orange-400 uppercase tracking-widest">Retorno</span>
              <span className="text-[10px] font-black text-orange-600">{emp.returnDate || '--'}</span>
           </div>
         )}
      </div>
    </div>
  );

  if (!isAuthenticated) {
    return (
      <div className="min-h-screen bg-slate-900 flex items-center justify-center p-4">
        <div className="bg-white rounded-3xl p-8 w-full max-w-md shadow-2xl space-y-8">
          <div className="text-center space-y-2">
            <div className="w-16 h-16 bg-blue-600 rounded-2xl flex items-center justify-center text-white mx-auto shadow-lg mb-4">
              <ShieldCheck size={32} />
            </div>
            <h1 className="text-2xl font-black text-slate-800">Manupackaging</h1>
            <p className="text-slate-400 text-[10px] font-bold uppercase tracking-widest">Painel de Acesso</p>
          </div>
          <button onClick={() => { localStorage.setItem('manupackaging_session', 'true'); setIsAuthenticated(true); }} className="w-full bg-blue-600 hover:bg-blue-700 text-white font-black py-4 rounded-2xl transition-all shadow-xl shadow-blue-200 uppercase text-xs">Entrar no Sistema</button>
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-slate-50 pb-24">
      <header className="bg-white px-6 py-4 flex items-center justify-between sticky top-0 z-40 border-b border-slate-100 no-print">
        <div className="flex items-center gap-3">
          <div className="w-10 h-10 bg-blue-600 rounded-xl flex items-center justify-center text-white shadow-lg"><Cpu size={20} /></div>
          <h1 className="text-lg font-black text-slate-800 tracking-tighter uppercase">Controle de Produção</h1>
        </div>
        <div className="flex items-center gap-2">
          <button onClick={() => { setEditingEntry(null); setIsModalOpen(true); }} className="bg-blue-600 hover:bg-blue-700 text-white p-2.5 rounded-xl shadow-lg shadow-blue-100 transition-all flex items-center justify-center"><Plus size={24} /></button>
          <button onClick={() => setIsMenuOpen(true)} className="p-2.5 text-slate-400 hover:bg-slate-100 rounded-xl transition-all relative"><Settings size={20} /></button>
        </div>
      </header>

      <div className="max-w-6xl mx-auto px-4 mt-6 no-print">
        <div className="flex items-center p-1 bg-slate-200/50 rounded-2xl w-fit">
          <button onClick={() => setActiveTab('dashboard')} className={`flex items-center gap-2 px-6 py-2.5 rounded-xl transition-all ${activeTab === 'dashboard' ? 'bg-white text-blue-600 shadow-md' : 'text-slate-500 hover:text-slate-700'}`}>
            <BarChart3 size={18} /><span className="text-[11px] font-black uppercase tracking-tighter">Indicadores</span>
          </button>
          <button onClick={() => setActiveTab('reports')} className={`flex items-center gap-2 px-6 py-2.5 rounded-xl transition-all ${activeTab === 'reports' ? 'bg-white text-blue-600 shadow-md' : 'text-slate-500 hover:text-slate-700'}`}>
            <FileText size={18} /><span className="text-[11px] font-black uppercase tracking-tighter">Relatórios</span>
          </button>
          <button onClick={() => setActiveTab('personnel')} className={`flex items-center gap-2 px-6 py-2.5 rounded-xl transition-all ${activeTab === 'personnel' ? 'bg-white text-blue-600 shadow-md' : 'text-slate-500 hover:text-slate-700'}`}>
            <Users2 size={18} /><span className="text-[11px] font-black uppercase tracking-tighter">Gestão de Pessoal</span>
          </button>
        </div>
      </div>

      <main className="max-w-6xl mx-auto p-4 space-y-8 overflow-x-hidden">
        {activeTab === 'dashboard' ? (
          <div ref={dashboardRef} className="space-y-6">
            <div className="bg-white rounded-3xl p-5 shadow-sm border border-slate-100 flex flex-wrap items-center gap-4 no-print">
              <div className="flex items-center gap-2 mr-2">
                <div className="w-8 h-8 bg-blue-50 text-blue-600 rounded-lg flex items-center justify-center"><Filter size={16} /></div>
                <span className="text-[10px] font-black uppercase text-slate-400 tracking-widest">Filtros</span>
              </div>
              <div className="flex-1 min-w-[180px]">
                <label className="text-[9px] font-black text-slate-400 uppercase tracking-widest ml-1 mb-1 block">Operador</label>
                <select value={dashOperator} onChange={(e) => setDashOperator(e.target.value)} className="w-full bg-slate-50 border border-slate-100 rounded-xl px-4 py-2.5 text-[11px] font-bold text-slate-700 outline-none">
                  <option value="">Todos os Operadores</option>
                  {operators.map(op => <option key={op} value={op}>{op}</option>)}
                </select>
              </div>
              <div className="flex-1 min-w-[180px]">
                <label className="text-[9px] font-black text-slate-400 uppercase tracking-widest ml-1 mb-1 block">Máquina</label>
                <select value={dashMachine} onChange={(e) => setDashMachine(e.target.value)} className="w-full bg-slate-50 border border-slate-100 rounded-xl px-4 py-2.5 text-[11px] font-bold text-slate-700 outline-none">
                  <option value="">Todas as Máquinas</option>
                  {activeMachines.map(m => <option key={m} value={m}>{m}</option>)}
                </select>
              </div>
              <div className="flex-1 min-w-[150px]">
                <label className="text-[9px] font-black text-slate-400 uppercase tracking-widest ml-1 mb-1 block">Data Inicial</label>
                <input type="date" value={dashStartDate} onChange={(e) => setDashStartDate(e.target.value)} className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2.5 text-[11px] font-bold text-slate-700 outline-none" />
              </div>
              <div className="flex-1 min-w-[150px]">
                <label className="text-[9px] font-black text-slate-400 uppercase tracking-widest ml-1 mb-1 block">Data Final</label>
                <input type="date" value={dashEndDate} onChange={(e) => setDashEndDate(e.target.value)} className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2.5 text-[11px] font-bold text-slate-700 outline-none" />
              </div>
              {(dashOperator || dashMachine || dashStartDate || dashEndDate) && (
                <button onClick={() => { setDashOperator(''); setDashMachine(''); setDashStartDate(''); setDashEndDate(''); }} className="self-end px-4 py-2.5 text-red-500 hover:bg-red-50 rounded-xl text-[10px] font-black uppercase">Limpar</button>
              )}
            </div>

            <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
              <button 
                onClick={() => setSelectedTimeframe('month')} 
                className={`flex flex-col p-6 rounded-2xl border transition-all duration-300 h-full ${ 
                  selectedTimeframe === 'month' 
                  ? 'bg-blue-600 border-blue-500 shadow-xl text-white' 
                  : 'bg-white border-slate-100 text-slate-800 shadow-sm' 
                }`}
              >
                <div className="flex items-center justify-between mb-4">
                  <div className={`p-3 rounded-xl ${selectedTimeframe === 'month' ? 'bg-white/20' : 'bg-blue-50 text-blue-600'}`}>
                    <Target size={24} />
                  </div>
                  <div className="text-right">
                    <span className={`text-[10px] font-black uppercase tracking-widest block ${selectedTimeframe === 'month' ? 'text-blue-100' : 'text-slate-400'}`}>Meta Mensal</span>
                    <span className={`text-[9px] font-bold ${selectedTimeframe === 'month' ? 'text-blue-200' : 'text-slate-400'}`}>Alvo: {formatWeight(goalValue)}</span>
                  </div>
                </div>
                <div className="space-y-4">
                  <div>
                    <div className="text-2xl font-black leading-none mb-1">{formatWeight(totals.month)}</div>
                    <p className={`text-[9px] font-bold uppercase tracking-tighter ${selectedTimeframe === 'month' ? 'text-blue-200' : 'text-slate-400'}`}>Produzido até agora</p>
                  </div>
                  <div className="space-y-1.5">
                    <div className="w-full bg-black/10 h-2.5 rounded-full overflow-hidden border border-white/10">
                      <div className={`h-full transition-all duration-700 ${selectedTimeframe === 'month' ? 'bg-white' : 'bg-blue-500'}`} style={{ width: `${Math.min((totals.month / (goalValue || 1)) * 100, 100)}%` }}></div>
                    </div>
                    <div className="flex justify-between items-center text-[10px] font-black uppercase tracking-tighter">
                      <span>{((totals.month / (goalValue || 1)) * 100).toFixed(1)}%</span>
                      <span>Faltam {formatWeight(projection.remainingToGoal)}</span>
                    </div>
                  </div>
                </div>
              </button>
              <button onClick={() => setSelectedTimeframe('yesterday')} className={`flex flex-col p-6 rounded-2xl border transition-all duration-300 h-full ${ selectedTimeframe === 'yesterday' ? 'bg-emerald-600 border-emerald-500 shadow-xl text-white' : 'bg-white border-slate-100 text-slate-800 shadow-sm' }`}>
                <div className="flex items-center justify-between mb-4">
                  <div className={`p-3 rounded-xl ${selectedTimeframe === 'yesterday' ? 'bg-white/20' : 'bg-emerald-50 text-emerald-600'}`}>
                    {isRangeActive ? <Cpu size={24} /> : <TrendingUp size={24} />}
                  </div>
                  <span className={`text-[10px] font-black uppercase tracking-widest ${selectedTimeframe === 'yesterday' ? 'text-emerald-100' : 'text-slate-400'}`}>{isRangeActive ? 'Total Cast 1' : 'Ontem'}</span>
                </div>
                <div className="space-y-3">
                  <div className="text-2xl font-black">{formatWeight(isRangeActive ? totals.rangeCast1 : totals.yesterday)}</div>
                  <p className="text-[9px] font-bold uppercase tracking-tighter opacity-80">{isRangeActive ? 'Produção Cast 1 no Período' : 'Produção do Dia Anterior'}</p>
                </div>
              </button>
              <button onClick={() => setSelectedTimeframe('week')} className={`flex flex-col p-6 rounded-2xl border transition-all duration-300 h-full ${ selectedTimeframe === 'week' ? 'bg-violet-600 border-violet-500 shadow-xl text-white' : 'bg-white border-slate-100 text-slate-800 shadow-sm' }`}>
                <div className="flex items-center justify-between mb-4">
                  <div className={`p-3 rounded-xl ${selectedTimeframe === 'week' ? 'bg-white/20' : 'bg-violet-50 text-violet-600'}`}>
                    {isRangeActive ? <Cpu size={24} /> : <CalendarDays size={24} />}
                  </div>
                  <span className={`text-[10px] font-black uppercase tracking-widest ${selectedTimeframe === 'week' ? 'text-violet-100' : 'text-slate-400'}`}>{isRangeActive ? 'Total Cast 2' : 'Esta Semana'}</span>
                </div>
                <div className="space-y-3">
                  <div className="text-2xl font-black">{formatWeight(isRangeActive ? totals.rangeCast2 : totals.week)}</div>
                  <p className="text-[9px] font-bold uppercase tracking-tighter opacity-80">{isRangeActive ? 'Produção Cast 2 no Período' : 'Acumulado da Semana'}</p>
                </div>
              </button>
            </div>
            {/* Visualização de Gráficos Omitida por Brevidade - Mesma Lógica Anterior */}
          </div>
        ) : activeTab === 'reports' ? (
          <div className="bg-white rounded-3xl shadow-sm border border-slate-100 overflow-hidden">
             {/* Conteúdo de Relatórios Omitido por Brevidade - Mesma Lógica Anterior */}
          </div>
        ) : (
          <div className="space-y-12 animate-in fade-in duration-500">
            {/* Header de Gestão */}
            <div className="flex flex-wrap items-center justify-between gap-6 pb-6 border-b border-slate-200">
               <div className="flex items-center gap-5">
                  <div className="w-16 h-16 bg-blue-600 text-white rounded-[1.5rem] flex items-center justify-center shadow-2xl shadow-blue-100 transition-transform hover:scale-105">
                    <Users2 size={32} />
                  </div>
                  <div>
                    <h2 className="text-3xl font-black text-slate-800 leading-none">Gestão de Pessoal</h2>
                    <p className="text-[11px] font-black uppercase tracking-[0.2em] text-slate-400 mt-2 flex items-center gap-2">
                       <Factory size={12} className="text-blue-500" /> Escala Operacional Manupackaging
                    </p>
                  </div>
               </div>
               <button onClick={() => { setEditingEmployee(null); setIsEmployeeModalOpen(true); }} className="px-8 py-4 bg-blue-600 text-white rounded-2xl font-black text-xs uppercase shadow-2xl shadow-blue-100 hover:bg-blue-700 transition-all flex items-center gap-3">
                  <Plus size={20} /> Novo Colaborador
               </button>
            </div>

            {/* Sumário de Pessoal */}
            <div className="grid grid-cols-2 md:grid-cols-4 gap-6">
               {[
                 { label: 'Ativos', val: employees.filter(e => e.status === 'Ativo').length, icon: UserCheck, color: 'text-blue-600', bg: 'bg-blue-50' },
                 { label: 'Férias', val: employees.filter(e => e.status === 'Férias').length, icon: Calendar, color: 'text-orange-600', bg: 'bg-orange-50' },
                 { label: 'Vagas', val: employees.filter(e => e.status === 'Vaga Aberta').length, icon: AlertCircle, color: 'text-red-600', bg: 'bg-red-50' },
                 { label: 'Total', val: employees.length, icon: Briefcase, color: 'text-slate-600', bg: 'bg-slate-50' }
               ].map((item, i) => (
                 <div key={i} className="bg-white p-6 rounded-3xl border border-slate-100 shadow-sm flex items-center gap-5">
                    <div className={`w-14 h-14 ${item.bg} ${item.color} rounded-2xl flex items-center justify-center shadow-sm`}>
                       <item.icon size={28} />
                    </div>
                    <div>
                      <p className="text-[10px] font-black text-slate-400 uppercase tracking-widest">{item.label}</p>
                      <p className="text-2xl font-black text-slate-800">{item.val}</p>
                    </div>
                 </div>
               ))}
            </div>

            {/* Seções por Setor */}
            <div className="space-y-16">
              {Object.entries(groupedPersonnel).sort((a, b) => {
                const order = ['Extrusão', 'Erema', 'Fita', 'Ghezzi', 'Lintech', 'Wutec'];
                return order.indexOf(a[0]) - order.indexOf(b[0]);
              }).map(([sector, machines]) => (
                <section key={sector} className="space-y-8">
                  <div className="flex items-center gap-4">
                     <h3 className="text-xl font-black text-slate-800 uppercase tracking-tighter px-4 py-2 bg-slate-200/50 rounded-xl">{sector}</h3>
                     <div className="h-px bg-slate-200 flex-1"></div>
                  </div>

                  <div className="space-y-10">
                    {Object.entries(machines).map(([machine, shifts]) => (
                      <div key={machine} className="bg-white/50 rounded-[2.5rem] p-8 border border-slate-100 shadow-sm">
                        <div className="flex items-center gap-3 mb-8">
                           <div className="p-2 bg-blue-100 text-blue-600 rounded-lg">
                             <Cpu size={18} />
                           </div>
                           <h4 className="text-lg font-black text-slate-700">{machine}</h4>
                        </div>

                        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-8">
                           {Object.entries(shifts).map(([shiftName, staff]: [string, any]) => (
                             <div key={shiftName} className="space-y-4">
                                <div className="flex items-center justify-between border-b border-slate-200 pb-2">
                                   <span className="text-[10px] font-black text-slate-400 uppercase tracking-widest">{shiftName}</span>
                                   <span className="text-[10px] font-bold text-slate-600 bg-slate-100 px-2 py-0.5 rounded-md">{staff.length} Membros</span>
                                </div>
                                <div className="space-y-3">
                                   {staff.map((emp: Employee) => (
                                     <EmployeeCard key={emp.id} emp={emp} />
                                   ))}
                                </div>
                             </div>
                           ))}
                        </div>
                      </div>
                    ))}
                  </div>
                </section>
              ))}
            </div>
          </div>
        )}
      </main>

      {/* Menu e Modais Omitidos por Brevidade - Mesma Lógica Anterior */}
      {isMenuOpen && (
        <div className="fixed inset-0 z-50 flex justify-end">
          <div className="absolute inset-0 bg-slate-900/40 backdrop-blur-sm" onClick={() => setIsMenuOpen(false)}></div>
          <div className="relative w-85 bg-white shadow-2xl p-0 flex flex-col h-full animate-in slide-in-from-right duration-300">
            <div className="bg-[#1e293b] p-8 text-white"><h2 className="text-xl font-black uppercase tracking-tighter flex items-center gap-3"><Settings size={26} className="text-blue-500" /> Configurações</h2></div>
            <div className="flex-1 overflow-y-auto p-6 space-y-6 custom-scrollbar">
              <div className="space-y-3"><label className="text-[10px] font-black uppercase tracking-widest text-slate-400 ml-1">Meta Mensal (KG)</label><input type="number" value={goalValue} onChange={(e) => setGoalValue(Number(e.target.value))} className="w-full bg-slate-50 border border-slate-100 p-4 rounded-2xl font-black text-blue-600 outline-none focus:ring-4 focus:ring-blue-500/10" /></div>
              <div className="grid grid-cols-1 gap-3">
                <button onClick={() => { setIsOperatorModalOpen(true); setIsMenuOpen(false); }} className="w-full flex items-center justify-between p-4 bg-slate-50 hover:bg-blue-50 text-slate-700 rounded-2xl text-[10px] font-black uppercase transition-all group"><span>Gerenciar Equipe Produção</span><ChevronRight size={16} /></button>
                <button onClick={() => { setIsShiftModalOpen(true); setIsMenuOpen(false); }} className="w-full flex items-center justify-between p-4 bg-slate-50 hover:bg-blue-50 text-slate-700 rounded-2xl text-[10px] font-black uppercase transition-all group"><span>Gerenciar Turnos</span><ChevronRight size={16} /></button>
              </div>
              <div className="pt-4 border-t border-slate-100 space-y-3"><button onClick={() => fileInputRef.current?.click()} className="w-full flex items-center gap-3 p-4 bg-emerald-50 text-emerald-700 rounded-2xl text-[10px] font-black uppercase"><Upload size={18} /> Importar Backup</button><input type="file" ref={fileInputRef} onChange={handleImport} accept=".json" className="hidden" /><button onClick={handleExportBackup} className="w-full flex items-center gap-3 p-4 bg-blue-50 text-blue-700 rounded-2xl text-[10px] font-black uppercase"><Download size={18} /> Exportar Backup</button><button onClick={handleResetSystem} className="w-full flex items-center gap-3 p-4 bg-red-50 text-red-700 rounded-2xl text-[10px] font-black uppercase"><RotateCcw size={18} /> Resetar Sistema</button></div>
            </div>
            <div className="p-6 bg-slate-50 border-t border-slate-100"><button onClick={() => { localStorage.removeItem('manupackaging_session'); setIsAuthenticated(false); }} className="w-full p-4 bg-white border border-red-100 text-red-600 rounded-2xl text-[10px] font-black uppercase flex items-center justify-center gap-2 hover:bg-red-50 transition-all"><LogOut size={16}/> Sair</button></div>
          </div>
        </div>
      )}

      {confirmDialog.isOpen && (
        <div className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm">
          <div className="bg-white rounded-3xl p-8 w-full max-sm shadow-2xl space-y-6 animate-in zoom-in-95 duration-200">
            <div className="flex flex-col items-center text-center space-y-4"><div className="w-16 h-16 bg-red-100 text-red-600 rounded-2xl flex items-center justify-center"><AlertTriangle size={32} /></div><h3 className="text-xl font-black text-slate-800 uppercase tracking-tight">{confirmDialog.title}</h3><p className="text-slate-500 text-sm font-medium">{confirmDialog.message}</p></div>
            <div className="flex gap-3"><button onClick={() => setConfirmDialog(prev => ({...prev, isOpen: false}))} className="flex-1 px-4 py-3 bg-slate-100 text-slate-600 rounded-xl font-black text-[10px] uppercase hover:bg-slate-200 transition-all">Cancelar</button><button onClick={confirmDialog.onConfirm} className="flex-1 px-4 py-3 bg-red-600 text-white rounded-xl font-black text-[10px] uppercase shadow-lg shadow-red-100 hover:bg-red-700 transition-all">Confirmar</button></div>
          </div>
        </div>
      )}

      <LaunchModal isOpen={isModalOpen} onClose={() => { setIsModalOpen(false); setEditingEntry(null); }} onSave={handleSaveEntry} operators={operators} activeMachines={activeMachines} availableShifts={shifts} initialData={editingEntry} />
      <OperatorModal isOpen={isOperatorModalOpen} onClose={() => setIsOperatorModalOpen(false)} operators={operators} onUpdate={setOperators} />
      <ShiftModal isOpen={isShiftModalOpen} onClose={() => setIsShiftModalOpen(false)} onSave={(newShift) => setShifts(prev => [...prev, { ...newShift, id: Math.random().toString(36).substr(2, 9) }])} />
      <EmployeeModal isOpen={isEmployeeModalOpen} onClose={() => { setIsEmployeeModalOpen(false); setEditingEmployee(null); }} onSave={handleSaveEmployee} availableShifts={shifts} availableMachines={activeMachines} editingEmployee={editingEmployee} />
    </div>
  );
};

export default App;
