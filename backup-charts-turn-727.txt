const renderCustomizedLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) => {
  if (percent < 0.03) return null;
  const RADIAN = Math.PI / 180;
  const radius = outerRadius * 1.2;
  const x = cx + radius * Math.cos(-midAngle * RADIAN);
  const y = cy + radius * Math.sin(-midAngle * RADIAN);
  
  return (
    <text 
      x={x} 
      y={y} 
      fill="#64748b" 
      textAnchor={x > cx ? 'start' : 'end'} 
      dominantBaseline="central" 
      className="text-[11px] font-black"
    >
      {`${(percent * 100).toFixed(1)}%`}
    </text>
  );
};

  const syncStructureWithEmployeeChange = (oldEmployee: Employee | null, newEmployeeData: Partial<Employee>, action: string) => {
    setPersonnelStructure(prevStructure => {
        const newStructure = JSON.parse(JSON.stringify(prevStructure));

        const sortRoles = (roles: string[]) => {
            const priorityMap: Record<string, number> = {};
            jobRoles.forEach((r, i) => priorityMap[r] = i);
            
            roles.sort((a, b) => {
                const pA = priorityMap[a] !== undefined ? priorityMap[a] : 999;
                const pB = priorityMap[b] !== undefined ? priorityMap[b] : 999;
                return pA - pB;
            });
        };

        const findRolesArray = (struct: PersonnelStructure, sector: string, machine: string, shift: string) => {
             if (sector === 'Liderança') {
                const sIdx = struct.leadership.findIndex((s: any) => s.shift === shift);
                if (sIdx >= 0) return struct.leadership[sIdx].roles;
             } else if (sector === 'Extrusão') {
                const sIdx = struct.extrusao.findIndex((s: any) => s.shift === shift);
                if (sIdx >= 0) {
                    const mIdx = struct.extrusao[sIdx].machines.findIndex((m: any) => m.name === machine);
                    if (mIdx >= 0) return struct.extrusao[sIdx].machines[mIdx].roles;
                }
             } else {
                const secKey = sector === 'Reciclagem' ? 'erema' : 'fitaAdesiva';
                const sectorArr = struct[secKey];
                const mIdx = sectorArr.findIndex((m: any) => m.name === machine);
                if (mIdx >= 0) {
                    const sIdx = sectorArr[mIdx].shifts.findIndex((s: any) => s.name === shift);
                    if (sIdx >= 0) return sectorArr[mIdx].shifts[sIdx].roles;
                }
             }
             return null;
        };

        if (newEmployeeData.status !== 'Desligado' && newEmployeeData.sector && newEmployeeData.machine && newEmployeeData.shift && newEmployeeData.role) {
            
            const rolesArr = findRolesArray(newStructure, newEmployeeData.sector, newEmployeeData.machine, newEmployeeData.shift);
            
            if (rolesArr) {
                const futureOccupants = employees.filter(e => 
                    e.sector === newEmployeeData.sector && 
                    e.machine === newEmployeeData.machine && 
                    e.shift === newEmployeeData.shift && 
                    e.role === newEmployeeData.role && 
                    e.status !== 'Desligado' &&
                    e.id !== newEmployeeData.id
                ).length + 1;

                const currentCapacity = rolesArr.filter((r: string) => r === newEmployeeData.role).length;

                if (futureOccupants > currentCapacity) {
                    rolesArr.push(newEmployeeData.role);
                    sortRoles(rolesArr);
                }
            }
        }

        if (oldEmployee && (action === 'Transferência' || action === 'Alteração')) {
             const hasLeft = oldEmployee.sector !== newEmployeeData.sector || 
                             oldEmployee.machine !== newEmployeeData.machine || 
                             oldEmployee.shift !== newEmployeeData.shift || 
                             oldEmployee.role !== newEmployeeData.role;

             if (hasLeft) {
                 const rolesArr = findRolesArray(newStructure, oldEmployee.sector, oldEmployee.machine, oldEmployee.shift);
                 if (rolesArr) {
                     const remainingOccupants = employees.filter(e => 
                        e.sector === oldEmployee.sector && 
                        e.machine === oldEmployee.machine && 
                        e.shift === oldEmployee.shift && 
                        e.role === oldEmployee.role && 
                        e.status !== 'Desligado' &&
                        e.id !== oldEmployee.id
                     ).length;

                     const currentCapacity = rolesArr.filter((r: string) => r === oldEmployee.role).length;

                     if (currentCapacity > remainingOccupants) {
                         const idx = rolesArr.indexOf(oldEmployee.role);
                         if (idx > -1) {
                             rolesArr.splice(idx, 1);
                             sortRoles(rolesArr);
                         }
                     }
                 }
             }
        }
        
        return newStructure;
    });
  };

             const mIdx = sectorArr.findIndex((m: any) => m.name === machine);
             if (mIdx >= 0) {
                 const sIdx = sectorArr[mIdx].shifts.findIndex((s: any) => s.name === shift);
                 if (sIdx >= 0) {
                     sectorArr[mIdx].shifts[sIdx].roles.push('Vaga');
                 }
             }
        }
        return newStruct;
    });
  };

  const handleRemoveSlot = (sector: 'leadership' | 'extrusao' | 'erema' | 'fitaAdesiva', machine: string, shift: string, roleIndex: number, employee?: Employee) => {
    setConfirmDialog({
      isOpen: true,
      title: 'Remover Vaga',
      message: employee 
        ? `Tem certeza que deseja remover "${employee.name}" e excluir esta posição do quadro? O colaborador será desligado.` 
        : 'Deseja excluir esta vaga permanentemente?',
      onConfirm: () => {
         if (employee) {
             handleEmployeeAction({ id: employee.id, status: 'Desligado' }, 'Desligamento', 'Exclusão manual da vaga');
         }

         setPersonnelStructure(prev => {
            const newStruct = JSON.parse(JSON.stringify(prev));
             if (sector === 'leadership') {
                const sIdx = newStruct.leadership.findIndex((s: any) => s.shift === shift);
                if (sIdx >= 0) newStruct.leadership[sIdx].roles.splice(roleIndex, 1);
             } else if (sector === 'extrusao') {
                const sIdx = newStruct.extrusao.findIndex((s: any) => s.shift === shift);
                if (sIdx >= 0) {
                    const mIdx = newStruct.extrusao[sIdx].machines.findIndex((m: any) => m.name === machine);
                    if (mIdx >= 0) newStruct.extrusao[sIdx].machines[mIdx].roles.splice(roleIndex, 1);
                }
            } else {
                 const secKey = sector === 'erema' ? 'erema' : 'fitaAdesiva';
                 const sectorArr = newStruct[secKey];
                 const mIdx = sectorArr.findIndex((m: any) => m.name === machine);
                 if (mIdx >= 0) {
                     const sIdx = sectorArr[mIdx].shifts.findIndex((s: any) => s.name === shift);
                     if (sIdx >= 0) sectorArr[mIdx].shifts[sIdx].roles.splice(roleIndex, 1);
                 }
            }
            setConfirmDialog(prev => ({ ...prev, isOpen: false }));
            return newStruct;
         });
      }
    });
  };

  const handleEmployeeAction = (data: Partial<Employee>, action: string, details?: string) => {
    const timestamp = new Date().toISOString();
    const oldEmployee = data.id ? employees.find(e => e.id === data.id) || null : null;

    const newLog: PersonnelLog = {
        id: Math.random().toString(36).substr(2, 9),
        date: timestamp,
        employeeName: data.name || 'Desconhecido',
        action: action as any,
        details: details || '',
        user: 'Admin'
    };
    setPersonnelLogs(prev => [newLog, ...prev]);

    if (action === 'Exclusão' && oldEmployee) {
        setEmployees(prev => prev.filter(e => e.id !== oldEmployee.id));

        setPersonnelStructure(prevStructure => {
            const newStruct = JSON.parse(JSON.stringify(prevStructure));
            const { sector, machine, shift, role } = oldEmployee;

            const removeRoleInstance = (roles: string[]) => {
                const idx = roles.indexOf(role);
                if (idx > -1) roles.splice(idx, 1);
            };

            if (sector === 'Liderança') {
                const sIdx = newStruct.leadership.findIndex((s: any) => s.shift === shift);
                if (sIdx >= 0) removeRoleInstance(newStruct.leadership[sIdx].roles);
            } else if (sector === 'Extrusão') {
                const sIdx = newStruct.extrusao.findIndex((s: any) => s.shift === shift);
                if (sIdx >= 0) {
                    const mIdx = newStruct.extrusao[sIdx].machines.findIndex((m: any) => m.name === machine);
                    if (mIdx >= 0) removeRoleInstance(newStruct.extrusao[sIdx].machines[mIdx].roles);
                }
            } else {
                const secKey = sector === 'Reciclagem' ? 'erema' : 'fitaAdesiva';
                const sectorArr = newStruct[secKey];
                const mIdx = sectorArr.findIndex((m: any) => m.name === machine);
                if (mIdx >= 0) {
                    const sIdx = sectorArr[mIdx].shifts.findIndex((s: any) => s.name === shift);
                    if (sIdx >= 0) removeRoleInstance(sectorArr[mIdx].shifts[sIdx].roles);
                }
            }
            return newStruct;
        });
        return;
    }

    if (['Contratação', 'Alteração', 'Transferência', 'Desligamento'].includes(action)) {
        syncStructureWithEmployeeChange(oldEmployee, data, action);
    }
    
    if (action === 'Contratação' && newEmployeeSlot && newEmployeeSlot.role === 'Vaga') {
         setPersonnelStructure(prev => {
            const newStruct = JSON.parse(JSON.stringify(prev));
            const { sector, machine, shift } = newEmployeeSlot;
            
            const removeVaga = (roles: string[]) => {
                const idx = roles.indexOf('Vaga');
                if (idx > -1) roles.splice(idx, 1);
            };

            if (sector === 'Liderança') {
                 const sIdx = newStruct.leadership.findIndex((s: any) => s.shift === shift);
                 if (sIdx >= 0) removeVaga(newStruct.leadership[sIdx].roles);
            } else if (sector === 'Extrusão') {
                 const sIdx = newStruct.extrusao.findIndex((s: any) => s.shift === shift);
                 if (sIdx >= 0) {
                     const mIdx = newStruct.extrusao[sIdx].machines.findIndex((m: any) => m.name === machine);
                     if (mIdx >= 0) removeVaga(newStruct.extrusao[sIdx].machines[mIdx].roles);
                 }
            } else {
                 const secKey = sector === 'Reciclagem' ? 'erema' : 'fitaAdesiva';
                 const sectorArr = newStruct[secKey];
                 const mIdx = sectorArr.findIndex((m: any) => m.name === machine);
                 if (mIdx >= 0) {
                     const sIdx = sectorArr[mIdx].shifts.findIndex((s: any) => s.name === shift);
                     if (sIdx >= 0) removeVaga(sectorArr[mIdx].shifts[sIdx].roles);
                 }
            }
            return newStruct;
         });
    }

    if (data.id) {
        setEmployees(prev => prev.map(e => e.id === data.id ? { ...e, ...data, updatedAt: timestamp } : e));
    } else {
        setEmployees(prev => [...prev, { 
            ...data, 
            id: Math.random().toString(36).substr(2, 9), 
            updatedAt: timestamp 
        } as Employee]);
    }
  };

  const formatWeight = (val: number) => {
    if (val >= 1000) {
      return (val / 1000).toLocaleString('pt-BR', { minimumFractionDigits: 3, maximumFractionDigits: 3 }) + ' T';
    }
    return val.toLocaleString('pt-BR') + ' KG';
  };
  
  const formatMinutes = (val: number) => val >= 60 ? `${Math.floor(val / 60)}h ${val % 60}m` : `${val} min`;

  const getShiftDuration = (shiftName: string) => {
    const shift = shifts.find(s => s.name === shiftName);
    if (!shift) return 720;
    const [h1, m1] = shift.startTime.split(':').map(Number);
    const [h2, m2] = shift.endTime.split(':').map(Number);
    let minutes = (h2 * 60 + m2) - (h1 * 60 + m1);
    if (minutes < 0) minutes += 1440;
    return minutes === 0 ? 1440 : minutes;
  };

  const totals = useMemo(() => {
    const now = new Date();
    const todayStr = now.toLocaleDateString('sv-SE');
    const yesterday = new Date(now);
    yesterday.setDate(now.getDate() - 1);
    const yesterdayStr = yesterday.toLocaleDateString('sv-SE');
    const startOfWeek = new Date(now);
    startOfWeek.setHours(0,0,0,0);
    startOfWeek.setDate(now.getDate() - now.getDay());

    const currentYear = parseInt(dashboardMonth.split('-')[0]);
    const currentMonth = parseInt(dashboardMonth.split('-')[1]) - 1;
    const isCurrentMonth = now.getFullYear() === currentYear && now.getMonth() === currentMonth;
    const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();
    const currentDay = isCurrentMonth ? now.getDate() : daysInMonth;

    const baseFiltered = productionData.filter(e => {
      const matchesOp = dashOperator ? e.operator === dashOperator : true;
      const matchesMac = dashMachine ? e.machine === dashMachine : true;
      return matchesOp && matchesMac;
    });

    const results = { 
      month: 0, 
      eremaMonth: 0,
      today: 0, 
      yesterday: 0, 
      week: 0, 
      range: 0, 
      projection: 0,
      remaining: 0,
      dailyRequired: 0
    };

    baseFiltered.forEach(e => {
      const isErema = e.machine.toLowerCase().includes('erema');

      if (e.date.startsWith(dashboardMonth)) {
        if (isErema) {
          results.eremaMonth += (e.netWeight || 0);
        } else {
          results.month += (e.netWeight || 0);
        }
      }
      
      if (e.date === todayStr) results.today += (e.netWeight || 0);
      if (e.date === yesterdayStr) results.yesterday += (e.netWeight || 0);
      const entryDate = new Date(e.date + 'T00:00:00');
      if (entryDate >= startOfWeek) results.week += (e.netWeight || 0);

      if (dashStartDate || dashEndDate) {
        const matchesStart = dashStartDate ? e.date >= dashStartDate : true;
        const matchesEnd = dashEndDate ? e.date <= dashEndDate : true;
        if (matchesStart && matchesEnd) {
          results.range += (e.netWeight || 0);
        }
      }
    });

    results.projection = currentDay > 0 ? (results.month / currentDay) * daysInMonth : 0;
    results.remaining = Math.max(0, goalValue - results.month);
    
    const daysRemaining = Math.max(0, daysInMonth - currentDay);
    results.dailyRequired = daysRemaining > 0 ? results.remaining / daysRemaining : results.remaining;

    return results;
  }, [productionData, dashboardMonth, dashOperator, dashMachine, goalValue, dashStartDate, dashEndDate]);

  const filteredDataForCharts = useMemo(() => {
    const now = new Date();
    const todayStr = now.toLocaleDateString('sv-SE');
    const yesterday = new Date(now);
    yesterday.setDate(now.getDate() - 1);
    const yesterdayStr = yesterday.toLocaleDateString('sv-SE');
    const startOfWeek = new Date(now);
    startOfWeek.setHours(0,0,0,0);
    startOfWeek.setDate(now.getDate() - now.getDay());

      const matchesOp = dashOperator ? e.operator === dashOperator : true;
      const matchesMac = dashMachine ? e.machine === dashMachine : true;
      
      if (dashStartDate || dashEndDate) {
        const matchesStart = dashStartDate ? e.date >= dashStartDate : true;
        const matchesEnd = dashEndDate ? e.date <= dashEndDate : true;
        return matchesOp && matchesMac && matchesStart && matchesEnd;
      }

      let matchesTime = false;
      if (selectedTimeframe === 'month') {
        matchesTime = e.date.startsWith(dashboardMonth);
      } else if (selectedTimeframe === 'yesterday') {
        matchesTime = e.date === yesterdayStr;
      } else if (selectedTimeframe === 'week') {
        const entryDate = new Date(e.date + 'T00:00:00');
        matchesTime = entryDate >= startOfWeek;
      } else if (selectedTimeframe === 'today') {
        matchesTime = e.date === todayStr;
      } else {
        matchesTime = e.date.startsWith(dashboardMonth);

        ? Object.values(e).some(val => String(val).toLowerCase().includes(searchTerm.toLowerCase()))
        : true;

  }, [productionData, searchTerm, filterStartDate, filterEndDate]);

  const reportTotals = useMemo(() => {
    return filteredReports.reduce((acc, item) => ({
      gross: acc.gross + (item.grossWeight || 0),
      tara: acc.tara + (item.tara || 0),
      net: acc.net + (item.netWeight || 0),
      volumes: acc.volumes + (item.volumes || 0),
      tubetes: acc.tubetes + (item.tubetes || 0),
      ecoA: acc.ecoA + (item.ecoA || 0),
      ecoBP: acc.ecoBP + (item.ecoBP || 0),
      ecoBM: acc.ecoBM + (item.ecoBM || 0),

      manut: acc.manut + (item.manutencaoMin || 0),
      proc: acc.proc + (item.processoMin || 0),
      outros: acc.outros + (item.outrosMin || 0),
    }), { gross: 0, tara: 0, net: 0, volumes: 0, tubetes: 0, ecoA: 0, ecoBP: 0, ecoBM: 0, borra: 0, manut: 0, proc: 0, outros: 0 });
  }, [filteredReports]);

  const stats = useMemo(() => {
    const initialStats: Record<string, OperatorStat> = {};
    operators.forEach(op => {
      initialStats[op] = { name: op, prod: 0, ecoA: 0, ecoBP: 0, ecoBM: 0, borra: 0, p_manut: 0, p_proc: 0, p_outros: 0 };

      if (!initialStats[e.operator]) {

      }
      const isErema = e.machine.toLowerCase().includes('erema');
      
      if (!isErema) {
        initialStats[e.operator].prod += (e.netWeight || 0);
      }
      initialStats[e.operator].ecoA += (e.ecoA || 0);
      initialStats[e.operator].ecoBP += (e.ecoBP || 0);
      initialStats[e.operator].ecoBM += (e.ecoBM || 0);
      initialStats[e.operator].borra += (e.borraTotal || 0);
      initialStats[e.operator].p_manut += (e.manutencaoMin || 0);
      initialStats[e.operator].p_proc += (e.processoMin || 0);
      initialStats[e.operator].p_outros += (e.outrosMin || 0);
    });

  }, [filteredDataForCharts, operators]);

  const personnelStats = useMemo(() => {
    let totalStaff = 0;
    let totalOperators = 0;
    let totalAuxiliaries = 0;
    let totalVacancies = 0;

    employees.forEach(emp => {
        if (emp.status === 'Ativo') {
            totalStaff++;
            if (emp.role.includes('Operador') || emp.role === 'Líder de Turno') totalOperators++;
            else totalAuxiliaries++;
        }
    });

    let expectedSlots = 0;
    personnelStructure.leadership.forEach(s => expectedSlots += s.roles.length);
    personnelStructure.extrusao.forEach(s => s.machines.forEach(m => expectedSlots += m.roles.length));
    personnelStructure.erema.forEach(m => m.shifts.forEach(s => expectedSlots += s.roles.length));
    personnelStructure.fitaAdesiva.forEach(m => m.shifts.forEach(s => expectedSlots += s.roles.length));
    
    const activeEmployeesCount = employees.filter(e => e.status !== 'Desligado').length;
    totalVacancies = Math.max(0, expectedSlots - activeEmployeesCount);

    return { totalStaff, totalOperators, totalAuxiliaries, totalVacancies };
  }, [employees, personnelStructure]);

  const eremaChartData = useMemo(() => {
    const data: Record<string, number> = {};
    filteredDataForCharts
      .filter(e => e.machine.toLowerCase().includes('erema'))

         data[dateKey] = (data[dateKey] || 0) + (e.netWeight || 0);

        // but since it's just day/month, reliable enough for same year. 
        // For perfect sort, we'd stick to ISO keys then format.
        return 0;

  if (!isAuthenticated) {

        <div className="bg-white rounded-3xl p-8 w-full max-w-md shadow-2xl space-y-8">
          <div className="text-center space-y-2">
            <div className="w-16 h-16 bg-blue-600 rounded-2xl flex items-center justify-center text-white mx-auto shadow-lg mb-4">

            <p className="text-slate-400 text-[10px] font-bold uppercase tracking-widest">Painel de Acesso</p>
          </div>
          <button onClick={() => { localStorage.setItem('manupackaging_session', 'true'); setIsAuthenticated(true); }} className="w-full bg-blue-600 hover:bg-blue-700 text-white font-black py-4 rounded-2xl transition-all shadow-xl shadow-blue-200 uppercase text-xs">Entrar no Sistema</button>

                          <div className="bg-emerald-50 p-1.5 rounded-lg text-emerald-500"><HardHat size={16} /></div>
                      </div>
                      <span className="text-3xl font-black text-slate-800">{personnelStats.totalOperators}</span>
                      <p className="text-[10px] font-bold text-slate-400 mt-1">Líderes</p>
                  </div>
                  {/* Card Auxiliares */}
                  <div className="bg-white p-4 rounded-xl border border-slate-100 shadow-sm min-w-[200px] flex-1">
                      <div className="flex justify-between items-start mb-2">
                          <h3 className="text-[10px] font-black text-orange-500 uppercase tracking-widest">Auxiliares</h3>
                          <div className="bg-orange-50 p-1.5 rounded-lg text-orange-500"><UserPlus size={16} /></div>
                      </div>
                      <span className="text-3xl font-black text-slate-800">{personnelStats.totalAuxiliaries}</span>
                      <p className="text-[10px] font-bold text-slate-400 mt-1">Suporte</p>
                  </div>
                  {/* Card Vagas */}
                  <div className="bg-white p-4 rounded-xl border border-slate-100 shadow-sm min-w-[200px] flex-1">
                      <div className="flex justify-between items-start mb-2">
                          <h3 className="text-[10px] font-black text-red-500 uppercase tracking-widest">Vagas</h3>
                          <div className="bg-red-50 p-1.5 rounded-lg text-red-500"><UserMinus size={16} /></div>
                      </div>
                      <span className="text-3xl font-black text-slate-800">{personnelStats.totalVacancies}</span>
                      <p className="text-[10px] font-bold text-slate-400 mt-1">Aberto</p>
                  </div>
              </div>

              {/* Toolbar */}
              <div className="flex justify-end items-center gap-2">
                  <button onClick={() => setIsHistoryModalOpen(true)} className="px-4 py-2 bg-white border border-slate-200 text-slate-600 rounded-xl text-[10px] font-black uppercase hover:bg-slate-50 transition-all flex items-center gap-2 shadow-sm">
                      <History size={14}/> Histórico de Movimentação
                  </button>
              </div>

              {/* Dynamic Structure Visualization - Matching Screenshot Visuals */}
              <div className="space-y-8">
                  
                  {/* Liderança Section */}
                  <div className="rounded-t-xl overflow-hidden shadow-sm border border-slate-200">
                      <div className="bg-[#4f46e5] p-4 flex items-center gap-3">
                          <div className="p-2 bg-white/10 text-white rounded-lg"><Crown size={18}/></div>
                          <div>
                              <h3 className="text-sm font-black uppercase tracking-widest text-white">Liderança & Coordenação</h3>
                              <p className="text-[10px] font-bold text-indigo-200/70 uppercase tracking-wider">Supervisão de Turnos</p>
                          </div>
                      </div>
                      <div className="bg-white p-4 grid grid-cols-1 md:grid-cols-2 gap-4">
                          {personnelStructure.leadership.map((shift, sIdx) => {
                              const isNight = shift.shift.toLowerCase().includes('noturno');
                              return (
                                <div key={sIdx} className={`p-4 rounded-xl border ${isNight ? 'bg-[#2a3441] border-[#374151]' : 'bg-slate-50 border-slate-100 shadow-sm'}`}>
                                    <div className="flex justify-between items-center mb-3">
                                        <h5 className={`text-[10px] font-black uppercase ${isNight ? 'text-slate-400' : 'text-slate-500'}`}>{shift.shift}</h5>
                                        <button onClick={() => handleAddSlot('leadership', 'Supervisão', shift.shift)} className={`p-1 rounded hover:bg-black/10 transition-colors ${isNight ? 'text-indigo-400' : 'text-indigo-500'}`} title="Adicionar Liderança">
                                            <Plus size={14}/>
                                        </button>
                                    </div>
                                    <div className="space-y-1.5">
                                        {shift.roles.map((role, rIdx) => (
                                            <React.Fragment key={rIdx}>
                                                {renderPersonnelSlot('leadership', 'Supervisão', shift.shift, role, isNight, rIdx, shift.roles)}
                                            </React.Fragment>
                                        ))}
                                    </div>
                                </div>
                              );
                          })}
                      </div>
                  </div>

                  {/* Extrusão Section */}
                  <div className="rounded-t-xl overflow-hidden shadow-sm border border-slate-200">
                      <div className="bg-[#1e293b] p-4 flex items-center gap-3">
                          <div className="p-2 bg-white/10 text-white rounded-lg"><Briefcase size={18}/></div>
                          <div>
                              <h3 className="text-sm font-black uppercase tracking-widest text-white">Setor: Extrusão</h3>
                              <p className="text-[10px] font-bold text-slate-400 uppercase tracking-wider">Escala de Turnos</p>
                          </div>
                      </div>
                      
                      <div className="bg-white p-4 grid grid-cols-1 md:grid-cols-2 gap-4">
                          {personnelStructure.extrusao.map((shift, sIdx) => {
                              const isNight = shift.shift.toLowerCase().includes('noturno');
                              // Night shift gets dark card bg, Day shift gets white
                              return (
                                <div key={sIdx} className={`rounded-xl border p-4 ${isNight ? 'bg-[#1e293b] border-[#334155]' : 'bg-white border-slate-100 shadow-sm'}`}>
                                    <div className="flex items-center gap-2 mb-4">
                                        <Clock size={16} className={isNight ? 'text-slate-400' : 'text-slate-400'} />
                                        <h4 className={`text-xs font-black uppercase tracking-widest ${isNight ? 'text-white' : 'text-slate-800'}`}>{shift.shift}</h4>
                                    </div>
                                    <div className="grid grid-cols-1 gap-4">
                                        {shift.machines.map((machine, mIdx) => (
                                            <div key={mIdx} className={`p-3 rounded-xl ${isNight ? 'bg-[#242f3d] border-[#374151]' : 'bg-slate-50 border border-slate-100'}`}>
                                                <div className="flex justify-between items-center mb-3">
                                                    <h5 className={`text-[10px] font-black uppercase ${isNight ? 'text-slate-400' : 'text-slate-500'}`}>{machine.name}</h5>
                                                    <button onClick={() => handleAddSlot('extrusao', machine.name, shift.shift)} className={`p-1 rounded hover:bg-black/10 transition-colors ${isNight ? 'text-blue-400' : 'text-blue-500'}`} title="Adicionar Vaga">
                                                        <Plus size={14}/>
                                                    </button>
                                                </div>
                                                <div className="space-y-1.5">
                                                    {machine.roles.map((role, rIdx) => (
                                                        <React.Fragment key={rIdx}>
                                                            {renderPersonnelSlot('extrusao', machine.name, shift.shift, role, isNight, rIdx, machine.roles)}
                                                        </React.Fragment>
                                                    ))}
                                                </div>
                                            </div>
                                        ))}
                                    </div>
                                </div>
                              );
                          })}
                      </div>
                  </div>

                  {/* Reciclagem */}
                  <div className="rounded-t-xl overflow-hidden shadow-sm border border-slate-200">
                       <div className="bg-[#064e3b] p-4 flex items-center gap-3">
                          <div className="p-2 bg-white/10 text-white rounded-lg"><RotateCcw size={18}/></div>
                          <div>
                              <h3 className="text-sm font-black uppercase tracking-widest text-white">Setor: Reciclagem</h3>
                              <p className="text-[10px] font-bold text-emerald-200/60 uppercase tracking-wider">Erema</p>
                          </div>
                       </div>
                       <div className="bg-white p-4">
                           {personnelStructure.erema.map((machine, mIdx) => (
                               <div key={mIdx} className="grid grid-cols-1 md:grid-cols-2 gap-4">
                                   {machine.shifts.map((shift, sIdx) => {
                                       const isNight = shift.name.toLowerCase().includes('noturno');
                                       return (
                                           <div key={sIdx} className="bg-slate-50 border border-slate-100 p-4 rounded-xl shadow-sm">
                                               <div className="flex justify-between items-center mb-4 pb-2 border-b border-slate-200">
                                                   <span className="text-xs font-black text-slate-700 uppercase">{shift.name}</span>
                                                   <button onClick={() => handleAddSlot('erema', machine.name, shift.name)} className="p-1 rounded bg-slate-200 hover:bg-slate-300 text-slate-500 transition-colors">
                                                        <Plus size={12}/>
                                                   </button>
                                               </div>
                                               <div className="space-y-2">
                                                   {shift.roles.map((role, rIdx) => (
                                                       <React.Fragment key={rIdx}>
                                                           {renderPersonnelSlot('erema', machine.name, shift.name, role, isNight, rIdx, shift.roles)}
                                                       </React.Fragment>
                                                   ))}
                                               </div>
                                           </div>
                                       );
                                   })}
                               </div>
                           ))}
                       </div>
                  </div>

                  {/* Fita Adesiva */}
                  <div className="rounded-t-xl overflow-hidden shadow-sm border border-slate-200">
                       <div className="bg-[#7c2d12] p-4 flex items-center gap-3">
                          <div className="p-2 bg-white/10 text-white rounded-lg"><Layers size={18}/></div>
                          <div>
                              <h3 className="text-sm font-black uppercase tracking-widest text-white">Setor: Fita Adesiva</h3>
                              <p className="text-[10px] font-bold text-orange-200/60 uppercase tracking-wider">Ghezzi/Lintech/Wutec</p>
                          </div>
                       </div>
                       <div className="bg-white p-4 grid grid-cols-1 md:grid-cols-2 gap-4">
                           {personnelStructure.fitaAdesiva.map((machine, mIdx) => (
                               <div key={mIdx} className="space-y-3 p-4 bg-white border border-slate-100 rounded-xl shadow-sm">
                                   <div className="flex items-center gap-2 mb-2">
                                       <div className="w-2 h-2 rounded-full bg-orange-500"></div>
                                       <h4 className="text-xs font-black uppercase text-slate-700">{machine.name}</h4>
                                   </div>
                                   <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
                                       {machine.shifts.map((shift, sIdx) => {
                                           const isNight = shift.name.toLowerCase().includes('noturno');
                                           return (
                                               <div key={sIdx} className="bg-slate-50 border border-slate-100 p-3 rounded-xl relative">
                                                   <div className="flex justify-between items-center mb-2">
                                                       <span className="text-[10px] font-bold text-slate-500 uppercase">{shift.name}</span>
                                                       <button onClick={() => handleAddSlot('fitaAdesiva', machine.name, shift.name)} className="text-orange-400 hover:text-orange-600 transition-colors">
                                                            <Plus size={14}/>
                                                       </button>
                                                   </div>
                                                   <div className="space-y-2">
                                                       {shift.roles.map((role, rIdx) => (
                                                           <React.Fragment key={rIdx}>
                                                               {renderPersonnelSlot('fitaAdesiva', machine.name, shift.name, role, isNight, rIdx, shift.roles)}
                                                           </React.Fragment>
                                                       ))}
                                                   </div>
                                               </div>
                                           );
                                       })}
                                   </div>
                               </div>
                           ))}
                       </div>
                  </div>
              </div>
           </div>
        )}
      </main>

      {/* Settings Menu Modal */}
      {isMenuOpen && (
        <div className="fixed inset-0 z-50 flex justify-end" onClick={() => setIsMenuOpen(false)}>
            <div className="w-80 bg-white h-full shadow-2xl p-6 animate-in slide-in-from-right duration-300" onClick={e => e.stopPropagation()}>
                <div className="flex justify-between items-center mb-8">
                    <h2 className="text-xl font-black uppercase text-slate-800">Configurações</h2>
                    <button onClick={() => setIsMenuOpen(false)}><X size={24} className="text-slate-400 hover:text-slate-800" /></button>
                </div>

                <div className="space-y-6">
                    <div className="space-y-2">
                        <label className="text-[10px] font-black uppercase text-slate-400 tracking-widest">Meta Mensal</label>
                        <input type="number" value={goalValue} onChange={e => setGoalValue(Number(e.target.value))} className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-3 font-bold text-slate-700" />
                    </div>

                    <button onClick={() => setIsShiftModalOpen(true)} className="w-full py-4 bg-slate-50 hover:bg-slate-100 rounded-xl flex items-center gap-3 px-4 transition-all">
                        <div className="p-2 bg-white rounded-lg shadow-sm text-blue-600"><Clock size={18}/></div>
                        <div className="text-left">
                            <p className="text-xs font-bold text-slate-700 uppercase">Configurar Turnos</p>
                            <p className="text-[9px] font-bold text-slate-400 uppercase">Horários de trabalho</p>
                        </div>
                    </button>

                    <button onClick={() => setIsOperatorModalOpen(true)} className="w-full py-4 bg-slate-50 hover:bg-slate-100 rounded-xl flex items-center gap-3 px-4 transition-all">
                        <div className="p-2 bg-white rounded-lg shadow-sm text-blue-600"><Users size={18}/></div>
                        <div className="text-left">
                            <p className="text-xs font-bold text-slate-700 uppercase">Gerenciar Operadores</p>
                            <p className="text-[9px] font-bold text-slate-400 uppercase">Equipe de produção</p>
                        </div>
                    </button>

                    <div className="pt-6 border-t border-slate-100 space-y-3">
                        <p className="text-[10px] font-black uppercase text-slate-400 tracking-widest">Dados & Sistema</p>
                        
                        <button onClick={handleExportBackup} className="w-full py-3 bg-white border border-slate-200 hover:bg-slate-50 rounded-xl flex items-center justify-between px-4 transition-all text-slate-600">
                             <span className="text-[10px] font-black uppercase flex items-center gap-2"><Download size={14}/> Backup</span>
                        </button>
                        
                        <div className="relative">
                            <input 
                                type="file" 
                                ref={fileInputRef}
                                onChange={handleImport}
                                accept=".json"
                                className="hidden"
                            />
                            <button onClick={() => fileInputRef.current?.click()} className="w-full py-3 bg-white border border-slate-200 hover:bg-slate-50 rounded-xl flex items-center justify-between px-4 transition-all text-slate-600">
                                <span className="text-[10px] font-black uppercase flex items-center gap-2"><Upload size={14}/> Restaurar</span>
                            </button>
                        </div>

                        <button onClick={handleResetSystem} className="w-full py-3 bg-red-50 border border-red-100 hover:bg-red-100 rounded-xl flex items-center justify-between px-4 transition-all text-red-600">
                             <span className="text-[10px] font-black uppercase flex items-center gap-2"><RotateCcw size={14}/> Resetar Sistema</span>
                        </button>
                    </div>

                    <div className="pt-6 border-t border-slate-100">
                        <button onClick={() => { setIsAuthenticated(false); localStorage.removeItem('manupackaging_session'); }} className="w-full py-4 bg-slate-800 text-white rounded-xl font-black text-xs uppercase shadow-lg shadow-slate-200 hover:bg-slate-900 transition-all flex items-center justify-center gap-2">
                            <LogOut size={16} /> Sair do Sistema
                        </button>
                    </div>
                </div>
            </div>
        </div>
      )}

      {/* Erema Chart Modal */}
      {isEremaChartModalOpen && (
        <div className="fixed inset-0 z-[60] flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-sm animate-in fade-in duration-200">
          <div className="bg-white rounded-[2rem] w-full max-w-4xl shadow-2xl relative overflow-hidden animate-in zoom-in-95 duration-200 flex flex-col max-h-[85vh]">
             <div className="bg-emerald-600 text-white p-6 flex items-center justify-between">
                <div className="flex items-center gap-4">
                   <div className="w-12 h-12 bg-white/20 rounded-2xl flex items-center justify-center text-white backdrop-blur-sm">
                      <RotateCcw size={24} />
                   </div>
                   <div>
                      <h2 className="text-xl font-black uppercase tracking-tight">Produção Erema</h2>
                      <p className="text-[10px] text-white/80 font-bold uppercase tracking-widest">Análise de Reciclagem</p>
                   </div>
                </div>
                <button onClick={() => setIsEremaChartModalOpen(false)} className="hover:bg-white/10 p-2 rounded-xl transition-colors text-white/80 hover:text-white">
                   <X size={24} />
                </button>
             </div>
             
             <div className="p-6 flex-1 flex flex-col">
                <div className="bg-slate-50 rounded-2xl p-4 border border-slate-100 flex-1 min-h-[400px]">
                   <ResponsiveContainer width="100%" height="100%">
                      <BarChart data={eremaChartData} margin={{ top: 20, right: 30, left: 20, bottom: 5 }}>
                        <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="#e2e8f0" />
                        <XAxis dataKey="name" axisLine={false} tickLine={false} tick={{fontSize: 12, fill: '#64748b'}} dy={10} />
                        <YAxis axisLine={false} tickLine={false} tick={{fontSize: 12, fill: '#64748b'}} tickFormatter={(val) => `${val}kg`} />
                        <Tooltip 
                            cursor={{fill: '#f1f5f9'}}
                            contentStyle={{ borderRadius: '12px', border: 'none', boxShadow: '0 4px 6px -1px rgb(0 0 0 / 0.1)' }}
                        />
                        <Bar dataKey="value" name="Reciclado (Kg)" fill="#10b981" radius={[4, 4, 0, 0]} maxBarSize={60} />
                      </BarChart>
                   </ResponsiveContainer>
                </div>
                <div className="mt-4 flex justify-end">
                   <div className="px-4 py-2 bg-emerald-50 text-emerald-700 rounded-xl text-sm font-black border border-emerald-100">
                      Total no Período: {formatWeight(eremaChartData.reduce((acc, curr) => acc + curr.value, 0))}
                   </div>
                </div>
             </div>
          </div>
        </div>
      )}

      {/* Modals */}
      <LaunchModal 
        isOpen={isModalOpen} 
        onClose={() => { setIsModalOpen(false); setEditingEntry(null); }}
        onSave={handleSaveEntry}
        operators={operators}
        activeMachines={activeMachines}
        availableShifts={shifts}
        initialData={editingEntry}
      />

      <ShiftModal 
        isOpen={isShiftModalOpen}
        onClose={() => setIsShiftModalOpen(false)}
        onSave={(newShift) => setShifts(prev => [...prev, { ...newShift, id: Math.random().toString(36).substr(2, 9) }])}
      />

      <OperatorModal
        isOpen={isOperatorModalOpen}
        onClose={() => setIsOperatorModalOpen(false)}
        operators={operators}
        onUpdate={setOperators}
      />
      
      <EmployeeModal 
          isOpen={isEmployeeModalOpen}
          onClose={() => { setIsEmployeeModalOpen(false); setSelectedEmployee(null); setNewEmployeeSlot(undefined); }}
          onSave={handleEmployeeAction}
          availableShifts={[{id:'l1', name: 'Diurno', type: 'Diurno', startTime: '', endTime: ''}, {id:'l2', name: 'Noturno', type: 'Noturno', startTime: '', endTime: ''}, ...shifts]}
          availableMachines={activeMachines}
          availableRoles={jobRoles}
          initialData={selectedEmployee}
          slotInfo={newEmployeeSlot}
      />

      <HistoryModal
          isOpen={isHistoryModalOpen}
          onClose={() => setIsHistoryModalOpen(false)}
          logs={personnelLogs}
      />

      <RoleModal 
        isOpen={isRoleModalOpen}
        onClose={() => setIsRoleModalOpen(false)}
        roles={jobRoles}
        onUpdate={setJobRoles}
      />

      {/* Confirm Dialog */}
      {confirmDialog.isOpen && (
        <div className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm animate-in fade-in duration-200">
          <div className="bg-white rounded-3xl w-full max-w-sm shadow-2xl p-6 text-center space-y-4 animate-in zoom-in-95 duration-200">
             <div className="w-16 h-16 bg-red-100 text-red-600 rounded-full flex items-center justify-center mx-auto">
                <AlertTriangle size={32} />
             </div>
             <div>
                <h3 className="text-lg font-black text-slate-800 uppercase">{confirmDialog.title}</h3>
                <p className="text-sm text-slate-500 font-medium mt-2">{confirmDialog.message}</p>
             </div>
             <div className="grid grid-cols-2 gap-3 pt-2">
                <button onClick={() => setConfirmDialog(prev => ({ ...prev, isOpen: false }))} className="py-3 bg-white border border-slate-200 rounded-xl text-slate-500 text-xs font-black uppercase hover:bg-slate-50">Cancelar</button>
                <button onClick={confirmDialog.onConfirm} className="py-3 bg-red-600 text-white rounded-xl text-xs font-black uppercase hover:bg-red-700 shadow-lg shadow-red-200">Confirmar</button>
             </div>
          </div>
        </div>
      )}
    </div>
  );
};
