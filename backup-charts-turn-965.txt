const renderCustomizedLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) => {
  const radius = outerRadius * 1.15;
  const x = cx + radius * Math.cos(-midAngle * Math.PI / 180);
  const y = cy + radius * Math.sin(-midAngle * Math.PI / 180);
  return (
    <text x={x} y={y} fill="#64748b" textAnchor={x > cx ? 'start' : 'end'} dominantBaseline="central" className="text-[10px] font-black">
      {`${(percent * 100).toFixed(1)}%`}
    </text>
  );
};

  name: string;
  prod: number;
  ecoA: number;
  ecoBP: number;
  ecoBM: number;
  borra: number;
  p_manut: number;
  p_proc: number;
  p_outros: number;
}

const App: React.FC = () => {
  const [isAuthenticated, setIsAuthenticated] = useState(() => !!localStorage.getItem('manupackaging_session'));
  const [authData] = useState({ user: 'adaiasmelo', pass: '123456' }); 
  const [activeTab, setActiveTab] = useState<'dashboard' | 'reports' | 'personnel'>('dashboard');

    const saved = localStorage.getItem('manupackaging_data');
    return saved ? JSON.parse(saved) : INITIAL_DATA;
  });

    const saved = localStorage.getItem('manupackaging_goal');
    return saved ? parseFloat(saved) : DEFAULT_GOAL;
  });

  const [operators, setOperators] = useState<string[]>(() => {
    const saved = localStorage.getItem('manupackaging_operators');
    return saved ? JSON.parse(saved) : DEFAULT_OPERATORS;
  });

  const [machines, setMachines] = useState<string[]>(() => {
    const saved = localStorage.getItem('manupackaging_machines');
    return saved ? JSON.parse(saved) : ['Cast 1', 'Cast 2', 'Erema 1', 'Ghezzi', 'Lintech', 'Wutec'];
  });

    const saved = localStorage.getItem('manupackaging_employees');
    return saved ? JSON.parse(saved) : [];
  });

    const saved = localStorage.getItem('manupackaging_vacancies');
    return saved ? JSON.parse(saved) : [];
  });

    const saved = localStorage.getItem('manupackaging_shifts');
    return saved ? JSON.parse(saved) : [
      { id: '1', name: 'Diurno 1', startTime: '07:00', endTime: '19:00', type: 'Diurno' },
      { id: '2', name: 'Noturno 1', startTime: '19:00', endTime: '07:00', type: 'Noturno' },
      { id: '3', name: 'Diurno 2', startTime: '07:00', endTime: '19:00', type: 'Diurno' },
      { id: '4', name: 'Noturno 2', startTime: '19:00', endTime: '07:00', type: 'Noturno' },
      { id: '5', name: 'Comercial', startTime: '08:00', endTime: '18:00', type: 'Diurno' }
    ];
  });

  const [isShiftModalOpen, setIsShiftModalOpen] = useState(false);
  const [isEmployeeModalOpen, setIsEmployeeModalOpen] = useState(false);
  const [editingEntry, setEditingEntry] = useState<ProductionEntry | null>(null);
  const [editingEmployee, setEditingEmployee] = useState<any>(null);
  const [isMenuOpen, setIsMenuOpen] = useState(false);
  const [searchTerm, setSearchTerm] = useState('');

  useEffect(() => { localStorage.setItem('manupackaging_employees', JSON.stringify(employees)); }, [employees]);
  useEffect(() => { localStorage.setItem('manupackaging_vacancies', JSON.stringify(vacancies)); }, [vacancies]);
  useEffect(() => { localStorage.setItem('manupackaging_shifts', JSON.stringify(shifts)); }, [shifts]);
  useEffect(() => { localStorage.setItem('manupackaging_goal', goalValue.toString()); }, [goalValue]);
  useEffect(() => { localStorage.setItem('manupackaging_operators', JSON.stringify(operators)); }, [operators]);
  useEffect(() => { localStorage.setItem('manupackaging_machines', JSON.stringify(machines)); }, [machines]);

    if (newEntry.id) {

    } else {
      setProductionData(prev => [{ ...newEntry, id: Math.random().toString(36).substr(2, 9) } as ProductionEntry, ...prev]);
    }
    setEditingEntry(null);
  };

  const handleSaveShift = (newShift: Omit<Shift, 'id'>) => {
    const shift: Shift = { ...newShift, id: Math.random().toString(36).substr(2, 9) };
    setShifts(prev => [...prev, shift]);
  };

        id,
        name: newEmp.name || `Vaga ${newEmp.role}`,

        group: newEmp.shift,
        sector: newEmp.sector,
        machine: newEmp.machine,
        openSince: new Date().toLocaleDateString('sv-SE')
      };
      setVacancies(prev => [...prev.filter(v => v.id !== id), vacancy]);
      setEmployees(prev => prev.filter(e => e.id !== id));
    } else {
      const employee = {
        id,
        name: newEmp.name,
        role: newEmp.role,
        sector: newEmp.sector,
        machine: newEmp.machine,
        group: newEmp.shift,

        startDate: newEmp.startDate || new Date().toLocaleDateString('sv-SE')

      { name: 'Manutenção', value: totalManut },
      { name: 'Processo', value: totalProc },
      { name: 'Outros', value: totalOutros }

            <input type="password" value={authData.pass} readOnly className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-3 text-sm font-bold" />

                        {prodChartData.map((_, i) => <Cell key={i} fill={COLORS[i % COLORS.length]} stroke="none" />)}
                      </Pie>
                      <Tooltip formatter={(val: number) => formatWeight(val)} />

                  <h3 className="font-black text-xs uppercase tracking-widest text-slate-800 mb-4">♻️ Eco A (KG)</h3>

                    <PieChart>

                  <ResponsiveContainer width="100%" height="100%">
                    <PieChart>
                      <Pie data={ecoBPChartData} cx="50%" cy="50%" outerRadius={100} innerRadius={60} dataKey="ecoBP" nameKey="name" label={renderCustomizedLabel} paddingAngle={2}>
                        {ecoBPChartData.map((_, i) => <Cell key={i} fill={COLORS[i % COLORS.length]} stroke="none" />)}

                      <Legend verticalAlign="bottom" height={120} align="center" iconType="circle" />
                    </PieChart>

                    <PieChart>

                        <Cell fill="#f43f5e" stroke="none" />

                      <Legend verticalAlign="bottom" height={120} align="center" iconType="circle" />
                    </PieChart>
                  </ResponsiveContainer>

               <div className="bg-white rounded-3xl p-6 shadow-sm border border-slate-100 flex flex-col h-[520px]">
                  <h3 className="font-black text-xs uppercase tracking-widest text-slate-800 mb-4">🗑️ Borra Total por Operador (KG)</h3>
                  <ResponsiveContainer width="100%" height="100%">
                    <PieChart>
                      <Pie data={borraChartData} cx="50%" cy="50%" outerRadius={100} innerRadius={60} dataKey="borra" nameKey="name" label={renderCustomizedLabel} paddingAngle={2}>

                      <Legend verticalAlign="bottom" height={120} align="center" iconType="circle" />
                    </PieChart>

               <div className="bg-white rounded-3xl p-6 shadow-sm border border-slate-100 flex flex-col h-[520px]">
                  <h3 className="font-black text-xs uppercase tracking-widest text-slate-800 mb-4">⏱️ Tempo de Parada por Categoria</h3>
                  <ResponsiveContainer width="100%" height="100%">
                    <PieChart>
                      <Pie data={stopsOverviewData} cx="50%" cy="50%" outerRadius={100} innerRadius={60} dataKey="value" nameKey="name" label={renderCustomizedLabel} paddingAngle={2}>
                        <Cell fill="#ef4444" stroke="none" />
                        <Cell fill="#f59e0b" stroke="none" />
                        <Cell fill="#64748b" stroke="none" />
                      </Pie>
                      <Tooltip formatter={(val: number) => formatMinutes(val)} />
                      <Legend verticalAlign="bottom" height={120} align="center" iconType="circle" />

            <div className="p-4 border-b border-slate-100 flex items-center justify-between gap-4 bg-slate-50/50">
               <div className="relative flex-1 max-w-md">
                 <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" size={16} />
                 <input type="text" value={searchTerm} onChange={(e) => setSearchTerm(e.target.value)} placeholder="Pesquisar registros..." className="w-full pl-10 pr-4 py-2 bg-white border border-slate-200 rounded-xl text-xs font-bold text-slate-700 focus:outline-none" />
               </div>
               <button onClick={() => window.print()} className="flex items-center gap-2 px-4 py-2 bg-slate-800 text-white rounded-xl text-xs font-bold uppercase no-print"><Printer size={16} /> Imprimir</button>
            </div>
            <div className="overflow-x-auto">
              <table className="w-full text-left border-collapse min-w-[900px]">
                <thead>
                  <tr className="bg-slate-50 text-[9px] font-black text-slate-400 uppercase tracking-widest border-b border-slate-100">
                    <th className="px-4 py-3">Data</th>
                    <th className="px-4 py-3">Operador</th>
                    <th className="px-4 py-3">Máquina</th>
                    <th className="px-4 py-3 font-black text-blue-700">Líquido (KG)</th>
                    <th className="px-4 py-3 text-emerald-600">Eco A</th>
                    <th className="px-4 py-3 text-red-500">Borra</th>
                    <th className="px-4 py-3 text-right">Ações</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100 text-[10px] font-bold">
                  {filteredDataForTable.map((e) => (
                    <tr key={e.id} className="hover:bg-slate-50 transition-colors">
                      <td className="px-4 py-3">{e.date}</td>
                      <td className="px-4 py-3">{e.operator}</td>
                      <td className="px-4 py-3">{e.machine}</td>
                      <td className="px-4 py-3 font-black">{e.netWeight}</td>
                      <td className="px-4 py-3 text-emerald-600">{e.ecoA}</td>
                      <td className="px-4 py-3 text-red-500">{e.borraTotal}</td>
                      <td className="px-4 py-3 text-right space-x-2">
                        <button onClick={() => { setEditingEntry(e); setIsModalOpen(true); }} className="text-blue-500 p-1.5 hover:bg-blue-50 rounded-lg transition-all"><FileText size={14} /></button>
                        <button onClick={() => handleDeleteEntry(e.id)} className="text-red-400 hover:text-red-600 p-1.5 hover:bg-red-50 rounded-lg transition-all"><Trash2 size={14} /></button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        ) : (
          <div className="space-y-6">
            <div className="flex flex-col md:flex-row justify-between items-start md:items-center bg-white p-4 rounded-3xl shadow-sm border border-slate-100 gap-4">
               <div className="flex gap-4 overflow-x-auto no-scrollbar w-full md:w-auto">
                  <button onClick={() => setTeamViewMode('active')} className={`text-xs font-black uppercase tracking-widest pb-1 border-b-2 whitespace-nowrap transition-all ${teamViewMode === 'active' ? 'border-blue-600 text-blue-600' : 'border-transparent text-slate-400 hover:text-slate-600'}`}>Equipe Ativa</button>
                  <button onClick={() => setTeamViewMode('shifts')} className={`text-xs font-black uppercase tracking-widest pb-1 border-b-2 whitespace-nowrap transition-all ${teamViewMode === 'shifts' ? 'border-orange-600 text-orange-600' : 'border-transparent text-slate-400 hover:text-slate-600'}`}>Turnos</button>
                  <button onClick={() => setTeamViewMode('vacancies')} className={`text-xs font-black uppercase tracking-widest pb-1 border-b-2 whitespace-nowrap transition-all ${teamViewMode === 'vacancies' ? 'border-red-600 text-red-600' : 'border-transparent text-slate-400 hover:text-slate-600'}`}>Vagas</button>
               </div>
               <div className="flex items-center gap-2 w-full md:w-auto">
                 <button onClick={() => setIsShiftModalOpen(true)} className="flex-1 md:flex-none flex items-center justify-center gap-2 px-4 py-2 bg-orange-50 text-orange-600 border border-orange-100 rounded-xl text-[10px] font-black uppercase hover:bg-orange-100 transition-all">
                   <Clock size={16} /> Adicionar Turno
                 </button>
                 <button onClick={() => { setEditingEmployee(null); setIsEmployeeModalOpen(true); }} className="flex-1 md:flex-none flex items-center justify-center gap-2 px-4 py-2 bg-slate-50 text-slate-600 border border-slate-200 rounded-xl text-[10px] font-black uppercase hover:bg-slate-100 transition-all">
                   <Plus size={16} /> Nova Vaga / Colaborador
                 </button>
               </div>
            </div>
            
            {teamViewMode === 'active' ? (
              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
                {Object.keys(groupedPersonnel).map((key) => {
                  const emps = groupedPersonnel[key];
                  return (
                    <div key={key} className="bg-white rounded-3xl border border-slate-200 shadow-sm overflow-hidden">
                      <div className="bg-slate-50 px-4 py-2 font-black text-[9px] text-slate-500 uppercase border-b">{key}</div>
                      <div className="divide-y divide-slate-100">
                        {emps.map(emp => (
                          <div key={emp.id} className="p-4 hover:bg-slate-50 transition-all group">
                            <div className="flex items-center justify-between">
                              <div className="flex items-center gap-3">
                                 <div className={`w-10 h-10 rounded-2xl flex items-center justify-center font-bold text-white ${emp.group && emp.group.includes('Noturno') ? 'bg-indigo-600' : 'bg-orange-500'}`}>{emp.name[0]}</div>
                                 <div>
                                   <h4 className="font-bold text-slate-800 text-sm">{emp.name}</h4>
                                   <p className="text-[10px] text-slate-400 font-bold uppercase">{emp.role} {emp.status === 'Férias' && <span className="text-orange-500">(Férias)</span>}</p>
                                 </div>
                              </div>
                              <div className="flex gap-1 opacity-0 group-hover:opacity-100 transition-opacity">
                                <button onClick={() => { setEditingEmployee(emp); setIsEmployeeModalOpen(true); }} className="text-blue-500 p-1.5 hover:bg-blue-50 rounded-lg transition-all"><Edit2 size={14} /></button>
                                <button onClick={() => setEmployees(prev => prev.filter(x => x.id !== emp.id))} className="text-slate-300 hover:text-red-500 transition-colors p-1.5"><Trash2 size={14} /></button>
                              </div>
                            </div>
                          </div>
                        ))}
                      </div>
                    </div>
                  );
                })}
              </div>
            ) : teamViewMode === 'shifts' ? (
              <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
                {shifts.map(s => (
                  <div key={s.id} className="bg-white p-5 rounded-3xl border border-slate-100 shadow-sm space-y-3 relative group">
                    <div className="flex items-center gap-2">
                      <span className={`w-2 h-2 rounded-full ${s.type === 'Diurno' ? 'bg-orange-400' : 'bg-indigo-400'}`}></span>
                      <h4 className="font-black text-slate-800 text-sm uppercase tracking-tight">{s.name}</h4>
                    </div>
                    <div className="flex items-center justify-between">
                      <div><p className="text-[9px] text-slate-400 font-bold uppercase tracking-widest">Horário</p><p className="text-xs font-black text-slate-700">{s.startTime} - {s.endTime}</p></div>
                      <div className={`px-2 py-1 rounded-lg text-[8px] font-black uppercase ${s.type === 'Diurno' ? 'bg-orange-50 text-orange-600' : 'bg-indigo-50 text-indigo-600'}`}>{s.type}</div>
                    </div>
                    <button onClick={() => setShifts(prev => prev.filter(x => x.id !== s.id))} className="absolute top-4 right-4 text-slate-300 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-all"><Trash2 size={14} /></button>
                  </div>
                ))}
              </div>
            ) : (
              <div className="space-y-4">
                {vacancies.length === 0 ? (
                  <div className="bg-white p-12 text-center rounded-3xl border border-slate-100"><AlertTriangle size={32} className="mx-auto text-slate-300 mb-2" /><p className="text-slate-400 font-bold text-sm">Nenhuma vaga aberta no momento.</p></div>
                ) : (
                  <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                    {vacancies.map(v => (
                      <div key={v.id} className="bg-white p-5 rounded-3xl border border-slate-200 flex items-center justify-between hover:border-blue-200 transition-all group">
                        <div className="flex items-center gap-4">
                          <div className="w-12 h-12 bg-red-50 text-red-500 rounded-2xl flex items-center justify-center font-black">?</div>
                          <div>
                            <h4 className="font-black text-slate-800 text-sm uppercase tracking-tight">{v.name}</h4>
                            <p className="text-[10px] text-slate-400 font-bold uppercase">Setor: {v.sector} | {v.group} | Aberta desde: {v.openSince}</p>
                            <div className="mt-1 inline-flex px-2 py-0.5 bg-red-50 text-red-600 rounded-lg text-[8px] font-black uppercase">{v.role}</div>
                          </div>
                        </div>
                        <div className="flex gap-2">
                           <button onClick={() => { setEditingEmployee({...v, status: 'Vaga Aberta'}); setIsEmployeeModalOpen(true); }} className="text-blue-500 p-2 hover:bg-blue-50 rounded-xl transition-all"><Edit2 size={18} /></button>
                           <button onClick={() => setVacancies(prev => prev.filter(x => x.id !== v.id))} className="text-slate-200 hover:text-red-500 transition-colors p-2"><Trash2 size={18} /></button>
                        </div>
                      </div>
                    ))}
                  </div>
                )}
              </div>
            )}
          </div>
        )}
      </main>

      {/* MENU DE OPÇÕES (SIDEBAR) */}
      <div className={`fixed inset-0 z-50 transition-all duration-300 ${isMenuOpen ? 'visible' : 'invisible'}`}>
        <div className={`absolute inset-0 bg-slate-900/40 backdrop-blur-sm transition-opacity duration-300 ${isMenuOpen ? 'opacity-100' : 'opacity-0'}`} onClick={() => setIsMenuOpen(false)} />
        <div className={`absolute right-0 top-0 bottom-0 w-80 bg-white shadow-2xl transition-transform duration-300 transform p-6 flex flex-col gap-6 ${isMenuOpen ? 'translate-x-0' : 'translate-x-full'} overflow-y-auto custom-scrollbar`}>
          <div className="flex items-center justify-between">
            <h2 className="text-lg font-black text-slate-800 flex items-center gap-2">
              <Settings size={20} className="text-blue-600" /> MENU DE OPÇÕES
            </h2>
            <button onClick={() => setIsMenuOpen(false)} className="p-2 text-slate-400"><X size={24} /></button>
          </div>
          
          <div className="space-y-8 divide-y divide-slate-100">
             <div className="pt-4 space-y-4">
                <h3 className="text-[10px] font-black text-slate-400 uppercase tracking-widest flex items-center gap-2"><Users2 size={14} /> Operadores</h3>
                <div className="flex gap-2">
                  <input type="text" value={newOpName} onChange={e => setNewOpName(e.target.value)} placeholder="Novo operador..." className="flex-1 bg-slate-50 border border-slate-200 rounded-xl px-3 py-2 text-xs font-bold outline-none focus:border-blue-500" />
                  <button onClick={() => { if (newOpName.trim()) { setOperators(prev => [...prev, newOpName.trim()].sort()); setNewOpName(''); } }} className="bg-blue-600 text-white p-2 rounded-xl"><Plus size={16} /></button>
                </div>
                <div className="space-y-2 max-h-40 overflow-y-auto pr-2 custom-scrollbar">
                  {operators.map(op => (
                    <div key={op} className="flex items-center justify-between bg-slate-50 p-2 rounded-xl border border-slate-100 group">
                      <span className="text-[11px] font-bold text-slate-700">{op}</span>
                      <button onClick={() => setOperators(prev => prev.filter(x => x !== op))} className="text-slate-300 hover:text-red-500"><Trash2 size={14} /></button>
                    </div>
                  ))}
                </div>
             </div>

             <div className="pt-6 space-y-4">
                <h3 className="text-[10px] font-black text-slate-400 uppercase tracking-widest flex items-center gap-2"><Cpu size={14} /> Máquinas</h3>
                <div className="flex gap-2">
                  <input type="text" value={newMachineName} onChange={e => setNewMachineName(e.target.value)} placeholder="Nova máquina..." className="flex-1 bg-slate-50 border border-slate-200 rounded-xl px-3 py-2 text-xs font-bold outline-none focus:border-blue-500" />
                  <button onClick={() => { if (newMachineName.trim()) { setMachines(prev => [...prev, newMachineName.trim()].sort()); setNewMachineName(''); } }} className="bg-emerald-600 text-white p-2 rounded-xl"><Plus size={16} /></button>
                </div>
                <div className="space-y-2 max-h-32 overflow-y-auto pr-2 custom-scrollbar">
                  {machines.map(m => (
                    <div key={m} className="flex items-center justify-between bg-slate-50 p-2 rounded-xl border border-slate-100 group">
                      <span className="text-[11px] font-bold text-slate-700">{m}</span>
                      <button onClick={() => setMachines(prev => prev.filter(x => x !== m))} className="text-slate-300 hover:text-red-500"><Trash2 size={14} /></button>
                    </div>
                  ))}
                </div>
             </div>

             <div className="pt-6 space-y-2">
                <label className="text-[10px] font-black text-slate-400 uppercase tracking-widest flex items-center gap-2"><BarChart3 size={14} /> Meta Mensal (KG)</label>
                <input type="number" value={goalValue} onChange={(e) => setGoalValue(parseFloat(e.target.value) || 0)} className="w-full bg-slate-50 border border-slate-200 rounded-xl px-4 py-2 text-sm font-black text-blue-600 outline-none focus:border-blue-500" />
             </div>

             <div className="pt-6 space-y-4">
               <h3 className="text-[10px] font-black text-slate-400 uppercase tracking-widest flex items-center gap-2"><RefreshCw size={14} /> Sincronização de Dados</h3>
               <div className="grid grid-cols-2 gap-2">
                 <button onClick={handleExportData} className="flex items-center justify-center gap-2 bg-slate-100 hover:bg-blue-600 hover:text-white text-slate-600 py-2.5 rounded-xl text-[10px] font-black uppercase transition-all border border-slate-200"><Download size={14} /> Exportar</button>
                 <button onClick={() => fileInputRef.current?.click()} className="flex items-center justify-center gap-2 bg-slate-100 hover:bg-emerald-600 hover:text-white text-slate-600 py-2.5 rounded-xl text-[10px] font-black uppercase transition-all border border-slate-200"><Upload size={14} /> Importar</button>
               </div>
               <input type="file" ref={fileInputRef} onChange={handleImportData} className="hidden" accept=".json" />
             </div>

             <div className="pt-6">
                <button onClick={() => { localStorage.removeItem('manupackaging_session'); setIsAuthenticated(false); }} className="w-full flex items-center justify-center gap-2 py-3 bg-red-50 text-red-600 rounded-xl text-xs font-black uppercase hover:bg-red-100 transition-all"><LogOut size={16} /> Sair do Sistema</button>
             </div>
          </div>
        </div>
      </div>

      <ShiftModal isOpen={isShiftModalOpen} onClose={() => setIsShiftModalOpen(false)} onSave={handleSaveShift} />
      <EmployeeModal 
        isOpen={isEmployeeModalOpen} 
        onClose={() => setIsEmployeeModalOpen(false)} 
        onSave={handleSaveEmployee} 
        availableShifts={shifts} 
        availableMachines={machines} 
        editingEmployee={editingEmployee}
      />
      <LaunchModal 
        isOpen={isModalOpen} 
        onClose={() => { setIsModalOpen(false); setEditingEntry(null); }} 
        onSave={handleSaveEntry} 
        operators={operators} 
        activeMachines={machines} 
        availableShifts={shifts}
        initialData={editingEntry} 
      />
    </div>
  );
};

export default App;
