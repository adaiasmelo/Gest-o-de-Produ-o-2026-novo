
import React, { useState, useMemo, useEffect } from 'react';
import { Menu, Plus, TrendingUp, BarChart2, Calendar, RotateCcw, Download, Clock, Zap, Trash2, Cpu, BrainCircuit, Sparkles, CheckCircle2, AlertTriangle } from 'lucide-react';
import { BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, Cell, PieChart, Pie, Legend } from 'recharts';
import { GoogleGenAI } from "@google/genai";
import { ProductionEntry } from './types';
import { INITIAL_DATA, GOAL_VALUE } from './constants';
import LaunchModal from './components/LaunchModal';
import StatsCard from './components/StatsCard';

const App: React.FC = () => {
  const [productionData, setProductionData] = useState<ProductionEntry[]>(() => {
    const saved = localStorage.getItem('manupackaging_data');
    return saved ? JSON.parse(saved) : INITIAL_DATA;
  });

  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isAiLoading, setIsAiLoading] = useState(false);
  const [aiInsights, setAiInsights] = useState<string | null>(null);
  
  const [startDate, setStartDate] = useState<string>('');
  const [endDate, setEndDate] = useState<string>('');
  const [selectedMachine, setSelectedMachine] = useState<string>('Todas');

  useEffect(() => {
    localStorage.setItem('manupackaging_data', JSON.stringify(productionData));
  }, [productionData]);

  const filteredData = useMemo(() => {
    return productionData.filter(entry => {
      const entryDate = entry.date;
      const matchesStart = !startDate || entryDate >= startDate;
      const matchesEnd = !endDate || entryDate <= endDate;
      const matchesMachine = selectedMachine === 'Todas' || entry.machine === selectedMachine;
      return matchesStart && matchesEnd && matchesMachine;
    });
  }, [productionData, startDate, endDate, selectedMachine]);

  const stats = useMemo(() => {
    const s = filteredData.reduce((acc, curr) => ({
      totalNet: acc.totalNet + curr.netWeight,
      totalGross: acc.totalGross + curr.grossWeight,
      ecoA: acc.ecoA + curr.ecoA,
      ecoBP: acc.ecoBP + curr.ecoBP,
      ecoBM: acc.ecoBM + curr.ecoBM,
      borra: acc.borra + curr.borraTotal,
      paradas: acc.paradas + curr.manutencaoMin + curr.processoMin + curr.outrosMin,
      manut: acc.manut + curr.manutencaoMin,
      proc: acc.proc + curr.processoMin,
      outros: acc.outros + curr.outrosMin,
    }), { totalNet: 0, totalGross: 0, ecoA: 0, ecoBP: 0, ecoBM: 0, borra: 0, paradas: 0, manut: 0, proc: 0, outros: 0 });
    
    const qualityEfficiency = s.totalGross > 0 ? (s.totalNet / s.totalGross) * 100 : 0;
    return { ...s, qualityEfficiency };
  }, [filteredData]);

  // Real-time Machine Status Logic
  const machineStatus = useMemo(() => {
    const machines = ['Cast 1', 'Cast 2'];
    return machines.map(m => {
      const lastEntry = [...productionData].reverse().find(e => e.machine === m);
      const totalDowntime = lastEntry ? (lastEntry.manutencaoMin + lastEntry.processoMin + lastEntry.outrosMin) : 0;
      
      let status = 'Operando';
      let color = 'emerald';
      
      if (!lastEntry) { status = 'Sem Dados'; color = 'slate'; }
      else if (lastEntry.manutencaoMin > 30) { status = 'Manutenção'; color = 'orange'; }
      else if (totalDowntime > 60) { status = 'Alerta'; color = 'amber'; }

      return { name: m, status, color, lastOp: lastEntry?.operator || '---' };
    });
  }, [productionData]);

  const generateAiInsights = async () => {
    setIsAiLoading(true);
    try {
      const ai = new GoogleGenAI({ apiKey: process.env.API_KEY });
      const prompt = `Analise os seguintes dados de produção industrial e forneça 3 insights rápidos e acionáveis:
        Dados: Produção Líquida Total: ${stats.totalNet}kg, Eficiência: ${stats.qualityEfficiency.toFixed(1)}%, 
        Total Paradas: ${stats.paradas}min (Manutenção: ${stats.manut}m, Processo: ${stats.proc}m).
        Considere que a meta é ${GOAL_VALUE}kg. Responda de forma curta e profissional.`;

      const response = await ai.models.generateContent({
        model: 'gemini-3-flash-preview',
        contents: prompt,
      });

      setAiInsights(response.text || "Não foi possível gerar insights no momento.");
    } catch (error) {
      setAiInsights("Erro ao conectar com o analista de IA. Verifique sua conexão.");
    } finally {
      setIsAiLoading(false);
    }
  };

  const handleSaveEntry = (newEntry: Omit<ProductionEntry, 'id'>) => {
    const entryWithId: ProductionEntry = {
      ...newEntry,
      id: Math.random().toString(36).substr(2, 9),
    };
    setProductionData(prev => [...prev, entryWithId]);
  };

  const handleDeleteEntry = (id: string) => {
    if (confirm('Deseja realmente excluir este lançamento?')) {
      setProductionData(prev => prev.filter(e => e.id !== id));
    }
  };

  return (
    <div className="min-h-screen bg-slate-50 pb-20">
      <header className="bg-white px-6 py-4 flex items-center justify-between sticky top-0 z-40 border-b border-slate-100">
        <h1 className="text-xl font-bold text-slate-800 flex items-center gap-2">
          <div className="w-8 h-8 bg-blue-600 rounded-lg flex items-center justify-center text-white">
            <Cpu size={18} />
          </div>
          <span className="tracking-tight">Manupackaging <span className="text-blue-600">Pro</span></span>
        </h1>
        <div className="flex items-center gap-4">
           <button onClick={() => setIsModalOpen(true)} className="bg-blue-600 hover:bg-blue-700 text-white font-bold px-4 py-2 rounded-lg flex items-center gap-2 shadow-lg shadow-blue-200 transition-all active:scale-95 text-sm">
              <Plus size={18} />
              <span className="hidden sm:inline">Lançar</span>
           </button>
        </div>
      </header>

      <main className="max-w-4xl mx-auto p-4 space-y-6">
        
        {/* Machine Status Trackers */}
        <div className="grid grid-cols-2 gap-4">
          {machineStatus.map((m) => (
            <div key={m.name} className="bg-white rounded-2xl p-4 shadow-sm border border-slate-100 flex items-center gap-4">
              <div className={`w-10 h-10 rounded-full flex items-center justify-center bg-${m.color}-100 text-${m.color}-600`}>
                {m.status === 'Operando' ? <CheckCircle2 size={20} /> : <AlertTriangle size={20} />}
              </div>
              <div>
                <span className="text-[10px] font-black text-slate-400 uppercase tracking-widest">{m.name}</span>
                <div className="flex items-center gap-2">
                  <span className="text-sm font-bold text-slate-700">{m.status}</span>
                  <div className={`w-2 h-2 rounded-full bg-${m.color}-500 animate-pulse`}></div>
                </div>
                <span className="text-[9px] text-slate-400 italic">Último op: {m.lastOp}</span>
              </div>
            </div>
          ))}
        </div>

        {/* AI Insights Card */}
        <div className="relative group">
          <div className="absolute -inset-0.5 bg-gradient-to-r from-blue-500 to-purple-600 rounded-3xl blur opacity-20 group-hover:opacity-30 transition duration-1000"></div>
          <div className="relative bg-white rounded-3xl p-6 shadow-sm border border-slate-100 overflow-hidden">
            <div className="flex items-center justify-between mb-4">
               <div className="flex items-center gap-2">
                  <div className="p-2 bg-purple-50 rounded-lg text-purple-600">
                    <BrainCircuit size={20} />
                  </div>
                  <div>
                    <h3 className="font-black text-slate-800 text-sm uppercase tracking-tight">Analista de IA</h3>
                    <p className="text-[10px] text-slate-400 font-bold uppercase">Processamento Gemini v3</p>
                  </div>
               </div>
               <button 
                onClick={generateAiInsights}
                disabled={isAiLoading}
                className="bg-slate-900 hover:bg-black text-white px-4 py-2 rounded-xl text-xs font-bold flex items-center gap-2 transition-all disabled:opacity-50"
               >
                 {isAiLoading ? <RotateCcw className="animate-spin" size={14} /> : <Sparkles size={14} />}
                 {aiInsights ? 'Atualizar Insights' : 'Gerar Insights'}
               </button>
            </div>
            
            {aiInsights ? (
              <div className="bg-slate-50 rounded-2xl p-4 border border-slate-100 text-sm text-slate-600 leading-relaxed italic animate-in fade-in slide-in-from-bottom-2">
                "{aiInsights}"
              </div>
            ) : (
              <div className="text-center py-4 text-slate-400 text-xs italic">
                Clique para analisar os gargalos de produção deste período...
              </div>
            )}
          </div>
        </div>

        {/* Global Stats Grid */}
        <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
          <div className="bg-white p-4 rounded-2xl shadow-sm border border-slate-100">
            <span className="text-[9px] font-black text-slate-400 uppercase block">Eficiência de Qualidade</span>
            <div className="flex items-baseline gap-1">
              <span className="text-xl font-black text-emerald-500">{stats.qualityEfficiency.toFixed(1)}%</span>
              <TrendingUp size={12} className="text-emerald-400" />
            </div>
          </div>
          <div className="bg-white p-4 rounded-2xl shadow-sm border border-slate-100">
            <span className="text-[9px] font-black text-slate-400 uppercase block">Total Borra (Perda)</span>
            <span className="text-xl font-black text-red-500">{stats.borra.toLocaleString('pt-BR')} kg</span>
          </div>
          <div className="bg-white p-4 rounded-2xl shadow-sm border border-slate-100">
            <span className="text-[9px] font-black text-slate-400 uppercase block">Horas de Parada</span>
            <span className="text-xl font-black text-orange-500">{(stats.paradas / 60).toFixed(1)}h</span>
          </div>
          <div className="bg-white p-4 rounded-2xl shadow-sm border border-slate-100">
            <span className="text-[9px] font-black text-slate-400 uppercase block">Produção Total</span>
            <span className="text-xl font-black text-blue-600">{(stats.totalNet / 1000).toFixed(1)}t</span>
          </div>
        </div>

        {/* Dynamic Filters */}
        <div className="bg-white rounded-2xl p-4 shadow-sm border border-slate-200 flex flex-col md:flex-row items-end md:items-center gap-4">
           <div className="flex bg-slate-100 p-1 rounded-xl w-full md:w-auto">
              {['Todas', 'Cast 1', 'Cast 2'].map(m => (
                <button 
                  key={m} onClick={() => setSelectedMachine(m)}
                  className={`flex-1 md:flex-none px-4 py-2 rounded-lg text-[10px] font-black uppercase tracking-tight transition-all ${selectedMachine === m ? 'bg-white text-blue-600 shadow-sm' : 'text-slate-500'}`}
                >
                  {m}
                </button>
              ))}
           </div>
           <div className="flex-1 grid grid-cols-2 gap-2 w-full">
              <input type="date" value={startDate} onChange={(e) => setStartDate(e.target.value)} className="bg-slate-50 border border-slate-200 rounded-xl px-3 py-2 text-xs font-bold text-slate-600 focus:outline-none" />
              <input type="date" value={endDate} onChange={(e) => setEndDate(e.target.value)} className="bg-slate-50 border border-slate-200 rounded-xl px-3 py-2 text-xs font-bold text-slate-600 focus:outline-none" />
           </div>
           <button onClick={() => { setStartDate(''); setEndDate(''); setSelectedMachine('Todas'); }} className="p-3 bg-slate-100 text-slate-500 rounded-xl hover:bg-slate-200 transition-colors">
              <RotateCcw size={16} />
           </button>
        </div>

        {/* Goal Tracking */}
        <div className="bg-white rounded-2xl p-6 shadow-sm border border-blue-500/10">
          <div className="flex items-center justify-between mb-4">
            <div>
               <span className="text-[10px] font-black text-blue-500 uppercase tracking-widest">Atingimento da Meta</span>
               <div className="flex items-baseline gap-2">
                  <span className="text-4xl font-black text-slate-800 tracking-tighter">{stats.totalNet.toLocaleString('pt-BR')}</span>
                  <span className="text-slate-400 font-medium text-xs">/ {GOAL_VALUE.toLocaleString('pt-BR')} kg</span>
               </div>
            </div>
            <div className="text-right">
              <span className="text-3xl font-black text-blue-600">{Math.round((stats.totalNet / GOAL_VALUE) * 100)}%</span>
            </div>
          </div>
          <div className="w-full bg-slate-100 h-3 rounded-full overflow-hidden">
            <div className="h-full bg-gradient-to-r from-blue-600 to-indigo-400 rounded-full transition-all duration-1000 shadow-[0_0_10px_rgba(37,99,235,0.3)]" style={{ width: `${Math.min((stats.totalNet / GOAL_VALUE) * 100, 100)}%` }}></div>
          </div>
        </div>

        {/* Charts Row */}
        <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
          <div className="bg-white rounded-2xl p-6 shadow-sm border border-slate-100">
            <h3 className="font-black text-slate-800 text-xs uppercase tracking-widest mb-6 flex items-center gap-2">
               <BarChart2 size={16} className="text-blue-500" /> Rendimento Operacional
            </h3>
            <div className="h-[240px] w-full">
              {filteredData.length > 0 ? (
                <ResponsiveContainer width="100%" height="100%">
                  <BarChart data={filteredData.map(e => ({ name: e.operator, kg: e.netWeight }))}>
                    <CartesianGrid vertical={false} stroke="#f1f5f9" />
                    <XAxis dataKey="name" axisLine={false} tickLine={false} tick={{ fontSize: 9, fontWeight: 'bold', fill: '#94a3b8' }} />
                    <YAxis axisLine={false} tickLine={false} tick={{ fontSize: 9, fill: '#94a3b8' }} />
                    <Tooltip cursor={{ fill: '#f8fafc' }} contentStyle={{ borderRadius: '12px', border: 'none', boxShadow: '0 4px 12px rgba(0,0,0,0.05)' }} />
                    <Bar dataKey="kg" fill="#3b82f6" radius={[4, 4, 0, 0]} barSize={25} />
                  </BarChart>
                </ResponsiveContainer>
              ) : <NoDataPlaceholder /> }
            </div>
          </div>

          <div className="bg-white rounded-2xl p-6 shadow-sm border border-slate-100">
            <h3 className="font-black text-slate-800 text-xs uppercase tracking-widest mb-6 flex items-center gap-2">
               <Clock size={16} className="text-orange-500" /> Motivos de Parada
            </h3>
            <div className="h-[240px] w-full">
              {stats.paradas > 0 ? (
                <ResponsiveContainer width="100%" height="100%">
                  <PieChart>
                    <Pie 
                      data={[
                        { name: 'Manut', value: stats.manut, color: '#f97316' },
                        { name: 'Proc', value: stats.proc, color: '#f59e0b' },
                        { name: 'Outros', value: stats.outros, color: '#94a3b8' }
                      ].filter(d => d.value > 0)} 
                      cx="50%" cy="50%" innerRadius={60} outerRadius={85} paddingAngle={8} dataKey="value"
                    >
                      { (d: any) => <Cell key={d.name} fill={d.color} /> }
                      <Cell fill="#f97316" /><Cell fill="#f59e0b" /><Cell fill="#94a3b8" />
                    </Pie>
                    <Tooltip />
                    <Legend verticalAlign="bottom" height={36} iconType="circle" wrapperStyle={{ fontSize: '10px', fontWeight: 'bold', textTransform: 'uppercase' }} />
                  </PieChart>
                </ResponsiveContainer>
              ) : <NoDataPlaceholder /> }
            </div>
          </div>
        </div>

        {/* History / Recent Entries */}
        <div className="bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden">
          <div className="p-4 border-b border-slate-100 bg-slate-50 flex items-center justify-between">
            <h3 className="font-black text-slate-800 text-[10px] uppercase tracking-widest">Lançamentos Recentes</h3>
            <button onClick={exportToCSV} className="text-blue-600 text-[10px] font-black uppercase flex items-center gap-1 hover:underline">
              <Download size={12} /> CSV
            </button>
          </div>
          <div className="divide-y divide-slate-50">
            {filteredData.slice().reverse().map((entry) => (
              <div key={entry.id} className="p-4 flex items-center justify-between hover:bg-slate-50/50 transition-colors">
                <div className="flex items-center gap-4">
                  <div className={`w-2 h-10 rounded-full ${entry.shift === 'Diurno' ? 'bg-amber-400' : 'bg-indigo-600'}`}></div>
                  <div>
                    <div className="flex items-center gap-2">
                      <span className="font-black text-slate-800 text-sm">{entry.operator}</span>
                      <span className="text-[10px] bg-slate-100 text-slate-500 px-2 py-0.5 rounded font-bold uppercase">{entry.machine}</span>
                    </div>
                    <p className="text-[10px] text-slate-400 font-medium">{entry.date} • {entry.shift}</p>
                  </div>
                </div>
                <div className="text-right flex items-center gap-6">
                  <div>
                    <span className="block font-black text-blue-600 text-sm">{entry.netWeight.toLocaleString('pt-BR')} kg</span>
                    <span className="text-[9px] text-slate-400 font-bold uppercase">Líquido</span>
                  </div>
                  <button onClick={() => handleDeleteEntry(entry.id)} className="text-slate-300 hover:text-red-500 transition-colors">
                    <Trash2 size={16} />
                  </button>
                </div>
              </div>
            ))}
          </div>
        </div>
      </main>

      <LaunchModal isOpen={isModalOpen} onClose={() => setIsModalOpen(false)} onSave={handleSaveEntry} />
      
      {/* Footer Branding */}
      <footer className="py-8 text-center">
        <p className="text-[10px] font-black text-slate-300 uppercase tracking-[0.2em]">Manupackaging Industrial System • 2026</p>
      </footer>
    </div>
  );

  function exportToCSV() {
    const headers = ['Data', 'Operador', 'Maquina', 'Turno', 'Peso Liquido', 'Paradas'];
    const rows = filteredData.map(e => [e.date, e.operator, e.machine, e.shift, e.netWeight, e.manutencaoMin + e.processoMin + e.outrosMin]);
    const csvContent = [headers, ...rows].map(r => r.join(';')).join('\n');
    const blob = new Blob([csvContent], { type: 'text/csv' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `relatorio_producao_${new Date().toISOString().split('T')[0]}.csv`;
    a.click();
  }
};

const NoDataPlaceholder = () => (
  <div className="h-full flex flex-col items-center justify-center text-slate-300 gap-2">
    <BarChart2 size={32} strokeWidth={1} />
    <span className="text-[10px] font-black uppercase tracking-tighter">Dados insuficientes</span>
  </div>
);

export default App;
