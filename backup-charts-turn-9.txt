
import React, { useState, useMemo } from 'react';
import { Menu, Plus, TrendingUp, BarChart2, Calendar, Pencil, Filter, RotateCcw, PieChart as PieChartIcon } from 'lucide-react';
import { BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, Cell, PieChart, Pie, Legend } from 'recharts';
import { ProductionEntry } from './types';
import { INITIAL_DATA, GOAL_VALUE } from './constants';
import LaunchModal from './components/LaunchModal';
import StatsCard from './components/StatsCard';

const App: React.FC = () => {
  const [productionData, setProductionData] = useState<ProductionEntry[]>(INITIAL_DATA);
  const [isModalOpen, setIsModalOpen] = useState(false);
  
  // Filter States
  const [startDate, setStartDate] = useState<string>('');
  const [endDate, setEndDate] = useState<string>('');

  // Filter Logic
  const filteredData = useMemo(() => {
    return productionData.filter(entry => {
      const entryDate = entry.date;
      const matchesStart = !startDate || entryDate >= startDate;
      const matchesEnd = !endDate || entryDate <= endDate;
      return matchesStart && matchesEnd;
    });
  }, [productionData, startDate, endDate]);

  const stats = useMemo(() => {
    return filteredData.reduce((acc, curr) => ({
      totalNet: acc.totalNet + curr.netWeight,
      ecoA: acc.ecoA + curr.ecoA,
      ecoBP: acc.ecoBP + curr.ecoBP,
      ecoBM: acc.ecoBM + curr.ecoBM,
      borra: acc.borra + curr.borraTotal,
      paradas: acc.paradas + curr.manutencaoMin + curr.processoMin + curr.outrosMin,
    }), { totalNet: 0, ecoA: 0, ecoBP: 0, ecoBM: 0, borra: 0, paradas: 0 });
  }, [filteredData]);

  const percentageGoal = Math.round((stats.totalNet / GOAL_VALUE) * 100);

  const chartData = useMemo(() => {
    return filteredData.map(entry => ({
      name: entry.operator,
      value: entry.netWeight,
      borra: entry.borraTotal
    }));
  }, [filteredData]);

  // Data for Loss/Material Distribution Chart
  const lossDistributionData = useMemo(() => {
    return [
      { name: 'Eco A', value: stats.ecoA, color: '#10b981' }, // Emerald-500
      { name: 'Eco B (Prod)', value: stats.ecoBP, color: '#fbbf24' }, // Amber-400
      { name: 'Eco B (Manut)', value: stats.ecoBM, color: '#fb923c' }, // Orange-400
      { name: 'Borra', value: stats.borra, color: '#ef4444' }, // Red-500
    ].filter(item => item.value > 0);
  }, [stats]);

  const handleSaveEntry = (newEntry: Omit<ProductionEntry, 'id'>) => {
    const entryWithId: ProductionEntry = {
      ...newEntry,
      id: Math.random().toString(36).substr(2, 9),
    };
    setProductionData(prev => [...prev, entryWithId]);
  };

  const clearFilters = () => {
    setStartDate('');
    setEndDate('');
  };

  return (
    <div className="min-h-screen bg-slate-50 pb-20">
      {/* Header */}
      <header className="bg-white px-6 py-4 flex items-center justify-between sticky top-0 z-40 border-b border-slate-100">
        <h1 className="text-xl font-bold text-slate-800 flex items-center gap-2">
          Controle de Produção 2026
        </h1>
        <button className="text-slate-600 p-1 hover:bg-slate-100 rounded-lg">
          <Menu size={24} />
        </button>
      </header>

      <main className="max-w-4xl mx-auto p-4 space-y-6">
        {/* Banner Card */}
        <div className="relative overflow-hidden rounded-3xl bg-slate-700 p-8 text-white shadow-xl min-h-[160px] flex flex-col justify-between">
          <div className="relative z-10 flex items-center gap-3">
             <div className="w-10 h-10 bg-white/20 rounded-lg backdrop-blur-md flex items-center justify-center">
                <div className="w-6 h-6 border-4 border-white border-t-transparent rounded-full opacity-60"></div>
             </div>
             <span className="text-xl font-bold tracking-tight">Manupackaging</span>
          </div>
          
          <div className="flex justify-end relative z-10 mt-4">
            <button 
              onClick={() => setIsModalOpen(true)}
              className="bg-yellow-400 hover:bg-yellow-500 text-slate-900 font-bold px-6 py-2 rounded-full flex items-center gap-2 transition-transform active:scale-95 shadow-lg shadow-yellow-900/20"
            >
              <Plus size={20} />
              Lançar
            </button>
          </div>

          <div className="absolute top-[-20%] right-[-10%] w-64 h-64 bg-slate-600 rounded-full opacity-20 blur-3xl pointer-events-none"></div>
          <div className="absolute bottom-[-10%] left-[5%] w-32 h-32 bg-slate-500 rounded-full opacity-10 blur-2xl pointer-events-none"></div>
        </div>

        {/* Filters Section */}
        <div className="bg-white rounded-2xl p-4 shadow-sm border border-slate-200 flex flex-col md:flex-row items-end md:items-center gap-4">
          <div className="flex-1 w-full space-y-1">
            <label className="text-[10px] font-bold text-slate-500 uppercase flex items-center gap-1">
              <Calendar size={12} /> Início
            </label>
            <input 
              type="date" 
              value={startDate}
              onChange={(e) => setStartDate(e.target.value)}
              className="w-full bg-slate-50 border border-slate-200 rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500/20 text-slate-700 font-medium"
            />
          </div>
          <div className="flex-1 w-full space-y-1">
            <label className="text-[10px] font-bold text-slate-500 uppercase flex items-center gap-1">
              <Calendar size={12} /> Fim
            </label>
            <input 
              type="date" 
              value={endDate}
              onChange={(e) => setEndDate(e.target.value)}
              className="w-full bg-slate-50 border border-slate-200 rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-500/20 text-slate-700 font-medium"
            />
          </div>
          <div className="flex gap-2 w-full md:w-auto mt-1">
            <button 
              onClick={clearFilters}
              disabled={!startDate && !endDate}
              className="flex-1 md:flex-none flex items-center justify-center gap-2 px-4 py-2 bg-slate-100 hover:bg-slate-200 disabled:opacity-50 disabled:cursor-not-allowed text-slate-600 rounded-lg text-sm font-bold transition-colors"
            >
              <RotateCcw size={16} /> Limpar
            </button>
          </div>
        </div>

        {/* Goal Section */}
        <div className="bg-white rounded-2xl p-6 shadow-sm border border-blue-500/20">
          <div className="flex items-center justify-between mb-2">
            <div className="flex items-center gap-2">
               <span className="text-[10px] font-bold text-blue-500 uppercase tracking-widest">Meta {startDate || endDate ? '(Filtrado)' : ''}</span>
               <button className="flex items-center gap-1 text-[9px] bg-blue-50 text-blue-600 px-2 py-0.5 rounded font-bold">
                 EDITAR <Pencil size={8} />
               </button>
            </div>
            <span className="text-2xl font-black text-blue-600">{percentageGoal}%</span>
          </div>
          
          <div className="flex items-baseline gap-2 mb-4">
             <span className="text-4xl font-black text-slate-800 tracking-tighter">
               {stats.totalNet.toLocaleString('pt-BR')}
             </span>
             <span className="text-slate-400 font-medium text-sm">/ {GOAL_VALUE.toLocaleString('pt-BR')} kg</span>
          </div>

          <div className="w-full bg-slate-100 h-3 rounded-full overflow-hidden">
            <div 
              className="h-full bg-emerald-400 rounded-full transition-all duration-1000 ease-out"
              style={{ width: `${Math.min(percentageGoal, 100)}%` }}
            ></div>
          </div>
          <div className="flex justify-between mt-2">
            <span className="text-[10px] text-slate-400">Base Mensal: <span className="font-bold">{GOAL_VALUE.toLocaleString('pt-BR')} kg</span></span>
            <span className="text-[10px] text-slate-400 italic">
              {filteredData.length} lançamento(s)
            </span>
          </div>
        </div>

        {/* Efficiency Chart */}
        <div className="bg-white rounded-2xl p-6 shadow-sm">
          <div className="flex items-center justify-between mb-6">
            <div className="flex items-center gap-2">
               <TrendingUp size={18} className="text-blue-500" />
               <h3 className="font-bold text-slate-800">Evolução e Eficiência</h3>
            </div>
          </div>

          <div className="h-[240px] w-full">
            {filteredData.length > 0 ? (
              <ResponsiveContainer width="100%" height="100%">
                <BarChart data={chartData} margin={{ top: 20, right: 0, left: -20, bottom: 0 }}>
                  <CartesianGrid vertical={false} strokeDasharray="3 3" stroke="#f1f5f9" />
                  <XAxis 
                    dataKey="name" 
                    axisLine={false} 
                    tickLine={false} 
                    tick={{ fontSize: 10, fill: '#94a3b8' }} 
                  />
                  <YAxis 
                    axisLine={false} 
                    tickLine={false} 
                    tick={{ fontSize: 10, fill: '#94a3b8' }} 
                  />
                  <Tooltip 
                    cursor={{ fill: 'transparent' }}
                    content={({ active, payload }) => {
                      if (active && payload && payload.length) {
                        return (
                          <div className="bg-white p-2 shadow-xl border border-slate-100 rounded-lg text-xs">
                            <p className="font-bold text-slate-700">{payload[0].payload.name}</p>
                            <p className="text-blue-600">{payload[0].value?.toLocaleString('pt-BR')} kg</p>
                          </div>
                        );
                      }
                      return null;
                    }}
                  />
                  <Bar dataKey="value" fill="#93c5fd" radius={[4, 4, 0, 0]} barSize={24}>
                    {chartData.map((entry, index) => (
                      <Cell key={`cell-${index}`} fill={index % 2 === 0 ? '#60a5fa' : '#93c5fd'} />
                    ))}
                  </Bar>
                </BarChart>
              </ResponsiveContainer>
            ) : (
              <div className="h-full flex flex-col items-center justify-center text-slate-400 gap-2">
                <BarChart2 size={40} className="opacity-20" />
                <span className="text-xs font-medium">Nenhum dado no período selecionado</span>
              </div>
            )}
          </div>
        </div>

        {/* Analysis Section - Loss/Material Distribution */}
        <div className="bg-white rounded-2xl p-6 shadow-sm">
          <div className="flex items-center gap-2 mb-8">
             <PieChartIcon size={18} className="text-purple-500" />
             <h3 className="font-bold text-slate-800">Distribuição de Materiais e Perdas</h3>
          </div>

          <div className="h-[280px] w-full">
            {lossDistributionData.length > 0 ? (
              <ResponsiveContainer width="100%" height="100%">
                <PieChart>
                  <Pie
                    data={lossDistributionData}
                    cx="50%"
                    cy="45%"
                    innerRadius={60}
                    outerRadius={80}
                    paddingAngle={5}
                    dataKey="value"
                  >
                    {lossDistributionData.map((entry, index) => (
                      <Cell key={`cell-${index}`} fill={entry.color} />
                    ))}
                  </Pie>
                  <Tooltip 
                    content={({ active, payload }) => {
                      if (active && payload && payload.length) {
                        const data = payload[0].payload;
                        return (
                          <div className="bg-white p-3 shadow-xl border border-slate-100 rounded-xl text-xs">
                            <p className="font-bold text-slate-700 mb-1">{data.name}</p>
                            <p className="font-bold" style={{ color: data.color }}>
                              {data.value.toLocaleString('pt-BR')} kg
                            </p>
                            <p className="text-slate-400 mt-1">
                              {((data.value / (stats.ecoA + stats.ecoBP + stats.ecoBM + stats.borra)) * 100).toFixed(1)}% do total
                            </p>
                          </div>
                        );
                      }
                      return null;
                    }}
                  />
                  <Legend 
                    verticalAlign="bottom" 
                    align="center"
                    iconType="circle"
                    formatter={(value) => <span className="text-[10px] font-bold text-slate-500 uppercase">{value}</span>}
                  />
                </PieChart>
              </ResponsiveContainer>
            ) : (
              <div className="h-full flex flex-col items-center justify-center text-slate-400 gap-2">
                <PieChartIcon size={40} className="opacity-20" />
                <span className="text-xs font-medium">Sem dados de perdas no período</span>
              </div>
            )}
          </div>
        </div>

        {/* Grid Stats */}
        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
          <StatsCard 
            label="PRODUÇÃO" 
            value={stats.totalNet.toLocaleString('pt-BR')} 
            sublabel="Kg Líquidos" 
            borderColor="border-blue-500" 
          />
          <StatsCard 
            label="ECO A" 
            value={stats.ecoA === 0 ? '0' : stats.ecoA.toLocaleString('pt-BR')} 
            sublabel="Material Nobre" 
            borderColor="border-emerald-500" 
          />
          <StatsCard 
            label="ECO B (PROD)" 
            value={stats.ecoBP.toLocaleString('pt-BR')} 
            sublabel="Processo" 
            borderColor="border-amber-500" 
          />
          <StatsCard 
            label="ECO B (MANUT)" 
            value={stats.ecoBM.toLocaleString('pt-BR')} 
            sublabel="Manutenção" 
            borderColor="border-orange-500" 
          />
          <StatsCard 
            label="BORRA" 
            value={stats.borra.toLocaleString('pt-BR')} 
            sublabel="Perda Total" 
            borderColor="border-red-500" 
          />
          <StatsCard 
            label="PARADAS" 
            value={stats.paradas === 0 ? '0' : `${stats.paradas} min`} 
            sublabel="Minutos" 
            borderColor="border-slate-400" 
          />
        </div>

        {/* Ranking Table */}
        <div className="bg-white rounded-2xl shadow-sm overflow-hidden border border-slate-100">
          <table className="w-full text-left text-xs">
            <thead>
              <tr className="bg-slate-50 border-b border-slate-100">
                <th className="px-4 py-3 font-bold text-slate-500 uppercase">POS</th>
                <th className="px-4 py-3 font-bold text-slate-500 uppercase">OP</th>
                <th className="px-4 py-3 font-bold text-slate-500 uppercase text-right">PROD</th>
                <th className="px-4 py-3 font-bold text-slate-500 uppercase text-right">BORRA</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-slate-50">
              {filteredData.length > 0 ? (
                filteredData.sort((a,b) => b.netWeight - a.netWeight).map((entry, idx) => (
                  <tr key={entry.id} className="hover:bg-slate-50 transition-colors">
                    <td className="px-4 py-3 font-medium text-slate-400">#{idx + 1}</td>
                    <td className="px-4 py-3 font-bold text-slate-800">{entry.operator}</td>
                    <td className="px-4 py-3 font-bold text-slate-700 text-right">{entry.netWeight.toLocaleString('pt-BR')}</td>
                    <td className="px-4 py-3 font-medium text-slate-500 text-right">{entry.borraTotal.toLocaleString('pt-BR')}</td>
                  </tr>
                ))
              ) : (
                <tr>
                  <td colSpan={4} className="px-4 py-8 text-center text-slate-400 italic">
                    Nenhum registro encontrado
                  </td>
                </tr>
              )}
            </tbody>
          </table>
        </div>
      </main>

      <LaunchModal 
        isOpen={isModalOpen} 
        onClose={() => setIsModalOpen(false)} 
        onSave={handleSaveEntry} 
      />
    </div>
  );
};

export default App;
