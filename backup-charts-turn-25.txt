
import React, { useState, useMemo, useEffect } from 'react';
import { Menu, Plus, TrendingUp, BarChart2, Calendar, Pencil, Filter, RotateCcw, PieChart as PieChartIcon, Download, Clock, Zap, Trash2, Cpu } from 'lucide-react';
import { BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, Cell, PieChart, Pie, Legend } from 'recharts';
import { ProductionEntry } from './types';
import { INITIAL_DATA, GOAL_VALUE } from './constants';
import LaunchModal from './components/LaunchModal';
import StatsCard from './components/StatsCard';

const App: React.FC = () => {
  // Persistence Logic
  const [productionData, setProductionData] = useState<ProductionEntry[]>(() => {
    const saved = localStorage.getItem('manupackaging_data');
    return saved ? JSON.parse(saved) : INITIAL_DATA;
  });

  useEffect(() => {
    localStorage.setItem('manupackaging_data', JSON.stringify(productionData));
  }, [productionData]);

  const [isModalOpen, setIsModalOpen] = useState(false);
  
  // Filter States
  const [startDate, setStartDate] = useState<string>('');
  const [endDate, setEndDate] = useState<string>('');
  const [selectedMachine, setSelectedMachine] = useState<string>('Todas');

  // Filter Logic
  const filteredData = useMemo(() => {
    return productionData.filter(entry => {
      const entryDate = entry.date;
      const matchesStart = !startDate || entryDate >= startDate;
      const matchesEnd = !endDate || entryDate <= endDate;
      const matchesMachine = selectedMachine === 'Todas' || entry.machine === selectedMachine;
      return matchesStart && matchesEnd && matchesMachine;
    });
  }, [productionData, startDate, endDate, selectedMachine]);

  const stats = useMemo(() => {
    const s = filteredData.reduce((acc, curr) => ({
      totalNet: acc.totalNet + curr.netWeight,
      totalGross: acc.totalGross + curr.grossWeight,
      ecoA: acc.ecoA + curr.ecoA,
      ecoBP: acc.ecoBP + curr.ecoBP,
      ecoBM: acc.ecoBM + curr.ecoBM,
      borra: acc.borra + curr.borraTotal,
      paradas: acc.paradas + curr.manutencaoMin + curr.processoMin + curr.outrosMin,
      manut: acc.manut + curr.manutencaoMin,
      proc: acc.proc + curr.processoMin,
      outros: acc.outros + curr.outrosMin,
    }), { totalNet: 0, totalGross: 0, ecoA: 0, ecoBP: 0, ecoBM: 0, borra: 0, paradas: 0, manut: 0, proc: 0, outros: 0 });
    
    const qualityEfficiency = s.totalGross > 0 ? (s.totalNet / s.totalGross) * 100 : 0;
    
    return { ...s, qualityEfficiency };
  }, [filteredData]);

  const downtimeDistribution = useMemo(() => [
    { name: 'Manutenção', value: stats.manut, color: '#f97316' },
    { name: 'Processo', value: stats.proc, color: '#f59e0b' },
    { name: 'Outros', value: stats.outros, color: '#94a3b8' },
  ].filter(d => d.value > 0), [stats]);

  const shiftData = useMemo(() => {
    const shifts = filteredData.reduce((acc, curr) => {
      const shift = curr.shift;
      if (!acc[shift]) acc[shift] = 0;
      acc[shift] += curr.netWeight;
      return acc;
    }, {} as Record<string, number>);
    return Object.entries(shifts).map(([name, value]) => ({ name, value }));
  }, [filteredData]);

  const percentageGoal = Math.round((stats.totalNet / GOAL_VALUE) * 100);

  const chartData = useMemo(() => {
    return filteredData.map(entry => ({
      name: entry.operator,
      value: entry.netWeight,
    }));
  }, [filteredData]);

  const handleSaveEntry = (newEntry: Omit<ProductionEntry, 'id'>) => {
    const entryWithId: ProductionEntry = {
      ...newEntry,
      id: Math.random().toString(36).substr(2, 9),
    };
    setProductionData(prev => [...prev, entryWithId]);
  };

  const handleDeleteEntry = (id: string) => {
    if (confirm('Deseja realmente excluir este lançamento?')) {
      setProductionData(prev => prev.filter(e => e.id !== id));
    }
  };

  const clearFilters = () => {
    setStartDate('');
    setEndDate('');
    setSelectedMachine('Todas');
  };

  const exportToCSV = () => {
    const headers = ['Data', 'Operador', 'Maquina', 'Turno', 'Peso Bruto', 'Tara', 'Peso Liquido', 'Eco A', 'Eco B P', 'Eco B M', 'Borra', 'Parada Manut (min)', 'Motivo Manut', 'Parada Proc (min)', 'Motivo Proc', 'Parada Outros (min)', 'Motivo Outros'];
    const rows = filteredData.map(e => [
      e.date, e.operator, e.machine, e.shift, e.grossWeight, e.tara, e.netWeight, e.ecoA, e.ecoBP, e.ecoBM, e.borraTotal, 
      e.manutencaoMin, e.manutencaoMotivo || '', e.processoMin, e.processoMotivo || '', e.outrosMin, e.outrosMotivo || ''
    ]);
    const csvContent = [headers, ...rows].map(r => r.join(';')).join('\n');
    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const link = document.createElement('a');
    link.href = URL.createObjectURL(blob);
    link.setAttribute('download', `producao_manupackaging_${new Date().toISOString().split('T')[0]}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

  return (
    <div className="min-h-screen bg-slate-50 pb-20">
      <header className="bg-white px-6 py-4 flex items-center justify-between sticky top-0 z-40 border-b border-slate-100">
        <h1 className="text-xl font-bold text-slate-800 flex items-center gap-2">
          Controle de Produção 2026
        </h1>
        <div className="flex items-center gap-2">
           <button onClick={exportToCSV} className="text-slate-500 p-2 hover:bg-slate-100 rounded-lg flex items-center gap-2 text-sm font-medium">
             <Download size={20} />
             <span className="hidden sm:inline">Exportar</span>
           </button>
           <button className="text-slate-600 p-1 hover:bg-slate-100 rounded-lg">
             <Menu size={24} />
           </button>
        </div>
      </header>

      <main className="max-w-4xl mx-auto p-4 space-y-6">
        {/* Machine Filter Buttons */}
        <div className="flex bg-white p-1 rounded-xl shadow-sm border border-slate-200 w-fit">
          {['Todas', 'Cast 1', 'Cast 2'].map((m) => (
            <button
              key={m}
              onClick={() => setSelectedMachine(m)}
              className={`px-6 py-2 rounded-lg text-xs font-bold transition-all ${selectedMachine === m ? 'bg-blue-600 text-white shadow-md' : 'text-slate-500 hover:bg-slate-50'}`}
            >
              {m}
            </button>
          ))}
        </div>

        {/* Hero Banner */}
        <div className="relative overflow-hidden rounded-3xl bg-slate-800 p-8 text-white shadow-xl min-h-[160px] flex flex-col justify-between">
          <div className="relative z-10 flex items-center gap-3">
             <div className="w-10 h-10 bg-blue-500 rounded-lg flex items-center justify-center shadow-lg shadow-blue-500/20">
                <Zap size={20} className="text-white fill-white" />
             </div>
             <div>
               <span className="text-xl font-bold tracking-tight block">Manupackaging</span>
               <span className="text-[10px] text-slate-400 font-bold uppercase tracking-widest">Dashboard Industrial</span>
             </div>
          </div>
          <div className="flex justify-end relative z-10 mt-4">
            <button onClick={() => setIsModalOpen(true)} className="bg-yellow-400 hover:bg-yellow-500 text-slate-900 font-bold px-6 py-2 rounded-full flex items-center gap-2 transition-transform active:scale-95 shadow-lg shadow-yellow-900/20">
              <Plus size={20} />
              Lançar Produção
            </button>
          </div>
          <div className="absolute top-[-20%] right-[-10%] w-64 h-64 bg-blue-600 rounded-full opacity-10 blur-3xl pointer-events-none"></div>
        </div>

        {/* Filters */}
        <div className="bg-white rounded-2xl p-4 shadow-sm border border-slate-200 flex flex-col md:flex-row items-end md:items-center gap-4">
          <div className="flex-1 w-full space-y-1">
            <label className="text-[10px] font-bold text-slate-500 uppercase flex items-center gap-1"><Calendar size={12} /> Início</label>
            <input type="date" value={startDate} onChange={(e) => setStartDate(e.target.value)} className="w-full bg-slate-50 border border-slate-200 rounded-lg px-3 py-2 text-sm focus:outline-none" />
          </div>
          <div className="flex-1 w-full space-y-1">
            <label className="text-[10px] font-bold text-slate-500 uppercase flex items-center gap-1"><Calendar size={12} /> Fim</label>
            <input type="date" value={endDate} onChange={(e) => setEndDate(e.target.value)} className="w-full bg-slate-50 border border-slate-200 rounded-lg px-3 py-2 text-sm focus:outline-none" />
          </div>
          <button onClick={clearFilters} className="w-full md:w-auto flex items-center justify-center gap-2 px-6 py-2 bg-slate-100 hover:bg-slate-200 text-slate-600 rounded-lg text-sm font-bold transition-colors">
            <RotateCcw size={16} /> Reset
          </button>
        </div>

        {/* Progress Bar */}
        <div className="bg-white rounded-2xl p-6 shadow-sm border border-blue-500/10">
          <div className="flex items-center justify-between mb-4">
            <div>
               <span className="text-[10px] font-black text-blue-500 uppercase tracking-widest">Meta {selectedMachine}</span>
               <div className="flex items-baseline gap-2">
                  <span className="text-4xl font-black text-slate-800 tracking-tighter">{stats.totalNet.toLocaleString('pt-BR')}</span>
                  <span className="text-slate-400 font-medium text-sm">/ {GOAL_VALUE.toLocaleString('pt-BR')} kg</span>
               </div>
            </div>
            <div className="text-right">
              <span className="text-3xl font-black text-blue-600">{percentageGoal}%</span>
            </div>
          </div>
          <div className="w-full bg-slate-100 h-4 rounded-full overflow-hidden p-1">
            <div className="h-full bg-gradient-to-r from-blue-500 to-emerald-400 rounded-full transition-all duration-1000 ease-out" style={{ width: `${Math.min(percentageGoal, 100)}%` }}></div>
          </div>
        </div>

        {/* Main Charts Row */}
        <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
          {/* Efficiency per Operator */}
          <div className="bg-white rounded-2xl p-6 shadow-sm">
            <div className="flex items-center gap-2 mb-6">
               <TrendingUp size={18} className="text-blue-500" />
               <h3 className="font-bold text-slate-800">Eficiência Operador</h3>
            </div>
            <div className="h-[240px] w-full">
              {filteredData.length > 0 ? (
                <ResponsiveContainer width="100%" height="100%">
                  <BarChart data={chartData}>
                    <CartesianGrid vertical={false} stroke="#f1f5f9" />
                    <XAxis dataKey="name" axisLine={false} tickLine={false} tick={{ fontSize: 10, fill: '#94a3b8' }} />
                    <YAxis axisLine={false} tickLine={false} tick={{ fontSize: 10, fill: '#94a3b8' }} />
                    <Tooltip cursor={{ fill: '#f8fafc' }} />
                    <Bar dataKey="value" fill="#3b82f6" radius={[4, 4, 0, 0]} barSize={30} />
                  </BarChart>
                </ResponsiveContainer>
              ) : <NoDataView /> }
            </div>
          </div>

          {/* Downtime Analysis */}
          <div className="bg-white rounded-2xl p-6 shadow-sm">
            <div className="flex items-center gap-2 mb-6">
               <Clock size={18} className="text-orange-500" />
               <h3 className="font-bold text-slate-800">Análise de Paradas (min)</h3>
            </div>
            <div className="h-[240px] w-full">
              {downtimeDistribution.length > 0 ? (
                <ResponsiveContainer width="100%" height="100%">
                  <PieChart>
                    <Pie data={downtimeDistribution} cx="50%" cy="50%" innerRadius={60} outerRadius={85} paddingAngle={5} dataKey="value">
                      {downtimeDistribution.map((entry, index) => (
                        <Cell key={`cell-${index}`} fill={entry.color} />
                      ))}
                    </Pie>
                    <Tooltip />
                    <Legend iconType="circle" />
                  </PieChart>
                </ResponsiveContainer>
              ) : <NoDataView /> }
            </div>
          </div>
        </div>

        {/* Stats Grid */}
        <div className="grid grid-cols-2 sm:grid-cols-3 gap-4">
          <StatsCard label="PRODUÇÃO LÍQUIDA" value={stats.totalNet.toLocaleString('pt-BR')} sublabel="Kg" borderColor="border-blue-500" />
          <StatsCard label="BORRA (PERDA)" value={stats.borra.toLocaleString('pt-BR')} sublabel="Kg" borderColor="border-red-500" />
          <StatsCard label="QUALIDADE" value={`${stats.qualityEfficiency.toFixed(1)}%`} sublabel="Eficiência" borderColor="border-emerald-500" />
          <StatsCard label="PARADAS" value={stats.paradas} sublabel="Minutos" borderColor="border-orange-500" />
          <StatsCard label="MÁQUINA ATUAL" value={selectedMachine} sublabel="Filtro Ativo" borderColor="border-slate-300" />
          <StatsCard label="ENTRADAS" value={filteredData.length} sublabel="Registros" borderColor="border-slate-300" />
        </div>

        {/* History Table */}
        <div className="bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden">
          <div className="p-4 border-b border-slate-100 bg-slate-50 flex items-center justify-between">
            <h3 className="font-bold text-slate-800 flex items-center gap-2 text-sm uppercase tracking-wider">
               <BarChart2 size={16} className="text-blue-500" /> Histórico de Lançamentos
            </h3>
            <span className="text-[10px] font-bold text-slate-400">{filteredData.length} registros</span>
          </div>
          <div className="overflow-x-auto">
            <table className="w-full text-left text-sm">
              <thead className="bg-slate-50 text-slate-400 text-[10px] font-bold uppercase">
                <tr>
                  <th className="px-4 py-3">Data</th>
                  <th className="px-4 py-3">Operador</th>
                  <th className="px-4 py-3">Máquina</th>
                  <th className="px-4 py-3 text-right">Líquido (kg)</th>
                  <th className="px-4 py-3 text-right">Paradas</th>
                  <th className="px-4 py-3 text-center">Ações</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {filteredData.slice().reverse().map((entry) => (
                  <tr key={entry.id} className="hover:bg-slate-50 transition-colors group">
                    <td className="px-4 py-3 font-medium text-slate-600 whitespace-nowrap">{entry.date}</td>
                    <td className="px-4 py-3 text-slate-800 font-bold">{entry.operator}</td>
                    <td className="px-4 py-3">
                      <span className="bg-slate-100 text-slate-600 text-[10px] px-2 py-0.5 rounded-full font-bold">
                        {entry.machine}
                      </span>
                    </td>
                    <td className="px-4 py-3 text-right font-black text-blue-600">{entry.netWeight.toLocaleString('pt-BR')}</td>
                    <td className="px-4 py-3 text-right text-orange-600 font-bold">
                      {entry.manutencaoMin + entry.processoMin + entry.outrosMin}m
                    </td>
                    <td className="px-4 py-3 text-center">
                      <button 
                        onClick={() => handleDeleteEntry(entry.id)}
                        className="text-slate-300 hover:text-red-500 p-1 rounded-lg transition-colors"
                      >
                        <Trash2 size={16} />
                      </button>
                    </td>
                  </tr>
                ))}
                {filteredData.length === 0 && (
                  <tr>
                    <td colSpan={6} className="px-4 py-10 text-center text-slate-400 italic">Nenhum registro encontrado para os filtros selecionados.</td>
                  </tr>
                )}
              </tbody>
            </table>
          </div>
        </div>
      </main>

      <LaunchModal isOpen={isModalOpen} onClose={() => setIsModalOpen(false)} onSave={handleSaveEntry} />
    </div>
  );
};

const NoDataView = () => (
  <div className="h-full flex flex-col items-center justify-center text-slate-400 gap-2 opacity-30">
    <BarChart2 size={40} />
    <span className="text-xs font-bold uppercase tracking-widest">Sem dados</span>
  </div>
);

export default App;
