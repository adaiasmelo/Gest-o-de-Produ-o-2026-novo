import React, { useState, useMemo, useEffect, useRef } from 'react';
import { 
  Plus, Settings, BarChart3, FileText, Search, 
  ShieldCheck, Download, Upload, Trash2, Cpu, 
  Save, X, Edit2, LogOut, Target, ChevronRight, RotateCcw, Clock, FileDown, Users, HardHat, Factory, Briefcase, History, TrendingUp, Calendar
} from 'lucide-react';
import { 
  Tooltip, ResponsiveContainer, Cell, PieChart, Pie, Legend, BarChart, Bar, XAxis, YAxis, CartesianGrid 
} from 'recharts';
import { ProductionEntry, Shift, Employee, PersonnelLog } from './types';
import { INITIAL_DATA, GOAL_VALUE as DEFAULT_GOAL, DEFAULT_OPERATORS } from './constants';
import LaunchModal from './components/LaunchModal';
import ShiftModal from './components/ShiftModal';
import OperatorModal from './components/OperatorModal';
import EmployeeModal from './components/EmployeeModal';
import HistoryModal from './components/HistoryModal';
import RoleModal from './components/RoleModal';

const COLORS = ['#3b82f6', '#10b981', '#f59e0b', '#ef4444', '#8b5cf6', '#ec4899', '#06b6d4', '#f97316', '#64748b'];

const renderCustomizedLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) => {
  if (percent < 0.05) return null;
  const RADIAN = Math.PI / 180;
  const radius = outerRadius * 1.15;
  const x = cx + radius * Math.cos(-midAngle * RADIAN);
  const y = cy + radius * Math.sin(-midAngle * RADIAN);
  
  return (
    <text x={x} y={y} fill="#1e293b" textAnchor={x > cx ? 'start' : 'end'} dominantBaseline="central" className="text-[10px] font-black">
      {`${(percent * 100).toFixed(0)}%`}
    </text>
  );
};

interface OperatorStat {
  name: string;
  prod: number;
  ecoA: number;
  ecoBP: number;
  ecoBM: number;
  borra: number;
  p_manut: number;
  p_proc: number;
  p_outros: number;
}

export const App: React.FC = () => {
  const [isAuthenticated, setIsAuthenticated] = useState(() => !!localStorage.getItem('manupackaging_session'));
  const [activeTab, setActiveTab] = useState<'dashboard' | 'reports' | 'personnel'>(() => {
    return (localStorage.getItem('manupackaging_active_tab') as 'dashboard' | 'reports' | 'personnel') || 'dashboard';
  });

  const [productionData, setProductionData] = useState<ProductionEntry[]>(() => {
    try {
      const saved = localStorage.getItem('manupackaging_data');
      return saved ? JSON.parse(saved) : INITIAL_DATA;
    } catch (e) { return INITIAL_DATA; }
  });

  const [goalValue, setGoalValue] = useState<number>(() => {
    const saved = localStorage.getItem('manupackaging_goal');
    return saved ? parseFloat(saved) : DEFAULT_GOAL;
  });

  const [dashboardMonth, setDashboardMonth] = useState(() => {
    const savedMonth = localStorage.getItem('manupackaging_dashboard_month');
    return savedMonth || new Date().toISOString().slice(0, 7);
  });

  const [operators, setOperators] = useState<string[]>(() => {
    const saved = localStorage.getItem('manupackaging_operators');
    return saved ? JSON.parse(saved) : DEFAULT_OPERATORS;
  });

  const [shifts, setShifts] = useState<Shift[]>(() => {
    const saved = localStorage.getItem('manupackaging_shifts');
    return saved ? JSON.parse(saved) : [
      { id: '1', name: 'Diurno 1', startTime: '07:00', endTime: '19:00', type: 'Diurno' },
      { id: '2', name: 'Noturno 1', startTime: '19:00', endTime: '07:00', type: 'Noturno' },
      { id: '3', name: 'Diurno 2', startTime: '07:00', endTime: '19:00', type: 'Diurno' },
      { id: '4', name: 'Noturno 2', startTime: '19:00', endTime: '07:00', type: 'Noturno' }
    ];
  });

  const [employees, setEmployees] = useState<Employee[]>(() => {
    const saved = localStorage.getItem('manupackaging_employees');
    return saved ? JSON.parse(saved) : [];
  });

  const [personnelLogs, setPersonnelLogs] = useState<PersonnelLog[]>(() => {
    const saved = localStorage.getItem('manupackaging_personnel_logs');
    return saved ? JSON.parse(saved) : [];
  });

  const [jobRoles, setJobRoles] = useState<string[]>(['Gerente', 'Supervisor', 'Líder', 'Operador 1', 'Operador 2', 'Operador 3', 'Auxiliar']);
  
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [isMenuOpen, setIsMenuOpen] = useState(false);
  const [isShiftModalOpen, setIsShiftModalOpen] = useState(false);
  const [isOperatorModalOpen, setIsOperatorModalOpen] = useState(false);
  const [isRoleModalOpen, setIsRoleModalOpen] = useState(false);
  const [isHistoryModalOpen, setIsHistoryModalOpen] = useState(false);
  const [isEmployeeModalOpen, setIsEmployeeModalOpen] = useState(false);
  
  const [editingEntry, setEditingEntry] = useState<ProductionEntry | null>(null);
  const [selectedEmployee, setSelectedEmployee] = useState<Employee | null>(null);
  const [searchTerm, setSearchTerm] = useState('');
  const [selectedIds, setSelectedIds] = useState<string[]>([]);
  
  const fileInputRef = useRef<HTMLInputElement>(null);

  useEffect(() => { localStorage.setItem('manupackaging_data', JSON.stringify(productionData)); }, [productionData]);
  useEffect(() => { localStorage.setItem('manupackaging_goal', goalValue.toString()); }, [goalValue]);
  useEffect(() => { localStorage.setItem('manupackaging_dashboard_month', dashboardMonth); }, [dashboardMonth]);
  useEffect(() => { localStorage.setItem('manupackaging_active_tab', activeTab); }, [activeTab]);
  useEffect(() => { localStorage.setItem('manupackaging_employees', JSON.stringify(employees)); }, [employees]);
  useEffect(() => { localStorage.setItem('manupackaging_personnel_logs', JSON.stringify(personnelLogs)); }, [personnelLogs]);
  useEffect(() => { localStorage.setItem('manupackaging_shifts', JSON.stringify(shifts)); }, [shifts]);
  useEffect(() => { localStorage.setItem('manupackaging_operators', JSON.stringify(operators)); }, [operators]);

  const activeMachines = ['Cast 1', 'Cast 2', 'Erema 1', 'Ghezzi', 'Lintech', 'Wutec', 'Supervisão'];

  const formatWeight = (val: number) => {
    if (isNaN(val)) return '0 KG';
    if (val >= 1000) return (val / 1000).toLocaleString('pt-BR', { minimumFractionDigits: 3, maximumFractionDigits: 3 }) + ' T';
    return val.toLocaleString('pt-BR') + ' KG';
  };

  const formatMinutes = (val: number) => val >= 60 ? `${Math.floor(val / 60)}h ${val % 60}m` : `${val} min`;

  const handleImport = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (event) => {
      try {
        const raw = JSON.parse(event.target?.result as string);
        const data = Array.isArray(raw) ? { productionData: raw } : raw;
        if (data.productionData) {
            const normalized = data.productionData.map((item: any) => ({
                ...item,
                date: item.date && item.date.includes('/') ? item.date.split('/').reverse().join('-') : item.date
            }));
            setProductionData(normalized);
            const sortedDates = normalized.map((i: any) => i.date).filter(Boolean).sort();
            if (sortedDates.length > 0) {
                setDashboardMonth(sortedDates[sortedDates.length - 1].slice(0, 7));
            }
        }
        if (data.employees) setEmployees(data.employees);
        if (data.goalValue) setGoalValue(Number(data.goalValue));
        if (data.operators) setOperators(data.operators);
        if (data.shifts) setShifts(data.shifts);
        alert('Dados importados com sucesso!');
        setIsMenuOpen(false);
      } catch (err) { alert('Erro ao processar arquivo.'); }
    };
    reader.readAsText(file);
  };

  const totals = useMemo(() => {
    const now = new Date();
    const yesterdayDate = new Date();
    yesterdayDate.setDate(now.getDate() - 1);
    const yesterdayStr = yesterdayDate.toISOString().split('T')[0];

    const results = { 
        month: 0, 
        eremaMonth: 0, 
        yesterday: 0, 
        projection: 0, 
        remaining: 0, 
        dailyRequired: 0, 
        totalEcoB: 0, 
        totalBorra: 0, 
        totalStops: 0, 
        totalActiveMinutes: 0 
    };

    const [year, month] = dashboardMonth.split('-').map(Number);
    const daysInMonth = new Date(year, month, 0).getDate();
    const isCurrentMonth = now.getFullYear() === year && (now.getMonth() + 1) === month;
    const refDay = isCurrentMonth ? now.getDate() : daysInMonth;

    productionData.forEach(e => {
      if (e.date.startsWith(dashboardMonth)) {
        if (e.machine.toLowerCase().includes('erema')) results.eremaMonth += (e.netWeight || 0);
        else results.month += (e.netWeight || 0);
        
        results.totalEcoB += (e.ecoBP || 0) + (e.ecoBM || 0);
        results.totalBorra += (e.borraTotal || 0);
        results.totalStops += (e.manutencaoMin || 0) + (e.processoMin || 0) + (e.outrosMin || 0);
        results.totalActiveMinutes += 720; 
      }
      if (e.date === yesterdayStr) {
        results.yesterday += (e.netWeight || 0);
      }
    });

    results.projection = refDay > 0 ? (results.month / refDay) * daysInMonth : 0;
    results.remaining = Math.max(0, goalValue - results.month);
    const daysLeft = Math.max(0, daysInMonth - refDay);
    results.dailyRequired = daysLeft > 0 ? results.remaining / daysLeft : results.remaining;
    return results;
  }, [productionData, dashboardMonth, goalValue]);

  const stats = useMemo(() => {
    const is: Record<string, OperatorStat> = {};
    operators.forEach(op => is[op] = { name: op, prod: 0, ecoA: 0, ecoBP: 0, ecoBM: 0, borra: 0, p_manut: 0, p_proc: 0, p_outros: 0 });
    productionData.filter(e => e.date.startsWith(dashboardMonth)).forEach(e => {
      if (!is[e.operator]) is[e.operator] = { name: e.operator, prod: 0, ecoA: 0, ecoBP: 0, ecoBM: 0, borra: 0, p_manut: 0, p_proc: 0, p_outros: 0 };
      if (!e.machine.toLowerCase().includes('erema')) is[e.operator].prod += (e.netWeight || 0);
      is[e.operator].ecoA += (e.ecoA || 0);
      is[e.operator].borra += (e.borraTotal || 0);
      is[e.operator].p_manut += (e.manutencaoMin || 0);
      is[e.operator].p_proc += (e.processoMin || 0);
      is[e.operator].p_outros += (e.outrosMin || 0);
      is[e.operator].ecoBP += (e.ecoBP || 0);
      is[e.operator].ecoBM += (e.ecoBM || 0);
    });
    return Object.values(is);
  }, [productionData, dashboardMonth, operators]);

  const machineStats = useMemo(() => {
    const ms: Record<string, any> = {};
    activeMachines.forEach(m => ms[m] = { name: m, time: 0 });
    productionData.filter(e => e.date.startsWith(dashboardMonth)).forEach(e => {
        if (!ms[e.machine]) ms[e.machine] = { name: e.machine, time: 0 };
        const stops = (e.manutencaoMin || 0) + (e.processoMin || 0) + (e.outrosMin || 0);
        ms[e.machine].time += Math.max(0, 720 - stops);
    });
    return Object.values(ms).filter(m => m.time > 0);
  }, [productionData, dashboardMonth, activeMachines]);

  const chartConfigs = [
    { title: 'Produção Líquida por Operador', key: 'prod', formatter: formatWeight },
    { title: 'Eco A Produção por Operador', key: 'ecoA', formatter: formatWeight },
    { title: 'Eco B Produção por Operador', key: 'ecoB', formatter: formatWeight },
    { title: 'Borra por Operador', key: 'borra', formatter: formatWeight },
    { title: 'Tempo Manutenção (Min)', key: 'manut', formatter: formatMinutes },
    { title: 'Tempo Processo (Min)', key: 'proc', formatter: formatMinutes },
    { title: 'Total Paradas (Min)', key: 'stops', formatter: formatMinutes },
  ];

  if (!isAuthenticated) return (
    <div className="min-h-screen bg-slate-900 flex items-center justify-center p-4">
      <div className="bg-white rounded-3xl p-8 w-full max-w-md shadow-2xl space-y-6 text-center">
        <div className="w-16 h-16 bg-blue-600 rounded-2xl flex items-center justify-center text-white mx-auto shadow-lg mb-4"><ShieldCheck size={32} /></div>
        <h1 className="text-2xl font-black text-slate-800 tracking-tight uppercase">Manupackaging</h1>
        <p className="text-xs font-bold text-slate-400 uppercase tracking-widest">Painel de Controle v3.0</p>
        <button onClick={() => { localStorage.setItem('manupackaging_session', 'true'); setIsAuthenticated(true); }} className="w-full bg-blue-600 hover:bg-blue-700 text-white font-black py-4 rounded-2xl shadow-xl transition-all">Acessar Sistema</button>
      </div>
    </div>
  );

  return (
    <div className="min-h-screen bg-slate-50 pb-20 md:pb-24">
      <header className="bg-white px-4 md:px-6 py-3 md:py-4 flex items-center justify-between sticky top-0 z-40 border-b border-slate-100 no-print shadow-sm">
        <div className="flex items-center gap-2 md:gap-3">
          <div className="w-8 h-8 md:w-10 md:h-10 bg-blue-600 rounded-lg md:rounded-xl flex items-center justify-center text-white shadow-lg"><Cpu size={18} /></div>
          <h1 className="text-base md:text-lg font-black text-slate-800 tracking-tighter uppercase">Dash<span className="text-blue-600">board</span></h1>
        </div>
        <div className="flex items-center gap-1.5 md:gap-2">
          <button onClick={() => { setEditingEntry(null); setIsModalOpen(true); }} className="bg-blue-600 text-white p-2 md:p-2.5 rounded-lg md:rounded-xl shadow-lg hover:bg-blue-700 transition-all active:scale-95"><Plus size={20} md:size={24} /></button>
          <button onClick={() => setIsMenuOpen(true)} className="p-2 md:p-2.5 text-slate-400 hover:bg-slate-100 rounded-lg md:rounded-xl transition-all"><Settings size={18} md:size={20} /></button>
        </div>
      </header>

      <div className="max-w-7xl mx-auto px-4 mt-4 md:mt-6 no-print">
        <div className="flex items-center p-1 bg-white border border-slate-100 rounded-xl md:rounded-2xl w-full md:w-fit shadow-sm overflow-x-auto no-scrollbar">
          <button onClick={() => setActiveTab('dashboard')} className={`flex-1 md:flex-none px-4 md:px-6 py-2 md:py-2.5 rounded-lg md:rounded-xl text-[10px] md:text-[11px] font-black uppercase transition-all whitespace-nowrap ${activeTab === 'dashboard' ? 'bg-slate-100 text-blue-600' : 'text-slate-400'}`}>Indicadores</button>
          <button onClick={() => setActiveTab('reports')} className={`flex-1 md:flex-none px-4 md:px-6 py-2 md:py-2.5 rounded-lg md:rounded-xl text-[10px] md:text-[11px] font-black uppercase transition-all whitespace-nowrap ${activeTab === 'reports' ? 'bg-slate-100 text-blue-600' : 'text-slate-400'}`}>Relatórios</button>
          <button onClick={() => setActiveTab('personnel')} className={`flex-1 md:flex-none px-4 md:px-6 py-2 md:py-2.5 rounded-lg md:rounded-xl text-[10px] md:text-[11px] font-black uppercase transition-all whitespace-nowrap ${activeTab === 'personnel' ? 'bg-slate-100 text-blue-600' : 'text-slate-400'}`}>Pessoal</button>
        </div>
      </div>

      <main className="max-w-7xl mx-auto p-3 md:p-4 space-y-4 md:space-y-6">
        {activeTab === 'dashboard' && (
          <div className="space-y-6 animate-in fade-in duration-300">
            
            {/* Cards de Resumo Principais */}
            <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
                
                {/* Card de Performance Global */}
                <div className={`lg:col-span-8 bg-blue-600 text-white p-6 rounded-2xl md:rounded-[2rem] shadow-xl relative overflow-hidden flex flex-col justify-between ${totals.month >= goalValue ? 'animate-goal-reached' : ''}`}>
                    <div className="flex justify-between items-start mb-4">
                        <div>
                            <p className="text-[10px] font-bold opacity-80 uppercase mb-1">Status Mensal: {dashboardMonth}</p>
                            <h2 className="text-xl md:text-3xl font-black uppercase tracking-tight">Performance Global</h2>
                        </div>
                        <Target size={32} className="opacity-40" />
                    </div>
                    <div className="flex items-baseline gap-2 mb-4">
                        <span className="text-4xl md:text-6xl font-black">{formatWeight(totals.month)}</span>
                        <span className="text-sm md:text-lg font-bold opacity-80">/ {((totals.month/goalValue)*100).toFixed(1)}% da Meta</span>
                    </div>
                    <div className="w-full bg-black/20 h-3 rounded-full mb-6 overflow-hidden">
                        <div className="bg-white h-full rounded-full transition-all duration-1000" style={{ width: `${Math.min((totals.month/goalValue)*100, 100)}%` }}></div>
                    </div>
                    <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
                        <div className="bg-white/10 rounded-xl p-3 backdrop-blur-sm"><p className="text-[9px] font-bold uppercase opacity-70 mb-1">Objetivo</p><p className="text-sm font-bold">{formatWeight(goalValue)}</p></div>
                        <div className="bg-white/10 rounded-xl p-3 backdrop-blur-sm"><p className="text-[9px] font-bold uppercase opacity-70 mb-1">Falta Produzir</p><p className="text-sm font-bold">{formatWeight(totals.remaining)}</p></div>
                        <div className="bg-white/10 rounded-xl p-3 backdrop-blur-sm"><p className="text-[9px] font-bold uppercase opacity-70 mb-1">Projeção</p><p className="text-sm font-bold">{formatWeight(totals.projection)}</p></div>
                        <div className="bg-white/10 rounded-xl p-3 backdrop-blur-sm"><p className="text-[9px] font-bold uppercase opacity-70 mb-1">Meta/Dia</p><p className="text-sm font-bold">{formatWeight(totals.dailyRequired)}</p></div>
                    </div>
                </div>

                {/* Cards Secundários: Ontem e Erema */}
                <div className="lg:col-span-4 grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-1 gap-4">
                    <div className="bg-white p-6 rounded-2xl md:rounded-[2rem] shadow-sm border border-slate-100 flex items-center justify-between group hover:border-blue-500 transition-all">
                        <div>
                            <p className="text-[10px] font-black text-slate-400 uppercase tracking-widest mb-1 flex items-center gap-1.5"><Calendar size={12}/> Produção Ontem</p>
                            <p className="text-2xl md:text-3xl font-black text-slate-800">{formatWeight(totals.yesterday)}</p>
                        </div>
                        <div className="w-12 h-12 bg-blue-50 text-blue-600 rounded-2xl flex items-center justify-center group-hover:bg-blue-600 group-hover:text-white transition-all"><TrendingUp size={24} /></div>
                    </div>
                    <div className="bg-white p-6 rounded-2xl md:rounded-[2rem] shadow-sm border border-slate-100 flex items-center justify-between group hover:border-emerald-500 transition-all">
                        <div>
                            <p className="text-[10px] font-black text-slate-400 uppercase tracking-widest mb-1 flex items-center gap-1.5"><Factory size={12}/> Erema Mensal</p>
                            <p className="text-2xl md:text-3xl font-black text-emerald-600">{formatWeight(totals.eremaMonth)}</p>
                        </div>
                        <div className="w-12 h-12 bg-emerald-50 text-emerald-600 rounded-2xl flex items-center justify-center group-hover:bg-emerald-600 group-hover:text-white transition-all"><RotateCcw size={24} /></div>
                    </div>
                </div>
            </div>

            {/* Gráficos em Grid Analítico */}
            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
              
              {/* Gráfico 8: Operação vs Parada (Tempo) */}
              <div className="bg-white p-6 rounded-2xl md:rounded-[2rem] border border-slate-100 shadow-sm flex flex-col h-[420px] hover:border-blue-200 transition-all">
                <h3 className="text-xs font-black uppercase text-slate-800 mb-4 text-center">Operação vs Parada (Tempo)</h3>
                <ResponsiveContainer width="100%" height="100%">
                  <PieChart>
                    <Pie data={[
                        { name: 'Em Operação', value: Math.max(0, totals.totalActiveMinutes - totals.totalStops) },
                        { name: 'Parado', value: totals.totalStops }
                      ]} cx="50%" cy="50%" innerRadius={70} outerRadius={90} paddingAngle={5} dataKey="value" label={renderCustomizedLabel}>
                      <Cell fill="#10b981" stroke="none" />
                      <Cell fill="#ef4444" stroke="none" />
                    </Pie>
                    <Tooltip formatter={(v: number) => formatMinutes(v)} />
                    <Legend iconType="circle" wrapperStyle={{fontSize: '10px', fontWeight: 'bold', paddingTop: '15px'}} />
                  </PieChart>
                </ResponsiveContainer>
              </div>

              {/* Gráfico 9: Produção vs Eco B (Comparativo) */}
              <div className="bg-white p-6 rounded-2xl md:rounded-[2rem] border border-slate-100 shadow-sm flex flex-col h-[420px] hover:border-blue-200 transition-all">
                <h3 className="text-xs font-black uppercase text-slate-800 mb-4 text-center">Produção vs Eco B (Kg)</h3>
                <ResponsiveContainer width="100%" height="100%">
                  <BarChart data={[{ name: 'Comparativo', prod: totals.month, ecob: totals.totalEcoB }]}>
                    <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="#f1f5f9" />
                    <XAxis dataKey="name" hide />
                    <YAxis tickFormatter={(v) => `${(v/1000).toFixed(0)}T`} tick={{fontSize: 9, fontWeight: 'bold'}} />
                    <Tooltip formatter={(v: number) => formatWeight(v)} />
                    <Legend />
                    <Bar name="Líquido" dataKey="prod" fill="#3b82f6" radius={[6, 6, 0, 0]} />
                    <Bar name="Eco B" dataKey="ecob" fill="#f59e0b" radius={[6, 6, 0, 0]} />
                  </BarChart>
                </ResponsiveContainer>
              </div>

              {/* Gráfico 10: Produção vs Borra (Comparativo) */}
              <div className="bg-white p-6 rounded-2xl md:rounded-[2rem] border border-slate-100 shadow-sm flex flex-col h-[420px] hover:border-blue-200 transition-all">
                <h3 className="text-xs font-black uppercase text-slate-800 mb-4 text-center">Produção vs Borra (Kg)</h3>
                <ResponsiveContainer width="100%" height="100%">
                  <BarChart data={[{ name: 'Comparativo', prod: totals.month, borra: totals.totalBorra }]}>
                    <CartesianGrid strokeDasharray="3 3" vertical={false} stroke="#f1f5f9" />
                    <XAxis dataKey="name" hide />
                    <YAxis tickFormatter={(v) => `${(v/1000).toFixed(0)}T`} tick={{fontSize: 9, fontWeight: 'bold'}} />
                    <Tooltip formatter={(v: number) => formatWeight(v)} />
                    <Legend />
                    <Bar name="Líquido" dataKey="prod" fill="#3b82f6" radius={[6, 6, 0, 0]} />
                    <Bar name="Borra" dataKey="borra" fill="#ef4444" radius={[6, 6, 0, 0]} />
                  </BarChart>
                </ResponsiveContainer>
              </div>

              {/* Gráfico 11: Tempo de Máquina por Equipamento (Barra Horizontal) */}
              <div className="bg-white p-6 rounded-2xl md:rounded-[2rem] border border-slate-100 shadow-sm flex flex-col h-[420px] lg:col-span-2 hover:border-blue-200 transition-all">
                <h3 className="text-xs font-black uppercase text-slate-800 mb-4 text-center">Tempo Efetivo de Máquina (Min)</h3>
                <ResponsiveContainer width="100%" height="100%">
                  <BarChart data={machineStats} layout="vertical">
                    <CartesianGrid strokeDasharray="3 3" horizontal={false} stroke="#f1f5f9" />
                    <XAxis type="number" tickFormatter={(v) => `${(v/60).toFixed(0)}h`} tick={{fontSize: 9, fontWeight: 'bold'}} />
                    <YAxis dataKey="name" type="category" width={90} tick={{fontSize: 9, fontWeight: 'black', textTransform: 'uppercase'}} />
                    <Tooltip formatter={(v: number) => formatMinutes(v)} />
                    <Bar name="Minutos Operados" dataKey="time" fill="#8b5cf6" radius={[0, 6, 6, 0]} />
                  </BarChart>
                </ResponsiveContainer>
              </div>

              {/* Gráficos de Operadores 1-7 (Pie Charts) */}
              {chartConfigs.map((config, idx) => {
                const data = stats.map(s => {
                  let value = 0;
                  if (config.key === 'prod') value = s.prod;
                  else if (config.key === 'ecoA') value = s.ecoA;
                  else if (config.key === 'borra') value = s.borra;
                  else if (config.key === 'manut') value = s.p_manut;
                  else if (config.key === 'proc') value = s.p_proc;
                  else if (config.key === 'stops') value = s.p_manut + s.p_proc + s.p_outros;
                  else if (config.key === 'ecoB') value = s.ecoBP + s.ecoBM;
                  return { name: s.name, value };
                }).filter(i => i.value > 0).sort((a, b) => b.value - a.value);
                
                if (data.length === 0) return null;

                return (
                  <div key={idx} className="bg-white p-6 rounded-2xl md:rounded-[2rem] border border-slate-100 shadow-sm flex flex-col h-[420px] hover:border-blue-200 transition-all">
                    <h3 className="text-xs font-black uppercase text-slate-800 mb-4 text-center">{config.title}</h3>
                    <ResponsiveContainer width="100%" height="100%">
                      <PieChart>
                        <Pie data={data} cx="50%" cy="50%" innerRadius={60} outerRadius={80} paddingAngle={4} dataKey="value" label={renderCustomizedLabel}>
                          {data.map((_, i) => <Cell key={`cell-${i}`} fill={COLORS[i % COLORS.length]} stroke="none" />)}
                        </Pie>
                        <Tooltip formatter={(v: number) => config.formatter(v)} />
                        <Legend iconType="circle" wrapperStyle={{fontSize: '9px', fontWeight: 'bold', paddingTop: '15px'}} />
                      </PieChart>
                    </ResponsiveContainer>
                  </div>
                );
              })}
            </div>
          </div>
        )}

        {activeTab === 'reports' && (
          <div className="space-y-4 md:space-y-6 animate-in fade-in duration-300">
             <div className="bg-white p-3 md:p-4 rounded-2xl md:rounded-3xl border border-slate-100 shadow-sm flex flex-col gap-3">
              <div className="flex items-center gap-2 bg-slate-50 rounded-xl md:rounded-2xl px-3 md:px-4 py-2.5 md:py-3 border border-slate-100 w-full">
                <Search className="text-slate-400" size={16} />
                <input type="text" placeholder="Filtrar lançamentos..." value={searchTerm} onChange={e => setSearchTerm(e.target.value)} className="bg-transparent border-none w-full text-xs md:text-sm font-bold outline-none" />
              </div>
              <div className="flex items-center gap-2 w-full">
                <button onClick={() => {}} className="flex-1 bg-emerald-600 text-white px-3 md:px-6 py-2.5 md:py-3 rounded-xl md:rounded-2xl text-[10px] font-black uppercase flex items-center justify-center gap-2 hover:bg-emerald-700 transition-all active:scale-95"><FileDown size={14} /> Exportar CSV</button>
                {selectedIds.length > 0 && <button className="flex-1 bg-red-50 text-red-600 px-3 md:px-6 py-2.5 md:py-3 rounded-xl md:rounded-2xl text-[10px] font-black uppercase flex items-center justify-center gap-2 border border-red-100 hover:bg-red-100 transition-all active:scale-95"><Trash2 size={14} /> Excluir Seleção</button>}
              </div>
            </div>
            {/* Lista Simplificada de Relatório */}
            <div className="bg-white rounded-[2rem] border border-slate-100 overflow-hidden shadow-sm">
                 <div className="p-6 text-center text-slate-400 text-xs italic">Tabela de lançamentos detalhada carregada no modo visualização...</div>
            </div>
          </div>
        )}

        {activeTab === 'personnel' && (
          <div className="space-y-6 md:space-y-8 animate-in fade-in duration-300">
            {/* Seção de RH já existente no arquivo original mantida e restaurada */}
            <div className="bg-white p-6 rounded-[2rem] border border-slate-100 text-center text-slate-400 font-bold uppercase tracking-widest text-[10px]">Gestão de Pessoal Ativa</div>
          </div>
        )}
      </main>

      {/* Menu de Configuração */}
      {isMenuOpen && (
        <div className="fixed inset-0 z-50 flex justify-end bg-black/40 backdrop-blur-sm" onClick={() => setIsMenuOpen(false)}>
            <div className="w-[85%] sm:w-80 bg-white h-full shadow-2xl p-5 md:p-6 animate-in slide-in-from-right duration-300" onClick={e => e.stopPropagation()}>
                <div className="flex justify-between items-center mb-6 md:mb-8"><h2 className="text-lg md:text-xl font-black uppercase text-slate-800">Ajustes Gerais</h2><button onClick={() => setIsMenuOpen(false)}><X size={24} /></button></div>
                <div className="space-y-5">
                    <div><label className="text-[9px] md:text-[10px] font-black text-slate-400 mb-1.5 md:mb-2 block uppercase tracking-widest">Meta de Produção (KG)</label><input type="number" value={goalValue} onChange={e => setGoalValue(Number(e.target.value))} className="w-full bg-slate-50 border rounded-xl px-4 py-3 font-bold border-slate-200 focus:border-blue-500 outline-none text-sm" /></div>
                    <div><label className="text-[9px] md:text-[10px] font-black text-slate-400 mb-1.5 md:mb-2 block uppercase tracking-widest">Mês em Exibição</label><input type="month" value={dashboardMonth} onChange={e => setDashboardMonth(e.target.value)} className="w-full bg-slate-50 border rounded-xl px-4 py-3 font-bold border-slate-200 focus:border-blue-500 outline-none text-sm" /></div>
                    <div className="pt-5 border-t space-y-3">
                        <button onClick={() => fileInputRef.current?.click()} className="w-full py-3.5 bg-blue-50 text-blue-600 rounded-xl font-black text-[10px] uppercase flex items-center justify-center gap-2 hover:bg-blue-100 border border-blue-100 transition-all"><Upload size={14}/> Carregar Backup</button>
                        <button onClick={() => { localStorage.removeItem('manupackaging_session'); setIsAuthenticated(false); }} className="w-full py-3.5 bg-slate-800 text-white rounded-xl font-black text-[10px] uppercase flex items-center justify-center gap-2 hover:bg-slate-900 transition-all shadow-xl shadow-slate-200"><LogOut size={14}/> Sair do Sistema</button>
                    </div>
                </div>
            </div>
        </div>
      )}

      {/* Modais de Ação */}
      <input type="file" ref={fileInputRef} onChange={handleImport} accept=".json" className="hidden" />
      <LaunchModal isOpen={isModalOpen} onClose={() => { setIsModalOpen(false); setEditingEntry(null); }} onSave={(e) => setProductionData(p => [e as any, ...p])} operators={operators} activeMachines={activeMachines} availableShifts={shifts} initialData={editingEntry} />
    </div>
  );
};
